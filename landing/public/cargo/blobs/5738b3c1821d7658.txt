use crate::modelo::Cliente;
use crate::modelo::PropsCliente;
use crate::valores::Email;
use crate::valores::Etiqueta;

pub struct MadreCliente {
    email: Option<Email>,
    nombre: Option<String>,
    etiquetas: Option<Vec<Etiqueta>>,
}

impl MadreCliente {
    pub fn new() -> Self {
        Self {
            email: None,
            nombre: None,
            etiquetas: None,
        }
    }

    pub fn con_email(mut self, email: Email) -> Self {
        self.email = Some(email);
        self
    }

    pub fn con_nombre(mut self, nombre: &str) -> Self {
        self.nombre = Some(nombre.to_string());
        self
    }

    pub fn con_etiquetas(mut self, etiquetas: Vec<Etiqueta>) -> Self {
        self.etiquetas = Some(etiquetas);
        self
    }

    pub fn con_nombre_vacio(mut self) -> Self {
        self.nombre = Some(String::new());
        self
    }

    pub fn construir(self) -> Cliente {
        Cliente::new(PropsCliente {
            email: self.email.unwrap_or_else(|| {
                Email::new("example".to_string()).expect("Email::new rechazó el valor de ejemplo")
            }),
            nombre: self.nombre.unwrap_or_else(|| "example".to_string()),
            etiquetas: self.etiquetas.unwrap_or_default(),
        })
        .expect("MadreCliente: Cliente::new rechazó los props")
    }

    pub fn construir_props(self) -> PropsCliente {
        PropsCliente {
            email: self.email.unwrap_or_else(|| {
                Email::new("example".to_string()).expect("Email::new rechazó el valor de ejemplo")
            }),
            nombre: self.nombre.unwrap_or_else(|| "example".to_string()),
            etiquetas: self.etiquetas.unwrap_or_default(),
        }
    }

    pub fn aleatorio() -> Cliente {
        Self::new().construir()
    }

    pub fn aleatorios(n: usize) -> Vec<Cliente> {
        (0..n).map(|_| Self::aleatorio()).collect()
    }
}

impl Default for MadreCliente {
    fn default() -> Self {
        Self::new()
    }
}
