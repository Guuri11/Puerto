use async_trait::async_trait;
use dominio_producto::errores::ErrorProducto;
use dominio_producto::modelo::IdProducto;
use dominio_producto::repositorio::RepositorioProducto;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsBorrarProducto {
    pub id: IdProducto,
}

#[async_trait]
pub trait BorrarProducto: Send + Sync {
    async fn ejecutar(&self, params: ParamsBorrarProducto) -> Result<(), ErrorProducto>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub BorrarProducto {}

        #[async_trait]
        impl BorrarProducto for BorrarProducto {
            async fn ejecutar(&self, params: ParamsBorrarProducto) -> Result<(), ErrorProducto>;
        }
    }
}

pub struct BorrarProductoImpl {
    pub repositorio: Arc<dyn RepositorioProducto>,
}

#[async_trait]
impl BorrarProducto for BorrarProductoImpl {
    async fn ejecutar(&self, params: ParamsBorrarProducto) -> Result<(), ErrorProducto> {
        tracing::info!(id = %params.id, "borrar_producto: inicio");
        let mut entidad = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "borrar_producto: fallo del repositorio"))?
            .ok_or_else(|| {
                let err = ErrorProducto::NoEncontrado;
                tracing::warn!(error = %err, "borrar_producto: rechazado");
                err
            })?;
        let ahora = chrono::Utc::now();
        entidad.borrado = true;
        entidad.borrado_en = Some(ahora);
        entidad.actualizado_en = ahora;
        self.repositorio.guardar(&entidad).await.inspect_err(
            |e| tracing::error!(error = %e, "borrar_producto: fallo del repositorio"),
        )?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_producto::{
        modelo::{IdProducto, Producto, PropsProducto},
        repositorio::dobles::MockRepositorioProducto,
    };

    #[tokio::test]
    async fn deberia_hacer_borrado_logico_de_producto_cuando_el_id_existe() {
        let entidad = Producto::new(PropsProducto {
            nombre: "example".to_string(),
            precio: 42,
            sku: "example".to_string(),
        })
        .unwrap();
        let id_entidad = entidad.id;
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        repositorio_doble.expect_guardar().returning(|_| Ok(()));
        let caso_de_uso = BorrarProductoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsBorrarProducto { id: id_entidad })
            .await;

        assert!(resultado.is_ok());
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_cuando_producto_no_existe() {
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let caso_de_uso = BorrarProductoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsBorrarProducto {
                id: IdProducto::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.no_encontrado");
    }
}
