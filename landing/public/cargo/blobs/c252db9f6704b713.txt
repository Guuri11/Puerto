use async_trait::async_trait;
use dominio_cliente::errores::ErrorCliente;
use dominio_cliente::modelo::{Cliente, PropsCliente};
use dominio_cliente::unit_of_work::UnitOfWorkFactoryCliente;
use dominio_cliente::valores::{Email, Etiqueta};
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsCrearCliente {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub email: String,
    pub nombre: String,
    pub etiquetas: Vec<String>,
}

#[async_trait]
pub trait CrearCliente: Send + Sync {
    async fn ejecutar(&self, params: ParamsCrearCliente) -> Result<Cliente, ErrorCliente>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub CrearCliente {}

        #[async_trait]
        impl CrearCliente for CrearCliente {
            async fn ejecutar(&self, params: ParamsCrearCliente) -> Result<Cliente, ErrorCliente>;
        }
    }
}

pub struct CrearClienteImpl {
    pub uow_factory: Arc<dyn UnitOfWorkFactoryCliente>,
}

#[async_trait]
impl CrearCliente for CrearClienteImpl {
    #[tracing::instrument(name = "crear_cliente", skip_all, err(level = "warn"))]
    async fn ejecutar(&self, params: ParamsCrearCliente) -> Result<Cliente, ErrorCliente> {
        let email = Email::new(params.email)?;
        let etiquetas: Vec<Etiqueta> = params
            .etiquetas
            .into_iter()
            .map(Etiqueta::new)
            .collect::<Result<Vec<_>, _>>()?;
        let (entidad, evento) = Cliente::crear(PropsCliente {
            email,
            nombre: params.nombre,
            etiquetas,
        })?;
        self.uow_factory.guardar(&entidad, evento).await?;
        Ok(entidad)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_cliente::pruebas::madres::MadreCliente;
    use dominio_cliente::unit_of_work::dobles::{FallarEn, UnitOfWorkFactoryGrabadora};
    use puerto_nucleo::ErrorRepositorio;
    use puerto_nucleo::EventoDominio;
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    #[tokio::test]
    async fn deberia_registrar_el_evento_cliente_creado_cuando_los_campos_son_validos() {
        // Preparar
        let props = MadreCliente::new().construir_props();
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = CrearClienteImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearCliente {
                actor: MadrePrincipal::aleatorio(),
                email: props.email.valor().to_string(),
                nombre: props.nombre,
                etiquetas: props
                    .etiquetas
                    .iter()
                    .map(|v| v.valor().to_string())
                    .collect::<Vec<_>>(),
            })
            .await;

        // Comprobar
        let entidad = match resultado {
            Ok(entidad) => entidad,
            Err(e) => panic!("la creación falló: {e}"),
        };
        assert_eq!(entidad.nombre, MadreCliente::new().construir_props().nombre);
        let grabado = uow_factory.grabado();
        assert!(grabado.confirmado);
        assert_eq!(grabado.guardados.len(), 1);
        assert_eq!(grabado.eventos.len(), 1);
        assert_eq!(grabado.eventos[0].tipo_evento(), "cliente_creado");
        assert_eq!(grabado.eventos[0].id_agregado(), entidad.id.valor());
    }

    #[tokio::test]
    async fn no_deberia_hacer_commit_cuando_el_almacen_rechaza_un_duplicado() {
        // Preparar
        let props = MadreCliente::new().construir_props();
        let uow_factory =
            UnitOfWorkFactoryGrabadora::fallando_en(FallarEn::Guardar, ErrorRepositorio::Duplicado);
        let caso_de_uso = CrearClienteImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearCliente {
                actor: MadrePrincipal::aleatorio(),
                email: props.email.valor().to_string(),
                nombre: props.nombre,
                etiquetas: props
                    .etiquetas
                    .iter()
                    .map(|v| v.valor().to_string())
                    .collect::<Vec<_>>(),
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.duplicado");
        let grabado = uow_factory.grabado();
        assert!(!grabado.confirmado);
        assert!(grabado.eventos.is_empty());
    }
