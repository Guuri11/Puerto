import { ProductError } from "../domain/product/errors";
import type { Product, ProductDraft } from "../domain/product/model";
import type { ProductRepository } from "../domain/product/repository";

/** Valid products for tests; pass `overrides` for the fields a test is about. */
export const ProductMother = {
  draft(overrides: Partial<ProductDraft> = {}): ProductDraft {
    return {
      name: "example",
      price: 42,
      sku: "example",
      ...overrides,
    };
  },

  random(overrides: Partial<Product> = {}): Product {
    return { id: crypto.randomUUID(), ...ProductMother.draft(), ...overrides };
  },
};

/** An in-memory `ProductRepository`. Override a method to make it fail. */
export function fakeProductRepository(seed: Product[] = []): ProductRepository {
  const store = new Map(seed.map((item) => [item.id, item]));
  const find = (id: string) => {
    const found = store.get(id);
    if (found === undefined) throw new ProductError("product.not_found");
    return found;
  };
  return {
    list: async ({ limit, offset }) => {
      const all = [...store.values()];
      return { items: all.slice(offset, offset + limit), total: all.length, limit, offset };
    },
    get: async (id) => find(id),
    create: async (draft) => {
      const created = { id: crypto.randomUUID(), ...draft };
      store.set(created.id, created);
      return created;
    },
    update: async (id, draft) => {
      const updated = { ...find(id), ...draft };
      store.set(id, updated);
      return updated;
    },
    remove: async (id) => {
      find(id);
      store.delete(id);
    },
  };
}
