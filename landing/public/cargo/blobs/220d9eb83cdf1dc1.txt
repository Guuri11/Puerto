import { describe, expect, test } from "bun:test";
import { fireEvent, screen } from "@testing-library/react";
import { apiFalsa } from "../../../pruebas/api-falsa";
import { coleccionCliente, MadreCliente } from "../../../pruebas/cliente";
import { renderizarPantalla } from "../../../pruebas/renderizar";
import { ContenedorListaCliente } from "./lista-cliente";

describe("ContenedorListaCliente", () => {
  test("debería mostrar una fila por cada cliente", async () => {
    // Preparar
    const semilla = [MadreCliente.aleatorio(), MadreCliente.aleatorio()];
    const api = apiFalsa([coleccionCliente(semilla)]);

    // Actuar
    renderizarPantalla(<ContenedorListaCliente />, { api });

    // Comprobar
    expect(await screen.findAllByText("Ver")).toHaveLength(2);
  });

  test("debería mostrar la página siguiente cuando se pulsa Siguiente", async () => {
    // Preparar
    const semilla = Array.from({ length: 25 }, () => MadreCliente.aleatorio());
    const api = apiFalsa([coleccionCliente(semilla)]);
    renderizarPantalla(<ContenedorListaCliente />, { api });
    expect(await screen.findAllByText("Ver")).toHaveLength(20);

    // Actuar
    fireEvent.click(screen.getByRole("button", { name: "Siguiente" }));

    // Comprobar
    expect(await screen.findByText("21–25 de 25")).toBeTruthy();
    expect(screen.getAllByText("Ver")).toHaveLength(5);
  });

  test("debería abrir la página que nombra la URL", async () => {
    // Preparar
    const semilla = Array.from({ length: 25 }, () => MadreCliente.aleatorio());
    const api = apiFalsa([coleccionCliente(semilla)]);
    const ruta = "/clientes?pagina=2";

    // Actuar
    renderizarPantalla(<ContenedorListaCliente />, { api }, ruta);

    // Comprobar
    expect(await screen.findByText("21–25 de 25")).toBeTruthy();
  });

  test("debería decirlo cuando no hay clientes", async () => {
    // Preparar
    const api = apiFalsa([coleccionCliente()]);

    // Actuar
    renderizarPantalla(<ContenedorListaCliente />, { api });

    // Comprobar
    expect(await screen.findByText("Todavía no hay clientes.")).toBeTruthy();
  });

  test("debería mostrar el código del error cuando la API no da la lista", async () => {
    // Preparar
    const fallos = { "GET /clientes": "cliente.repositorio" };
    const api = apiFalsa([coleccionCliente()], fallos);

    // Actuar
    renderizarPantalla(<ContenedorListaCliente />, { api });

    // Comprobar
    expect((await screen.findByRole("alert")).textContent).toBe("cliente.repositorio");
  });
});
