/** Lo que una pantalla muestra de un error de la API: bajo el input del que trata, o en la alerta. */
export interface ProblemaCliente {
  campo: string | null;
  texto: string;
}

/**
 * Los mensajes de los códigos de `errores.rs` del backend, generados de la misma lista. Un código que
 * no está aquí se muestra tal cual.
 */
const TEXTOS: Record<string, string> = {
  "cliente.email_invalido": "Email: no válido",
  "cliente.validacion.nombre_vacio": "Nombre: obligatorio",
  "cliente.etiqueta_invalido": "Etiquetas: no válido",
  "cliente.duplicado": "Estos valores ya están en uso",
  "cliente.no_encontrado": "Ya no existe",
  "cliente.repositorio": "No se ha podido guardar; inténtalo más tarde",
  "cliente.desconocido": "Algo ha ido mal",
};

/** El input del que trata el código de cada campo. */
const CAMPOS: Record<string, string> = {
  "cliente.email_invalido": "email",
  "cliente.validacion.nombre_vacio": "nombre",
  "cliente.etiqueta_invalido": "etiquetas",
};

export function aProblema(codigo: string): ProblemaCliente {
  return { campo: CAMPOS[codigo] ?? null, texto: TEXTOS[codigo] ?? codigo };
}
