import { PedidoError } from "./errors";

export interface Pedido {
  id: string;
  importe: number;
  estado: Estado;
  nota: string | null;
}

/** What a create or an update sends: a pedido without its id. */
export type PedidoDraft = Omit<Pedido, "id">;

export const estadoValues = ["Pendiente", "Confirmado", "Cancelado"] as const;

export type Estado = (typeof estadoValues)[number];

/** Narrows the API's `string` to a `Estado`, rejecting with `pedido.estado_invalido`. */
export function parseEstado(value: string): Estado {
  const found = estadoValues.find((candidate) => candidate === value);
  if (found === undefined) throw new PedidoError("pedido.estado_invalido");
  return found;
}
