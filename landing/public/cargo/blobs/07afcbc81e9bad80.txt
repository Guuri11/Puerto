use std::sync::Arc;

use crate::domain::order::value_objects::Amount;
use crate::domain::order::value_objects::Status;
use async_trait::async_trait;
use puerto_core::RepositoryError;

use crate::domain::order::{
    errors::OrderError,
    events::{OrderEvent, OrderSnapshot},
    model::{Order, OrderProps},
    unit_of_work::OrderUnitOfWorkFactory,
    use_cases::create_order::{CreateOrderParams, CreateOrderUseCaseTrait},
};

pub struct CreateOrderUseCaseImpl {
    pub uow_factory: Arc<dyn OrderUnitOfWorkFactory>,
}

#[async_trait]
impl CreateOrderUseCaseTrait for CreateOrderUseCaseImpl {
    async fn execute(&self, params: CreateOrderParams) -> Result<Order, OrderError> {
        tracing::info!("create_order: start");
        let failed =
            |e: &RepositoryError| tracing::error!(error = %e, "create_order: repository failed");
        let amount = Amount::new(params.amount)?;
        let status = params.status.parse::<Status>()?;
        let entity = Order::new(OrderProps {
            amount,
            status,
            note: params.note,
        })
        .inspect_err(|e| tracing::warn!(error = %e, "create_order: rejected"))?;
        let mut uow = self.uow_factory.begin().await.inspect_err(failed)?;
        uow.save(&entity).await.inspect_err(failed)?;
        uow.record(OrderEvent::Created(OrderSnapshot::from_entity(&entity)))
            .await
            .inspect_err(failed)?;
        uow.commit().await.inspect_err(failed)?;
        tracing::info!(id = %entity.id, "create_order: done");
        Ok(entity)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::order::unit_of_work::mocks::{FailAt, RecordingUnitOfWorkFactory};
    use puerto_core::DomainEvent;

    #[tokio::test]
    async fn should_record_order_created_when_fields_are_valid() {
        // Arrange
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = CreateOrderUseCaseImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(CreateOrderParams {
                amount: 1.5,
                status: "Pending".to_string(),
                note: None,
            })
            .await;

        // Assert
        let entity = match result {
            Ok(entity) => entity,
            Err(e) => panic!("create failed: {e}"),
        };
        assert_eq!(entity.status.as_str(), "Pending");
        let recorded = uow_factory.recorded();
        assert!(recorded.committed);
        assert_eq!(recorded.saved.len(), 1);
        assert_eq!(recorded.events.len(), 1);
        assert_eq!(recorded.events[0].event_type(), "order_created");
        assert_eq!(recorded.events[0].aggregate_id(), entity.id.value());
    }

    #[tokio::test]
    async fn should_not_commit_when_the_store_rejects_a_duplicate() {
        // Arrange
        let uow_factory =
            RecordingUnitOfWorkFactory::failing_at(FailAt::Save, RepositoryError::Duplicate);
        let use_case = CreateOrderUseCaseImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(CreateOrderParams {
                amount: 1.5,
                status: "Pending".to_string(),
                note: None,
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.duplicate");
        let recorded = uow_factory.recorded();
        assert!(!recorded.committed);
        assert!(recorded.events.is_empty());
    }

    #[tokio::test]
    async fn should_return_repository_error_when_the_commit_fails() {
        // Arrange
        let uow_factory =
            RecordingUnitOfWorkFactory::failing_at(FailAt::Commit, RepositoryError::Unavailable);
        let use_case = CreateOrderUseCaseImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(CreateOrderParams {
                amount: 1.5,
                status: "Pending".to_string(),
                note: None,
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.repository_error");
        assert!(!uow_factory.recorded().committed);
    }

    #[tokio::test]
    async fn should_return_error_and_record_nothing_when_status_is_empty() {
        // Arrange
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = CreateOrderUseCaseImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(CreateOrderParams {
                amount: 1.5,
                status: "".to_string(),
                note: None,
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.invalid_status");
        assert!(uow_factory.recorded().saved.is_empty());
    }
}
