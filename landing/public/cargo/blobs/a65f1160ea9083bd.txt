use async_trait::async_trait;
use dominio_producto::errores::ErrorProducto;
use dominio_producto::modelo::{IdProducto, Producto};
use dominio_producto::repositorio::RepositorioProducto;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsObtenerProducto {
    pub id: IdProducto,
}

#[async_trait]
pub trait ObtenerProducto: Send + Sync {
    async fn ejecutar(&self, params: ParamsObtenerProducto) -> Result<Producto, ErrorProducto>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub ObtenerProducto {}

        #[async_trait]
        impl ObtenerProducto for ObtenerProducto {
            async fn ejecutar(&self, params: ParamsObtenerProducto) -> Result<Producto, ErrorProducto>;
        }
    }
}

pub struct ObtenerProductoImpl {
    pub repositorio: Arc<dyn RepositorioProducto>,
}

#[async_trait]
impl ObtenerProducto for ObtenerProductoImpl {
    async fn ejecutar(&self, params: ParamsObtenerProducto) -> Result<Producto, ErrorProducto> {
        tracing::info!(id = %params.id, "obtener_producto: inicio");
        let resultado = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(
                |e| tracing::error!(error = %e, "obtener_producto: fallo del repositorio"),
            )?;
        resultado.ok_or_else(|| {
            let err = ErrorProducto::NoEncontrado;
            tracing::warn!(error = %err, "obtener_producto: rechazado");
            err
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_producto::{
        modelo::IdProducto, pruebas::madres::MadreProducto,
        repositorio::dobles::MockRepositorioProducto,
    };
    use puerto_nucleo::ErrorRepositorio;

    #[tokio::test]
    async fn deberia_devolver_producto_cuando_el_id_existe() {
        let entidad = MadreProducto::aleatorio();
        let id_entidad = entidad.id;
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        let caso_de_uso = ObtenerProductoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerProducto { id: id_entidad })
            .await;

        assert_eq!(resultado.unwrap().id, id_entidad);
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_cuando_el_id_no_existe() {
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let caso_de_uso = ObtenerProductoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerProducto {
                id: IdProducto::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.no_encontrado");
    }

    #[tokio::test]
    async fn deberia_devolver_error_de_repositorio_cuando_el_almacen_no_esta_disponible() {
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Err(ErrorRepositorio::NoDisponible));
        let caso_de_uso = ObtenerProductoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerProducto {
                id: IdProducto::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.repositorio");
    }
}
