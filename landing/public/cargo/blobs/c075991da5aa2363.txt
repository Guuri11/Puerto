use async_trait::async_trait;
use sqlx::PgPool;

use business::domain::product::{
    model::{Product, ProductId},
    repository::ProductRepositoryTrait,
};
use puerto_core::{Page, PageRequest, RepositoryError};

use super::entity::ProductDb;
use crate::db::map_sqlx_error;

pub struct PgProductRepository {
    pub pool: PgPool,
}

impl PgProductRepository {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }
}

/// A row as the domain's entity. A row the domain rejects is stored data gone bad, not the
/// caller's fault: it is logged with the domain's reason and reported as `InvalidData`.
fn to_domain(row: ProductDb) -> Result<Product, RepositoryError> {
    Product::try_from(row).map_err(|e| {
        tracing::error!(error = %e, "stored row is not a valid product");
        RepositoryError::InvalidData
    })
}

#[async_trait]
impl ProductRepositoryTrait for PgProductRepository {
    async fn find_all(&self, page: PageRequest) -> Result<Page<Product>, RepositoryError> {
        tracing::debug!(limit = page.limit(), offset = page.offset(), "find_all");
        let rows = sqlx::query_as!(
            ProductDb,
            "SELECT id, created_at, updated_at, deleted, deleted_at, name, price, sku
             FROM products WHERE deleted = false
             ORDER BY created_at DESC, id LIMIT $1 OFFSET $2",
            i64::from(page.limit()),
            i64::from(page.offset())
        )
        .fetch_all(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "find_all failed");
            map_sqlx_error(&e)
        })?;
        let total = sqlx::query_scalar!(
            r#"SELECT COUNT(*) AS "total!" FROM products WHERE deleted = false"#
        )
        .fetch_one(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "find_all count failed");
            map_sqlx_error(&e)
        })?;

        let items = rows
            .into_iter()
            .map(to_domain)
            .collect::<Result<Vec<_>, _>>()?;
        // COUNT(*) is never negative.
        let total = u64::try_from(total).unwrap_or_default();
        Ok(Page::new(items, total, page))
    }

    async fn find_by_id(&self, id: ProductId) -> Result<Option<Product>, RepositoryError> {
        tracing::debug!(%id, "find_by_id");
        let row = sqlx::query_as!(
            ProductDb,
            "SELECT id, created_at, updated_at, deleted, deleted_at, name, price, sku
             FROM products WHERE id = $1 AND deleted = false",
            id.value()
        )
        .fetch_optional(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "find_by_id failed");
            map_sqlx_error(&e)
        })?;

        row.map(to_domain).transpose()
    }
}

/// `pub(crate)`: an entity that references Product seeds its rows with this module's `seed`.
#[cfg(test)]
pub(crate) mod integration_tests {
    use super::*;
    use business::domain::product::model::ProductProps;
    use business::domain::product::unit_of_work::ProductUnitOfWorkFactory;

    use crate::product::unit_of_work::PgProductUnitOfWorkFactory;

    use uuid::Uuid;

    fn test_repo(pool: PgPool) -> PgProductRepository {
        PgProductRepository::new(pool)
    }

    /// The repository only reads: rows are written through the unit of work.
    async fn store(pool: &PgPool, entity: &Product) {
        let mut uow = PgProductUnitOfWorkFactory::new(pool.clone())
            .begin()
            .await
            .unwrap();
        uow.save(entity).await.unwrap();
        uow.commit().await.unwrap();
    }

    /// A valid product, not stored.
    fn build() -> Product {
        Product::new(ProductProps {
            name: "example".to_string(),
            price: 42,
            sku: Uuid::new_v4().to_string(),
        })
        .unwrap()
    }

    /// A stored product.
    pub(crate) async fn seed(pool: &PgPool) -> Product {
        let entity = build();
        store(pool, &entity).await;
        entity
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_persist_and_retrieve_by_id(pool: PgPool) {
        // Arrange
        let entity = seed(&pool).await;

        // Act
        let found = test_repo(pool)
            .find_by_id(entity.id)
            .await
            .unwrap()
            .unwrap();

        // Assert
        assert_eq!(found.id, entity.id);
        assert_eq!(found.name, entity.name);
        assert_eq!(found.price, entity.price);
        assert_eq!(found.sku, entity.sku);
        assert!(!found.deleted);
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_return_none_for_nonexistent_id(pool: PgPool) {
        // Act
        let result = test_repo(pool)
            .find_by_id(ProductId::generate())
            .await
            .unwrap();

        // Assert
        assert!(result.is_none());
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_list_all_products_excluding_deleted(pool: PgPool) {
        // Arrange
        seed(&pool).await;
        let mut deleted = seed(&pool).await;
        deleted.deleted = true;
        deleted.deleted_at = Some(deleted.updated_at);
        store(&pool, &deleted).await;

        // Act
        let page = test_repo(pool)
            .find_all(PageRequest::default())
            .await
            .unwrap();

        // Assert
        assert_eq!(page.items.len(), 1);
        assert_eq!(page.total, 1);
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_return_the_requested_page_of_a_larger_list(pool: PgPool) {
        // Arrange
        for _ in 0..3 {
            seed(&pool).await;
        }

        // Act
        let page = test_repo(pool)
            .find_all(PageRequest::new(Some(2), Some(2)))
            .await
            .unwrap();

        // Assert
        assert_eq!(page.items.len(), 1);
        assert_eq!(page.total, 3);
        assert_eq!((page.limit, page.offset), (2, 2));
    }
}
