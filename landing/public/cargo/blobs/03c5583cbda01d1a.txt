use puerto_nucleo::ErrorRepositorio;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum ErrorCliente {
    #[error("cliente.validacion.{0}")]
    Validacion(String),
    #[error("cliente.no_encontrado")]
    NoEncontrado,
    #[error("cliente.duplicado")]
    Duplicado,
    #[error("cliente.repositorio")]
    Repositorio,
    #[error("cliente.desconocido")]
    Desconocido,
    #[error("cliente.email_invalido")]
    EmailInvalido,
    #[error("cliente.etiqueta_invalido")]
    EtiquetaInvalido,
}

impl From<ErrorRepositorio> for ErrorCliente {
    /// Cada fallo que nombra `puerto-nucleo` tiene su brazo. `ErrorRepositorio` es
    /// `#[non_exhaustive]`: uno que añada una versión posterior es un problema del almacén (500)
    /// hasta que tenga aquí un significado.
    fn from(err: ErrorRepositorio) -> Self {
        match err {
            ErrorRepositorio::Duplicado => Self::Duplicado,
            ErrorRepositorio::DatosInvalidos | ErrorRepositorio::NoDisponible => Self::Repositorio,
            _ => Self::Repositorio,
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn deberia_convertir_un_duplicado_en_el_duplicado_de_la_entidad() {
        let err = ErrorCliente::from(ErrorRepositorio::Duplicado);
        assert_eq!(err.to_string(), "cliente.duplicado");
    }

    #[test]
    fn deberia_convertir_datos_invalidos_en_un_error_de_repositorio() {
        let err = ErrorCliente::from(ErrorRepositorio::DatosInvalidos);
        assert_eq!(err.to_string(), "cliente.repositorio");
    }

    #[test]
    fn deberia_convertir_un_almacen_no_disponible_en_un_error_de_repositorio() {
        let err = ErrorCliente::from(ErrorRepositorio::NoDisponible);
        assert_eq!(err.to_string(), "cliente.repositorio");
    }
}
