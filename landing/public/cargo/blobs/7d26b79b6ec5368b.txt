use async_trait::async_trait;
use puerto_core::auth::Principal;

use crate::domain::customer::{errors::CustomerError, model::Customer};

#[derive(Debug, Clone)]
pub struct CreateCustomerParams {
    /// Who asks (ADR-0021). A rule that depends on the data ("only the owner edits") decides on
    /// it here and answers `Forbidden`.
    pub actor: Principal,
    pub email: String,
    pub name: String,
    pub tags: Vec<String>,
}

#[async_trait]
pub trait CreateCustomerUseCaseTrait: Send + Sync {
    async fn execute(&self, params: CreateCustomerParams) -> Result<Customer, CustomerError>;
}

#[cfg(any(test, feature = "test-utils"))]
pub mod mocks {
    use mockall::mock;

    use super::*;

    mock! {
        pub CreateCustomerUseCase {}

        #[async_trait]
        impl CreateCustomerUseCaseTrait for CreateCustomerUseCase {
            async fn execute(&self, params: CreateCustomerParams) -> Result<Customer, CustomerError>;
        }
    }
}
