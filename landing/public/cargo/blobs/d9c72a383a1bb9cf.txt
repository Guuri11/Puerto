use async_trait::async_trait;
use dominio_pedido::errores::ErrorPedido;
use dominio_pedido::eventos::{EventoPedido, SnapshotPedido};
use dominio_pedido::modelo::{IdPedido, Pedido, PropsPedido};
use dominio_pedido::repositorio::RepositorioPedido;
use dominio_pedido::unit_of_work::UnitOfWorkFactoryPedido;
use dominio_pedido::valores::{Estado, Importe};
use puerto_nucleo::ErrorRepositorio;
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsActualizarPedido {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub id: IdPedido,
    pub importe: f64,
    pub estado: String,
    pub nota: Option<String>,
}

#[async_trait]
pub trait ActualizarPedido: Send + Sync {
    async fn ejecutar(&self, params: ParamsActualizarPedido) -> Result<Pedido, ErrorPedido>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub ActualizarPedido {}

        #[async_trait]
        impl ActualizarPedido for ActualizarPedido {
            async fn ejecutar(&self, params: ParamsActualizarPedido) -> Result<Pedido, ErrorPedido>;
        }
    }
}

pub struct ActualizarPedidoImpl {
    pub repositorio: Arc<dyn RepositorioPedido>,
    pub uow_factory: Arc<dyn UnitOfWorkFactoryPedido>,
}

#[async_trait]
impl ActualizarPedido for ActualizarPedidoImpl {
    async fn ejecutar(&self, params: ParamsActualizarPedido) -> Result<Pedido, ErrorPedido> {
        tracing::info!(id = %params.id, "actualizar_pedido: inicio");
        let fallido = |e: &ErrorRepositorio| tracing::error!(error = %e, "actualizar_pedido: fallo del repositorio");
        let mut entidad = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(fallido)?
            .ok_or_else(|| {
                let err = ErrorPedido::NoEncontrado;
                tracing::warn!(error = %err, "actualizar_pedido: rechazado");
                err
            })?;
        let importe = Importe::new(params.importe)?;
        let estado = params.estado.parse::<Estado>()?;
        entidad
            .actualizar(PropsPedido {
                importe,
                estado,
                nota: params.nota,
            })
            .inspect_err(|e| tracing::warn!(error = %e, "actualizar_pedido: rechazado"))?;
        let mut uow = self.uow_factory.begin().await.inspect_err(fallido)?;
        uow.guardar(&entidad).await.inspect_err(fallido)?;
        uow.registrar(EventoPedido::Actualizado(SnapshotPedido::from_entidad(
            &entidad,
        )))
        .await
        .inspect_err(fallido)?;
        uow.commit().await.inspect_err(fallido)?;
        Ok(entidad)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_pedido::{
        modelo::IdPedido, pruebas::madres::MadrePedido, repositorio::dobles::MockRepositorioPedido,
        unit_of_work::dobles::UnitOfWorkFactoryGrabadora,
    };
    use puerto_nucleo::EventoDominio;
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    fn repositorio_con(entidad: Pedido) -> MockRepositorioPedido {
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        repositorio_doble
    }

    #[tokio::test]
    async fn deberia_registrar_el_evento_pedido_actualizado_cuando_los_parametros_son_validos() {
        // Preparar
        let props = MadrePedido::new().construir_props();
        let entidad = MadrePedido::aleatorio();
        let id_entidad = entidad.id;
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = ActualizarPedidoImpl {
            repositorio: Arc::new(repositorio_con(entidad)),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsActualizarPedido {
                actor: MadrePrincipal::aleatorio(),
                id: id_entidad,
                importe: props.importe.valor(),
                estado: props.estado.as_str().to_string(),
                nota: props.nota,
            })
            .await;

        // Comprobar
        assert!(resultado.is_ok());
        let grabado = uow_factory.grabado();
        assert!(grabado.confirmado);
        assert_eq!(grabado.eventos.len(), 1);
        assert_eq!(grabado.eventos[0].tipo_evento(), "pedido_actualizado");
        assert_eq!(grabado.eventos[0].id_agregado(), id_entidad.valor());
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_sin_registrar_nada_cuando_pedido_no_existe() {
        // Preparar
        let props = MadrePedido::new().construir_props();
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = ActualizarPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsActualizarPedido {
                actor: MadrePrincipal::aleatorio(),
                id: IdPedido::generar(),
                importe: props.importe.valor(),
                estado: props.estado.as_str().to_string(),
                nota: props.nota,
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "pedido.no_encontrado");
        let grabado = uow_factory.grabado();
        assert!(grabado.guardados.is_empty());
        assert!(grabado.eventos.is_empty());
    }
}
