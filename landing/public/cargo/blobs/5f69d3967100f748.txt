use chrono::{DateTime, Utc};
use uuid::Uuid;

use crate::errores::ErrorProducto;
use crate::eventos::{EventoProducto, SnapshotProducto};

/// El id de cada `Producto`: un `Uuid` que no se puede pasar donde se espera el id de otra
/// entidad.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct IdProducto(Uuid);

impl IdProducto {
    /// Un id nuevo y aleatorio — para una entidad que se está creando.
    pub fn generar() -> Self {
        Self(Uuid::new_v4())
    }

    /// Un id leído de fuera (un parámetro de ruta, una fila) — sin validar: cualquier `Uuid` es un
    /// id.
    pub fn from_uuid(valor: Uuid) -> Self {
        Self(valor)
    }

    pub fn valor(&self) -> Uuid {
        self.0
    }
}

impl std::fmt::Display for IdProducto {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        self.0.fmt(f)
    }
}

#[derive(Debug, Clone)]
pub struct PropsProducto {
    pub nombre: String,
    pub precio: i64,
    pub sku: String,
}

#[derive(Debug, Clone)]
pub struct Producto {
    pub id: IdProducto,
    pub creado_en: DateTime<Utc>,
    pub actualizado_en: DateTime<Utc>,
    pub borrado: bool,
    pub borrado_en: Option<DateTime<Utc>>,
    pub nombre: String,
    pub precio: i64,
    pub sku: String,
}

impl Producto {
    pub fn new(props: PropsProducto) -> Result<Self, ErrorProducto> {
        Self::validar(&props)?;
        let ahora = chrono::Utc::now();
        Ok(Self {
            id: IdProducto::generar(),
            creado_en: ahora,
            actualizado_en: ahora,
            borrado: false,
            borrado_en: None,
            nombre: props.nombre,
            precio: props.precio,
            sku: props.sku,
        })
    }

    /// Crea producto y emite `Creado` (ADR-0026 de Puerto). `new` construye sin emitir nada:
    /// lo usan la madre y los tests.
    pub fn crear(props: PropsProducto) -> Result<(Self, EventoProducto), ErrorProducto> {
        let entidad = Self::new(props)?;
        let evento = EventoProducto::Creado(SnapshotProducto::from_entidad(&entidad));
        Ok((entidad, evento))
    }

    /// Cambia los campos propios, con las mismas reglas que `new`, y emite `Actualizado`.
    pub fn actualizar(&mut self, props: PropsProducto) -> Result<EventoProducto, ErrorProducto> {
        Self::validar(&props)?;
        self.nombre = props.nombre;
        self.precio = props.precio;
        self.sku = props.sku;
        self.actualizado_en = chrono::Utc::now();
        Ok(EventoProducto::Actualizado(SnapshotProducto::from_entidad(
            self,
        )))
    }

    /// El borrado lógico: la fila se queda, marcada como borrada, y emite `Borrado`.
    pub fn borrar(&mut self) -> EventoProducto {
        let ahora = chrono::Utc::now();
        self.borrado = true;
        self.borrado_en = Some(ahora);
        self.actualizado_en = ahora;
        EventoProducto::Borrado(SnapshotProducto::from_entidad(self))
    }

    pub fn from_repositorio(datos: Producto) -> Self {
        datos
    }

    /// Las reglas de los campos propios: las comprueban `new` y `actualizar`, así que una regla nueva
    /// se escribe solo aquí.
    fn validar(props: &PropsProducto) -> Result<(), ErrorProducto> {
        if props.nombre.trim().is_empty() {
            return Err(ErrorProducto::Validacion("nombre_vacio".into()));
        }
        if props.sku.trim().is_empty() {
            return Err(ErrorProducto::Validacion("sku_vacio".into()));
        }
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::pruebas::madres::MadreProducto;
    use puerto_nucleo::EventoDominio;

    #[test]
    fn deberia_crear_producto_cuando_los_campos_son_validos() {
        let resultado = Producto::new(MadreProducto::new().construir_props());
        assert_eq!(
            resultado.unwrap().nombre,
            MadreProducto::new().construir_props().nombre
        );
    }

    #[test]
    fn deberia_rechazar_producto_cuando_nombre_es_una_cadena_vacia() {
        let resultado = Producto::new(MadreProducto::new().con_nombre_vacio().construir_props());
        assert!(resultado.is_err());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "producto.validacion.nombre_vacio"
        );
    }

    #[test]
    fn deberia_rechazar_producto_cuando_nombre_solo_tiene_espacios() {
        let resultado = Producto::new(MadreProducto::new().con_nombre("   ").construir_props());
        assert!(resultado.is_err());
    }

    #[test]
    fn deberia_rechazar_producto_cuando_sku_es_una_cadena_vacia() {
        let resultado = Producto::new(MadreProducto::new().con_sku_vacio().construir_props());
        assert!(resultado.is_err());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "producto.validacion.sku_vacio"
        );
    }

    #[test]
    fn deberia_rechazar_producto_cuando_sku_solo_tiene_espacios() {
        let resultado = Producto::new(MadreProducto::new().con_sku("   ").construir_props());
        assert!(resultado.is_err());
    }

    #[test]
    fn deberia_rechazar_actualizar_producto_cuando_nombre_es_una_cadena_vacia() {
        let mut entidad = MadreProducto::aleatorio();
        let resultado =
            entidad.actualizar(MadreProducto::new().con_nombre_vacio().construir_props());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "producto.validacion.nombre_vacio"
        );
    }

    #[test]
    fn deberia_emitir_producto_creado_cuando_se_crea() {
        // Preparar
        let props = MadreProducto::new().construir_props();

        // Actuar
        let resultado = Producto::crear(props);

        // Comprobar
        let (entidad, evento) = match resultado {
            Ok(creado) => creado,
            Err(e) => panic!("la creación falló: {e}"),
        };
        assert_eq!(evento.tipo_evento(), "producto_creado");
        assert_eq!(evento.id_agregado(), entidad.id.valor());
    }

    #[test]
    fn deberia_emitir_producto_actualizado_cuando_se_actualiza() {
        // Preparar
        let mut entidad = MadreProducto::aleatorio();

        // Actuar
        let resultado = entidad.actualizar(MadreProducto::new().construir_props());

        // Comprobar
        let evento = match resultado {
            Ok(evento) => evento,
            Err(e) => panic!("la actualización falló: {e}"),
        };
        assert_eq!(evento.tipo_evento(), "producto_actualizado");
        assert_eq!(evento.id_agregado(), entidad.id.valor());
    }

    #[test]
    fn deberia_marcar_producto_como_borrado_cuando_se_borra() {
        // Preparar
        let mut entidad = MadreProducto::aleatorio();

        // Actuar
        let evento = entidad.borrar();

        // Comprobar
        assert!(entidad.borrado);
        assert!(entidad.borrado_en.is_some());
        assert_eq!(evento.tipo_evento(), "producto_borrado");
        assert_eq!(evento.id_agregado(), entidad.id.valor());
    }
}
