import { describe, expect, test } from "bun:test";
import { fireEvent, screen } from "@testing-library/react";
import { ProductError } from "../../../../domain/product/errors";
import { fakeProductRepository, ProductMother } from "../../../../test-support/product";
import { renderScreen } from "../../../../test-support/render";
import { ProductListContainer } from "./product-list-container";

describe("ProductListContainer", () => {
  test("should show a row per product when there are some", async () => {
    // Arrange
    const seed = [ProductMother.random(), ProductMother.random()];
    const productRepository = fakeProductRepository(seed);

    // Act
    renderScreen(<ProductListContainer />, { productRepository });

    // Assert
    expect(await screen.findAllByText("View")).toHaveLength(2);
  });

  test("should show the next page when Next is pressed", async () => {
    // Arrange
    const seed = Array.from({ length: 25 }, () => ProductMother.random());
    const productRepository = fakeProductRepository(seed);
    renderScreen(<ProductListContainer />, { productRepository });
    expect(await screen.findAllByText("View")).toHaveLength(20);

    // Act
    fireEvent.click(screen.getByRole("button", { name: "Next" }));

    // Assert
    expect(await screen.findByText("21–25 of 25")).toBeTruthy();
    expect(screen.getAllByText("View")).toHaveLength(5);
  });

  test("should open the page the URL names", async () => {
    // Arrange
    const seed = Array.from({ length: 25 }, () => ProductMother.random());
    const productRepository = fakeProductRepository(seed);
    const path = "/products?page=2";

    // Act
    renderScreen(<ProductListContainer />, { productRepository }, path);

    // Assert
    expect(await screen.findByText("21–25 of 25")).toBeTruthy();
  });

  test("should say so when there is no product", async () => {
    // Arrange
    const productRepository = fakeProductRepository();

    // Act
    renderScreen(<ProductListContainer />, { productRepository });

    // Assert
    expect(await screen.findByText("No products yet.")).toBeTruthy();
  });

  test("should show the error code when the products cannot be listed", async () => {
    // Arrange
    const productRepository = {
      ...fakeProductRepository(),
      list: () => Promise.reject(new ProductError("product.repository_error")),
    };

    // Act
    renderScreen(<ProductListContainer />, { productRepository });

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe("product.repository_error");
  });
});
