use async_trait::async_trait;
use dominio_producto::errores::ErrorProducto;
use dominio_producto::eventos::{EventoProducto, SnapshotProducto};
use dominio_producto::modelo::{IdProducto, Producto};
use dominio_producto::repositorio::RepositorioProducto;
use dominio_producto::unit_of_work::UnitOfWorkFactoryProducto;
use puerto_nucleo::ErrorRepositorio;
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsActualizarProducto {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub id: IdProducto,
    pub nombre: String,
    pub precio: i64,
    pub sku: String,
}

#[async_trait]
pub trait ActualizarProducto: Send + Sync {
    async fn ejecutar(&self, params: ParamsActualizarProducto) -> Result<Producto, ErrorProducto>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub ActualizarProducto {}

        #[async_trait]
        impl ActualizarProducto for ActualizarProducto {
            async fn ejecutar(&self, params: ParamsActualizarProducto) -> Result<Producto, ErrorProducto>;
        }
    }
}

pub struct ActualizarProductoImpl {
    pub repositorio: Arc<dyn RepositorioProducto>,
    pub uow_factory: Arc<dyn UnitOfWorkFactoryProducto>,
}

#[async_trait]
impl ActualizarProducto for ActualizarProductoImpl {
    async fn ejecutar(&self, params: ParamsActualizarProducto) -> Result<Producto, ErrorProducto> {
        tracing::info!(id = %params.id, "actualizar_producto: inicio");
        let fallido = |e: &ErrorRepositorio| tracing::error!(error = %e, "actualizar_producto: fallo del repositorio");
        let mut entidad = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(fallido)?
            .ok_or_else(|| {
                let err = ErrorProducto::NoEncontrado;
                tracing::warn!(error = %err, "actualizar_producto: rechazado");
                err
            })?;
        if params.nombre.trim().is_empty() {
            let err = ErrorProducto::Validacion("nombre_vacio".into());
            tracing::warn!(error = %err, "actualizar_producto: rechazado");
            return Err(err);
        }
        if params.sku.trim().is_empty() {
            let err = ErrorProducto::Validacion("sku_vacio".into());
            tracing::warn!(error = %err, "actualizar_producto: rechazado");
            return Err(err);
        }
        entidad.nombre = params.nombre;
        entidad.precio = params.precio;
        entidad.sku = params.sku;
        entidad.actualizado_en = chrono::Utc::now();
        let mut uow = self.uow_factory.begin().await.inspect_err(fallido)?;
        uow.guardar(&entidad).await.inspect_err(fallido)?;
        uow.registrar(EventoProducto::Actualizado(SnapshotProducto::from_entidad(
            &entidad,
        )))
        .await
        .inspect_err(fallido)?;
        uow.commit().await.inspect_err(fallido)?;
        Ok(entidad)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_producto::{
        modelo::{IdProducto, PropsProducto},
        repositorio::dobles::MockRepositorioProducto,
        unit_of_work::dobles::UnitOfWorkFactoryGrabadora,
    };
    use puerto_nucleo::EventoDominio;
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    fn repositorio_con(entidad: Producto) -> MockRepositorioProducto {
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        repositorio_doble
    }

    #[tokio::test]
    async fn deberia_registrar_el_evento_producto_actualizado_cuando_los_parametros_son_validos() {
        // Preparar
        let entidad = Producto::new(PropsProducto {
            nombre: "original".to_string(),
            precio: 42,
            sku: "original".to_string(),
        })
        .unwrap();
        let id_entidad = entidad.id;
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = ActualizarProductoImpl {
            repositorio: Arc::new(repositorio_con(entidad)),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsActualizarProducto {
                actor: MadrePrincipal::aleatorio(),
                id: id_entidad,
                nombre: "updated".to_string(),
                precio: 42,
                sku: "updated".to_string(),
            })
            .await;

        // Comprobar
        assert_eq!(resultado.unwrap().nombre, "updated");
        let grabado = uow_factory.grabado();
        assert!(grabado.confirmado);
        assert_eq!(grabado.eventos.len(), 1);
        assert_eq!(grabado.eventos[0].tipo_evento(), "producto_actualizado");
        assert_eq!(grabado.eventos[0].id_agregado(), id_entidad.valor());
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_sin_registrar_nada_cuando_producto_no_existe() {
        // Preparar
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = ActualizarProductoImpl {
            repositorio: Arc::new(repositorio_doble),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsActualizarProducto {
                actor: MadrePrincipal::aleatorio(),
                id: IdProducto::generar(),
                nombre: "new".to_string(),
                precio: 42,
                sku: "new".to_string(),
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.no_encontrado");
        let grabado = uow_factory.grabado();
        assert!(grabado.guardados.is_empty());
        assert!(grabado.eventos.is_empty());
    }
