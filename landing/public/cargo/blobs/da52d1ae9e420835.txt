import { Link, useLocation, useSearch } from "wouter";
import { PAGE_SIZE } from "../../../../domain/shared/page";
import { useDependencies } from "../../../core/dependencies";
import { useAsync } from "../../../core/hooks/use-async";
import { EmptyState } from "../../../shared/ui/empty-state";
import { ErrorAlert } from "../../../shared/ui/error-alert";
import { offsetFromSearch, Pagination, pageSearch } from "../../../shared/ui/pagination";
import { ClienteListView } from "../views/cliente-list-view";

/** The list, one page at a time; the page is in the URL (`?page=2`), so reload and back keep it. */
export function ClienteListContainer() {
  const { clienteRepository } = useDependencies();
  const [location, navigate] = useLocation();
  const offset = offsetFromSearch(useSearch(), PAGE_SIZE);
  const goTo = (next: number) => navigate(location + pageSearch(next, PAGE_SIZE));
  const clientes = useAsync(
    () => clienteRepository.list({ limit: PAGE_SIZE, offset }),
    [clienteRepository, offset],
  );

  return (
    <section className="space-y-4">
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-bold">Clientes</h1>
        <Link href="/clientes/new" className="btn btn-primary">
          New cliente
        </Link>
      </div>
      {clientes.status === "loading" && <span className="loading loading-spinner" />}
      {clientes.status === "error" && <ErrorAlert error={clientes.error} />}
      {clientes.status === "ok" &&
        (clientes.data.total === 0 ? (
          <EmptyState message="No clientes yet." />
        ) : (
          <>
            <ClienteListView clientes={clientes.data.items} />
            <Pagination {...clientes.data} onOffsetChange={goTo} />
          </>
        ))}
    </section>
  );
}
