use async_trait::async_trait;
use puerto_nucleo::{ErrorRepositorio, Pagina, PeticionPagina};

use crate::modelo::{IdPedido, Pedido};

#[async_trait]
pub trait RepositorioPedido: Send + Sync {
    async fn buscar_todos(
        &self,
        pagina: PeticionPagina,
    ) -> Result<Pagina<Pedido>, ErrorRepositorio>;
    async fn buscar_por_id(&self, id: IdPedido) -> Result<Option<Pedido>, ErrorRepositorio>;
    // Sin `guardar`: las escrituras pasan por `UnitOfWorkFactoryPedido`, que registra sus
    // eventos.
}

        }
