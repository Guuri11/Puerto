use async_trait::async_trait;

use crate::domain::order::{
    errors::OrderError,
    model::{Order, OrderId},
};

#[derive(Debug, Clone)]
pub struct GetOrderParams {
    pub id: OrderId,
}

#[async_trait]
pub trait GetOrderUseCaseTrait: Send + Sync {
    async fn execute(&self, params: GetOrderParams) -> Result<Order, OrderError>;
}

    }
