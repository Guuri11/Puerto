import { describe, expect, test } from "bun:test";
import { fireEvent, screen } from "@testing-library/react";
import { PedidoError } from "../../../../domain/pedido/errors";
import { fakePedidoRepository, PedidoMother } from "../../../../test-support/pedido";
import { renderScreen } from "../../../../test-support/render";
import { PedidoListContainer } from "./pedido-list-container";

describe("PedidoListContainer", () => {
  test("should show a row per pedido when there are some", async () => {
    // Arrange
    const seed = [PedidoMother.random(), PedidoMother.random()];
    const pedidoRepository = fakePedidoRepository(seed);

    // Act
    renderScreen(<PedidoListContainer />, { pedidoRepository });

    // Assert
    expect(await screen.findAllByText("View")).toHaveLength(2);
  });

  test("should show the next page when Next is pressed", async () => {
    // Arrange
    const seed = Array.from({ length: 25 }, () => PedidoMother.random());
    const pedidoRepository = fakePedidoRepository(seed);
    renderScreen(<PedidoListContainer />, { pedidoRepository });
    expect(await screen.findAllByText("View")).toHaveLength(20);

    // Act
    fireEvent.click(screen.getByRole("button", { name: "Next" }));

    // Assert
    expect(await screen.findByText("21–25 of 25")).toBeTruthy();
    expect(screen.getAllByText("View")).toHaveLength(5);
  });

  test("should open the page the URL names", async () => {
    // Arrange
    const seed = Array.from({ length: 25 }, () => PedidoMother.random());
    const pedidoRepository = fakePedidoRepository(seed);
    const path = "/pedidos?page=2";

    // Act
    renderScreen(<PedidoListContainer />, { pedidoRepository }, path);

    // Assert
    expect(await screen.findByText("21–25 of 25")).toBeTruthy();
  });

  test("should say so when there is no pedido", async () => {
    // Arrange
    const pedidoRepository = fakePedidoRepository();

    // Act
    renderScreen(<PedidoListContainer />, { pedidoRepository });

    // Assert
    expect(await screen.findByText("No pedidos yet.")).toBeTruthy();
  });

  test("should show the error code when the pedidos cannot be listed", async () => {
    // Arrange
    const pedidoRepository = {
      ...fakePedidoRepository(),
      list: () => Promise.reject(new PedidoError("pedido.repositorio")),
    };

    // Act
    renderScreen(<PedidoListContainer />, { pedidoRepository });

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe("pedido.repositorio");
  });
});
