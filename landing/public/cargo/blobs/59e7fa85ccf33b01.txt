use async_trait::async_trait;
use dominio_producto::errores::ErrorProducto;
use dominio_producto::modelo::Producto;
use dominio_producto::repositorio::RepositorioProducto;
use puerto_nucleo::{Pagina, PeticionPagina};
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsListarProductos {
    pub pagina: PeticionPagina,
}

#[async_trait]
pub trait ListarProductos: Send + Sync {
    async fn ejecutar(
        &self,
        params: ParamsListarProductos,
    ) -> Result<Pagina<Producto>, ErrorProducto>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub ListarProductos {}

        #[async_trait]
        impl ListarProductos for ListarProductos {
            async fn ejecutar(&self, params: ParamsListarProductos) -> Result<Pagina<Producto>, ErrorProducto>;
        }
    }
}

pub struct ListarProductosImpl {
    pub repositorio: Arc<dyn RepositorioProducto>,
}

#[async_trait]
impl ListarProductos for ListarProductosImpl {
    async fn ejecutar(
        &self,
        params: ParamsListarProductos,
    ) -> Result<Pagina<Producto>, ErrorProducto> {
        tracing::info!(
            limite = params.pagina.limite(),
            desplazamiento = params.pagina.desplazamiento(),
            "listar_productos: inicio"
        );
        self.repositorio
            .buscar_todos(params.pagina)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "listar_productos: fallo del repositorio"))
            .map_err(ErrorProducto::from)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_producto::{
        modelo::{Producto, PropsProducto},
        repositorio::dobles::MockRepositorioProducto,
    };
    use puerto_nucleo::PeticionPagina;

    #[tokio::test]
    async fn deberia_devolver_la_pagina_de_productos() {
        let entidades = vec![
            Producto::new(PropsProducto {
                nombre: "first".to_string(),
                precio: 42,
                sku: "first".to_string(),
            })
            .unwrap(),
            Producto::new(PropsProducto {
                nombre: "second".to_string(),
                precio: 42,
                sku: "second".to_string(),
            })
            .unwrap(),
        ];
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_todos()
            .returning(move |pagina| Ok(Pagina::new(entidades.clone(), 2, pagina)));
        let caso_de_uso = ListarProductosImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsListarProductos {
                pagina: PeticionPagina::default(),
            })
            .await;

        let pagina = resultado.unwrap();
        assert_eq!(pagina.elementos.len(), 2);
        assert_eq!(pagina.total, 2);
    }

    #[tokio::test]
    async fn deberia_devolver_una_pagina_vacia_cuando_no_hay_productos() {
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_todos()
            .returning(|pagina| Ok(Pagina::new(vec![], 0, pagina)));
        let caso_de_uso = ListarProductosImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsListarProductos {
                pagina: PeticionPagina::default(),
            })
            .await;

        let pagina = resultado.unwrap();
        assert!(pagina.elementos.is_empty());
        assert_eq!(pagina.total, 0);
    }

    #[tokio::test]
    async fn deberia_pasar_al_repositorio_la_pagina_pedida() {
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_todos()
            .withf(|pagina| pagina.limite() == 5 && pagina.desplazamiento() == 10)
            .returning(|pagina| Ok(Pagina::new(vec![], 0, pagina)));
        let caso_de_uso = ListarProductosImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsListarProductos {
                pagina: PeticionPagina::new(Some(5), Some(10)),
            })
            .await;

        assert_eq!(resultado.unwrap().desplazamiento, 10);
    }
}
