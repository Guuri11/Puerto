use puerto_nucleo::ErrorRepositorio;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum ErrorProducto {
    #[error("producto.validacion.{0}")]
    Validacion(String),
    #[error("producto.no_encontrado")]
    NoEncontrado,
    #[error("producto.duplicado")]
    Duplicado,
    #[error("producto.repositorio")]
    Repositorio,
    #[error("producto.desconocido")]
    Desconocido,
}

impl From<ErrorRepositorio> for ErrorProducto {
    /// Cada fallo que nombra `puerto-nucleo` tiene su brazo. `ErrorRepositorio` es
    /// `#[non_exhaustive]`: uno que añada una versión posterior es un problema del almacén (500)
    /// hasta que tenga aquí un significado.
    fn from(err: ErrorRepositorio) -> Self {
        match err {
            ErrorRepositorio::Duplicado => Self::Duplicado,
            ErrorRepositorio::DatosInvalidos | ErrorRepositorio::NoDisponible => Self::Repositorio,
            _ => Self::Repositorio,
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn deberia_convertir_un_duplicado_en_el_duplicado_de_la_entidad() {
        let err = ErrorProducto::from(ErrorRepositorio::Duplicado);
        assert_eq!(err.to_string(), "producto.duplicado");
    }

    #[test]
    fn deberia_convertir_datos_invalidos_en_un_error_de_repositorio() {
        let err = ErrorProducto::from(ErrorRepositorio::DatosInvalidos);
        assert_eq!(err.to_string(), "producto.repositorio");
    }

    #[test]
    fn deberia_convertir_un_almacen_no_disponible_en_un_error_de_repositorio() {
        let err = ErrorProducto::from(ErrorRepositorio::NoDisponible);
        assert_eq!(err.to_string(), "producto.repositorio");
    }
}
