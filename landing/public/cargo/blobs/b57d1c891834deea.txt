import { ClienteError } from "../domain/cliente/errors";
import type { Cliente, ClienteDraft } from "../domain/cliente/model";
import type { ClienteRepository } from "../domain/cliente/repository";

/** Valid clientes for tests; pass `overrides` for the fields a test is about. */
export const ClienteMother = {
  draft(overrides: Partial<ClienteDraft> = {}): ClienteDraft {
    return {
      email: "example",
      nombre: "example",
      etiquetas: ["example"],
      ...overrides,
    };
  },

  random(overrides: Partial<Cliente> = {}): Cliente {
    return { id: crypto.randomUUID(), ...ClienteMother.draft(), ...overrides };
  },
};

/** An in-memory `ClienteRepository`. Override a method to make it fail. */
export function fakeClienteRepository(seed: Cliente[] = []): ClienteRepository {
  const store = new Map(seed.map((item) => [item.id, item]));
  const find = (id: string) => {
    const found = store.get(id);
    if (found === undefined) throw new ClienteError("cliente.no_encontrado");
    return found;
  };
  return {
    list: async ({ limit, offset }) => {
      const all = [...store.values()];
      return { items: all.slice(offset, offset + limit), total: all.length, limit, offset };
    },
    get: async (id) => find(id),
    create: async (draft) => {
      const created = { id: crypto.randomUUID(), ...draft };
      store.set(created.id, created);
      return created;
    },
    update: async (id, draft) => {
      const updated = { ...find(id), ...draft };
      store.set(id, updated);
      return updated;
    },
    remove: async (id) => {
      find(id);
      store.delete(id);
    },
  };
}
