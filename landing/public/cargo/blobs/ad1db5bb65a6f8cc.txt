/**
 * A request about a customer that the API refused. `code` is the API's `ErrorResponse.name`:
 * `customer.not_found`, `customer.validation_error.<field>_empty`.
 */
export class CustomerError extends Error {
  readonly code: string;

  constructor(code: string) {
    super(code);
    this.name = "CustomerError";
    this.code = code;
  }
}
