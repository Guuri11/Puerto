use async_trait::async_trait;
use dominio_pedido::errores::ErrorPedido;
use dominio_pedido::modelo::{IdPedido, Pedido};
use dominio_pedido::repositorio::RepositorioPedido;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsObtenerPedido {
    pub id: IdPedido,
}

#[async_trait]
pub trait ObtenerPedido: Send + Sync {
    async fn ejecutar(&self, params: ParamsObtenerPedido) -> Result<Pedido, ErrorPedido>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub ObtenerPedido {}

        #[async_trait]
        impl ObtenerPedido for ObtenerPedido {
            async fn ejecutar(&self, params: ParamsObtenerPedido) -> Result<Pedido, ErrorPedido>;
        }
    }
}

pub struct ObtenerPedidoImpl {
    pub repositorio: Arc<dyn RepositorioPedido>,
}

#[async_trait]
impl ObtenerPedido for ObtenerPedidoImpl {
    async fn ejecutar(&self, params: ParamsObtenerPedido) -> Result<Pedido, ErrorPedido> {
        tracing::info!(id = %params.id, "obtener_pedido: inicio");
        let resultado = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(
                |e| tracing::error!(error = %e, "obtener_pedido: fallo del repositorio"),
            )?;
        resultado.ok_or_else(|| {
            let err = ErrorPedido::NoEncontrado;
            tracing::warn!(error = %err, "obtener_pedido: rechazado");
            err
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_pedido::valores::Estado;
    use dominio_pedido::valores::Importe;
    use dominio_pedido::{
        modelo::{IdPedido, Pedido, PropsPedido},
        repositorio::dobles::MockRepositorioPedido,
    };
    use puerto_nucleo::ErrorRepositorio;

    #[tokio::test]
    async fn deberia_devolver_pedido_cuando_el_id_existe() {
        let entidad = Pedido::new(PropsPedido {
            importe: Importe::new(1.5).expect("Importe::new rechazó el valor de ejemplo"),
            estado: Estado::Pendiente,
            nota: None,
        })
        .unwrap();
        let id_entidad = entidad.id;
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        let caso_de_uso = ObtenerPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerPedido { id: id_entidad })
            .await;

        assert!(resultado.is_ok());
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_cuando_el_id_no_existe() {
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let caso_de_uso = ObtenerPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerPedido {
                id: IdPedido::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "pedido.no_encontrado");
    }

    #[tokio::test]
    async fn deberia_devolver_error_de_repositorio_cuando_el_almacen_no_esta_disponible() {
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Err(ErrorRepositorio::NoDisponible));
        let caso_de_uso = ObtenerPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerPedido {
                id: IdPedido::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "pedido.repositorio");
    }
}
