use crate::valores::Email;
use crate::valores::Etiqueta;
use chrono::{DateTime, Utc};
use uuid::Uuid;

use crate::errores::ErrorCliente;

/// El id de cada `Cliente`: un `Uuid` que no se puede pasar donde se espera el id de otra
/// entidad.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct IdCliente(Uuid);

impl IdCliente {
    /// Un id nuevo y aleatorio — para una entidad que se está creando.
    pub fn generar() -> Self {
        Self(Uuid::new_v4())
    }

    /// Un id leído de fuera (un parámetro de ruta, una fila) — sin validar: cualquier `Uuid` es un
    /// id.
    pub fn from_uuid(valor: Uuid) -> Self {
        Self(valor)
    }

    pub fn valor(&self) -> Uuid {
        self.0
    }
}

impl std::fmt::Display for IdCliente {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        self.0.fmt(f)
    }
}

#[derive(Debug, Clone)]
pub struct PropsCliente {
    pub email: Email,
    pub nombre: String,
    pub etiquetas: Vec<Etiqueta>,
}

#[derive(Debug, Clone)]
pub struct Cliente {
    pub id: IdCliente,
    pub creado_en: DateTime<Utc>,
    pub actualizado_en: DateTime<Utc>,
    pub borrado: bool,
    pub borrado_en: Option<DateTime<Utc>>,
    pub email: Email,
    pub nombre: String,
    pub etiquetas: Vec<Etiqueta>,
}

impl Cliente {
    pub fn new(props: PropsCliente) -> Result<Self, ErrorCliente> {
        if props.nombre.trim().is_empty() {
            return Err(ErrorCliente::Validacion("nombre_vacio".into()));
        }
        let ahora = chrono::Utc::now();
        Ok(Self {
            id: IdCliente::generar(),
            creado_en: ahora,
            actualizado_en: ahora,
            borrado: false,
            borrado_en: None,
            email: props.email,
            nombre: props.nombre,
            etiquetas: props.etiquetas,
        })
    }

    pub fn from_repositorio(datos: Cliente) -> Self {
        datos
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn deberia_crear_cliente_cuando_los_campos_son_validos() {
        let resultado = Cliente::new(PropsCliente {
            email: Email::new("example".to_string())
                .expect("Email::new rechazó el valor de ejemplo"),
            nombre: "example".to_string(),
            etiquetas: vec![],
        });
        assert_eq!(resultado.unwrap().nombre, "example");
    }

    #[test]
    fn deberia_rechazar_cliente_cuando_nombre_es_una_cadena_vacia() {
        let resultado = Cliente::new(PropsCliente {
            email: Email::new("example".to_string())
                .expect("Email::new rechazó el valor de ejemplo"),
            nombre: "".into(),
            etiquetas: vec![],
        });
        assert!(resultado.is_err());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "cliente.validacion.nombre_vacio"
        );
    }

    #[test]
    fn deberia_rechazar_cliente_cuando_nombre_solo_tiene_espacios() {
        let resultado = Cliente::new(PropsCliente {
            email: Email::new("example".to_string())
                .expect("Email::new rechazó el valor de ejemplo"),
            nombre: "   ".into(),
            etiquetas: vec![],
        });
        assert!(resultado.is_err());
    }

    #[test]
    fn deberia_rechazar_cliente_cuando_email_es_una_cadena_vacia() {
        let resultado = Email::new("".to_string());
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.email_invalido");
    }
}
