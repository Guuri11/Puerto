import type { ClienteProblem } from "../../../../domain/cliente/messages";
import type { Cliente, ClienteDraft } from "../../../../domain/cliente/model";
import { splitList } from "../../../shared/forms";
import { FieldError } from "../../../shared/ui/field-error";

/** What the inputs hold: the text of every input, `checked` for a checkbox. */
export interface ClienteFormValues {
  email: string;
  nombre: string;
  etiquetas: string;
}

/** The inputs for `cliente`, empty when there is none yet. */
export function toClienteForm(cliente: Cliente | null): ClienteFormValues {
  if (cliente === null) {
    return {
      email: "",
      nombre: "",
      etiquetas: "",
    };
  }
  return {
    email: cliente.email,
    nombre: cliente.nombre,
    etiquetas: cliente.etiquetas.join(", "),
  };
}

export function toClienteDraft(values: ClienteFormValues): ClienteDraft {
  return {
    email: values.email,
    nombre: values.nombre,
    etiquetas: splitList(values.etiquetas),
  };
}

export function ClienteFormView({
  values,
  saving,
  problem,
  onChange,
  onSubmit,
}: {
  values: ClienteFormValues;
  saving: boolean;
  /** What the API said about the last submit: shown under its input. */
  problem: ClienteProblem | null;
  onChange: (values: ClienteFormValues) => void;
  onSubmit: () => void;
}) {
  return (
    <form
      className="space-y-2"
      onSubmit={(event) => {
        event.preventDefault();
        onSubmit();
      }}
    >
      <label className="fieldset">
        <span className="fieldset-legend">Email</span>
        <input
          className="input w-full"
          type="text"
          aria-invalid={problem?.field === "email"}
          required
          value={values.email}
          onChange={(event) => onChange({ ...values, email: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="email" />
      <label className="fieldset">
        <span className="fieldset-legend">Nombre</span>
        <input
          className="input w-full"
          type="text"
          aria-invalid={problem?.field === "nombre"}
          required
          value={values.nombre}
          onChange={(event) => onChange({ ...values, nombre: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="nombre" />
      <label className="fieldset">
        <span className="fieldset-legend">Etiquetas</span>
        <input
          className="input w-full"
          type="text"
          aria-invalid={problem?.field === "etiquetas"}
          value={values.etiquetas}
          onChange={(event) => onChange({ ...values, etiquetas: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="etiquetas" />
      <button type="submit" className="btn btn-primary" disabled={saving}>
        Save
      </button>
    </form>
  );
}
