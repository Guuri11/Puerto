import { useState } from "react";
import { Link, useLocation } from "wouter";
import { useAuth } from "../../../core/auth";
import { useDependencies } from "../../../core/dependencies";
import { useAsync } from "../../../core/hooks/use-async";
import { ConfirmDialog } from "../../../shared/ui/confirm-dialog";
import { ErrorAlert } from "../../../shared/ui/error-alert";
import { PedidoDetailView } from "../views/pedido-detail-view";

export function PedidoDetailContainer({ id }: { id: string }) {
  const { pedidoRepository } = useDependencies();
  const { can } = useAuth();
  const pedido = useAsync(() => pedidoRepository.get(id), [pedidoRepository, id]);
  const [, navigate] = useLocation();
  const [confirming, setConfirming] = useState(false);
  const [failure, setFailure] = useState<unknown>(null);

  const remove = async () => {
    setConfirming(false);
    try {
      await pedidoRepository.remove(id);
      navigate("/pedidos");
    } catch (error) {
      setFailure(error);
    }
  };

  return (
    <section className="space-y-4">
      <h1 className="text-2xl font-bold">Pedido</h1>
      {failure !== null && <ErrorAlert error={failure} />}
      {pedido.status === "loading" && <span className="loading loading-spinner" />}
      {pedido.status === "error" && <ErrorAlert error={pedido.error} />}
      {pedido.status === "ok" && (
        <div className="card bg-base-100">
          <div className="card-body">
            <PedidoDetailView pedido={pedido.data} />
            <div className="card-actions justify-end">
              <Link href={`/pedidos/${id}/edit`} className="btn">
                Edit
              </Link>
              {can("admin") && (
                <button type="button" className="btn btn-error" onClick={() => setConfirming(true)}>
                  Delete
                </button>
              )}
            </div>
          </div>
        </div>
      )}
      <Link href="/pedidos" className="link">
        Back to pedidos
      </Link>
      <ConfirmDialog
        open={confirming}
        message="Delete this pedido?"
        onConfirm={remove}
        onCancel={() => setConfirming(false)}
      />
    </section>
  );
}
