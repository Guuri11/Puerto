import type { ProductProblem } from "../../../../domain/product/messages";
import type { Product, ProductDraft } from "../../../../domain/product/model";
import { FieldError } from "../../../shared/ui/field-error";

/** What the inputs hold: the text of every input, `checked` for a checkbox. */
export interface ProductFormValues {
  name: string;
  price: string;
  sku: string;
}

/** The inputs for `product`, empty when there is none yet. */
export function toProductForm(product: Product | null): ProductFormValues {
  if (product === null) {
    return {
      name: "",
      price: "",
      sku: "",
    };
  }
  return {
    name: product.name,
    price: String(product.price),
    sku: product.sku,
  };
}

export function toProductDraft(values: ProductFormValues): ProductDraft {
  return {
    name: values.name,
    price: Number(values.price),
    sku: values.sku,
  };
}

export function ProductFormView({
  values,
  saving,
  problem,
  onChange,
  onSubmit,
}: {
  values: ProductFormValues;
  saving: boolean;
  /** What the API said about the last submit: shown under its input. */
  problem: ProductProblem | null;
  onChange: (values: ProductFormValues) => void;
  onSubmit: () => void;
}) {
  return (
    <form
      className="space-y-2"
      onSubmit={(event) => {
        event.preventDefault();
        onSubmit();
      }}
    >
      <label className="fieldset">
        <span className="fieldset-legend">Name</span>
        <input
          className="input w-full"
          type="text"
          aria-invalid={problem?.field === "name"}
          required
          value={values.name}
          onChange={(event) => onChange({ ...values, name: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="name" />
      <label className="fieldset">
        <span className="fieldset-legend">Price</span>
        <input
          className="input w-full"
          type="number"
          aria-invalid={problem?.field === "price"}
          required
          value={values.price}
          onChange={(event) => onChange({ ...values, price: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="price" />
      <label className="fieldset">
        <span className="fieldset-legend">Sku</span>
        <input
          className="input w-full"
          type="text"
          aria-invalid={problem?.field === "sku"}
          required
          value={values.sku}
          onChange={(event) => onChange({ ...values, sku: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="sku" />
      <button type="submit" className="btn btn-primary" disabled={saving}>
        Save
      </button>
    </form>
  );
}
