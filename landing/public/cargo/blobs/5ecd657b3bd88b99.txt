import { describe, expect, test } from "bun:test";
import { fireEvent, screen, waitFor } from "@testing-library/react";
import type { Esquemas } from "../../../api/cliente";
import { textoDe } from "../../../compartido/formularios";
import { apiFalsa } from "../../../pruebas/api-falsa";
import { coleccionPedido, MadrePedido } from "../../../pruebas/pedido";
import { renderizarPantalla } from "../../../pruebas/renderizar";
import { ContenedorFormularioPedido } from "./formulario-pedido";

function escribir(etiqueta: string, valor: string) {
  fireEvent.change(screen.getByLabelText(etiqueta), { target: { value: valor } });
}

/** Rellena el formulario vacío con `peticion`, como lo haría quien lo usa. */
async function rellenar(peticion: Esquemas["PeticionCrearPedido"]) {
  await screen.findByText("Guardar");
  escribir("Importe", String(peticion.importe));
  escribir("Estado", peticion.estado);
  escribir("Nota", textoDe(peticion.nota));
}

describe("ContenedorFormularioPedido", () => {
  test("debería crear con lo escrito e ir a su detalle", async () => {
    // Preparar
    const peticion = MadrePedido.peticion();
    const guardados = coleccionPedido();
    const api = apiFalsa([guardados]);
    const pantalla = <ContenedorFormularioPedido />;
    const { historial } = renderizarPantalla(pantalla, { api });
    await rellenar(peticion);

    // Actuar
    fireEvent.click(screen.getByText("Guardar"));

    // Comprobar
    await waitFor(() => expect(historial.at(-1)).toStartWith("/pedidos/"));
    const [creado] = guardados.elementos.values();
    expect(creado).toMatchObject(peticion);
    expect(historial.at(-1)).toBe(`/pedidos/${creado?.id}`);
  });

  test("debería guardar la edición cuando se envía el formulario", async () => {
    // Preparar
    const sembrado = MadrePedido.aleatorio();
    const guardados = coleccionPedido([sembrado]);
    const api = apiFalsa([guardados]);
    const pantalla = <ContenedorFormularioPedido id={sembrado.id} />;
    const { historial } = renderizarPantalla(pantalla, { api });

    // Actuar
    fireEvent.click(await screen.findByText("Guardar"));

    // Comprobar
    await waitFor(() => expect(historial.at(-1)).toBe(`/pedidos/${sembrado.id}`));
    expect(guardados.elementos.get(sembrado.id)).toEqual(sembrado);
  });

  test("debería mostrar el propio código cuando la API responde uno que el formulario no conoce", async () => {
    // Preparar
    const peticion = MadrePedido.peticion();
    const codigo = "pedido.validacion.invalido";
    const fallos = { "POST /pedidos": codigo };
    const api = apiFalsa([coleccionPedido()], fallos);
    renderizarPantalla(<ContenedorFormularioPedido />, { api });
    await rellenar(peticion);

    // Actuar
    fireEvent.click(screen.getByText("Guardar"));

    // Comprobar
    expect((await screen.findByRole("alert")).textContent).toBe(codigo);
  });

  test("debería decir que los valores ya están en uso cuando la API responde un duplicado", async () => {
    // Preparar
    const peticion = MadrePedido.peticion();
    const fallos = { "POST /pedidos": "pedido.duplicado" };
    const api = apiFalsa([coleccionPedido()], fallos);
    renderizarPantalla(<ContenedorFormularioPedido />, { api });
    await rellenar(peticion);

    // Actuar
    fireEvent.click(screen.getByText("Guardar"));

    // Comprobar
    expect((await screen.findByRole("alert")).textContent).toBe("Estos valores ya están en uso");
  });

  test("debería mostrar el mensaje bajo su input cuando la API rechaza un campo", async () => {
    // Preparar
    const peticion = MadrePedido.peticion();
    const fallos = { "POST /pedidos": "pedido.importe_invalido" };
    const api = apiFalsa([coleccionPedido()], fallos);
    renderizarPantalla(<ContenedorFormularioPedido />, { api });
    await rellenar(peticion);

    // Actuar
    fireEvent.click(screen.getByText("Guardar"));

    // Comprobar
    expect(await screen.findByText("Importe: no válido")).toBeTruthy();
    const campo = screen.getByLabelText("Importe");
    expect(campo.getAttribute("aria-invalid")).toBe("true");
    expect(screen.queryByRole("alert")).toBeNull();
  });
});
