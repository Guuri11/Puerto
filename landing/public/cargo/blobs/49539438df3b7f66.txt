import { describe, expect, test } from "bun:test";
import { fireEvent, screen, waitFor } from "@testing-library/react";
import { apiFalsa } from "../../../pruebas/api-falsa";
import { coleccionCliente, MadreCliente } from "../../../pruebas/cliente";
import { renderizarPantalla } from "../../../pruebas/renderizar";
import { ContenedorDetalleCliente } from "./detalle-cliente";

describe("ContenedorDetalleCliente", () => {
  test("debería mostrar el detalle cuando existe", async () => {
    // Preparar
    const sembrado = MadreCliente.aleatorio();
    const api = apiFalsa([coleccionCliente([sembrado])]);

    const pantalla = <ContenedorDetalleCliente id={sembrado.id} />;

    // Actuar
    renderizarPantalla(pantalla, { api });

    // Comprobar
    expect(await screen.findByText("Editar")).toBeTruthy();
  });

  test("debería mostrar el código de no encontrado cuando no existe", async () => {
    // Preparar
    const api = apiFalsa([coleccionCliente()]);
    const pantalla = <ContenedorDetalleCliente id={crypto.randomUUID()} />;

    // Actuar
    renderizarPantalla(pantalla, { api });

    // Comprobar
    expect((await screen.findByRole("alert")).textContent).toBe("cliente.no_encontrado");
  });

  test("debería borrar solo tras confirmar el borrado", async () => {
    // Preparar
    const sembrado = MadreCliente.aleatorio();
    const guardados = coleccionCliente([sembrado]);
    const api = apiFalsa([guardados]);
    const pantalla = <ContenedorDetalleCliente id={sembrado.id} />;
    const { historial } = renderizarPantalla(pantalla, { api });
    fireEvent.click(await screen.findByText("Borrar", { selector: ".card-actions button" }));
    expect(guardados.elementos.has(sembrado.id)).toBe(true);

    // Actuar
    fireEvent.click(screen.getByText("Borrar", { selector: ".modal-action button" }));

    // Comprobar
    await waitFor(() => expect(historial.at(-1)).toBe("/clientes"));
    expect(guardados.elementos.has(sembrado.id)).toBe(false);
  });
});
