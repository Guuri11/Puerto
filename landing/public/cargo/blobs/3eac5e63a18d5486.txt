use std::sync::Arc;

use crate::domain::customer::value_objects::Email;
use crate::domain::customer::value_objects::Tag;
use async_trait::async_trait;
use puerto_core::RepositoryError;

use crate::domain::customer::{
    errors::CustomerError,
    events::{CustomerEvent, CustomerSnapshot},
    model::Customer,
    repository::CustomerRepositoryTrait,
    unit_of_work::CustomerUnitOfWorkFactory,
    use_cases::update_customer::{UpdateCustomerParams, UpdateCustomerUseCaseTrait},
};

pub struct UpdateCustomerUseCaseImpl {
    pub repository: Arc<dyn CustomerRepositoryTrait>,
    pub uow_factory: Arc<dyn CustomerUnitOfWorkFactory>,
}

#[async_trait]
impl UpdateCustomerUseCaseTrait for UpdateCustomerUseCaseImpl {
    async fn execute(&self, params: UpdateCustomerParams) -> Result<Customer, CustomerError> {
        tracing::info!(id = %params.id, "update_customer: start");
        let failed =
            |e: &RepositoryError| tracing::error!(error = %e, "update_customer: repository failed");
        let mut entity = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(failed)?
            .ok_or_else(|| {
                let err = CustomerError::NotFound;
                tracing::warn!(error = %err, "update_customer: rejected");
                err
            })?;
        if params.name.trim().is_empty() {
            let err = CustomerError::ValidationError("name_empty".into());
            tracing::warn!(error = %err, "update_customer: rejected");
            return Err(err);
        }
        let email = Email::new(params.email)?;
        let tags: Vec<Tag> = params
            .tags
            .into_iter()
            .map(Tag::new)
            .collect::<Result<Vec<_>, _>>()?;
        entity.email = email;
        entity.name = params.name;
        entity.tags = tags;
        entity.updated_at = chrono::Utc::now();
        let mut uow = self.uow_factory.begin().await.inspect_err(failed)?;
        uow.save(&entity).await.inspect_err(failed)?;
        uow.record(CustomerEvent::Updated(CustomerSnapshot::from_entity(
            &entity,
        )))
        .await
        .inspect_err(failed)?;
        uow.commit().await.inspect_err(failed)?;
        Ok(entity)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::customer::value_objects::Email;
    use crate::domain::customer::{
        model::{CustomerId, CustomerProps},
        repository::mocks::MockCustomerRepository,
        unit_of_work::mocks::RecordingUnitOfWorkFactory,
    };
    use puerto_core::DomainEvent;

    fn repository_holding(entity: Customer) -> MockCustomerRepository {
        let mut mock_repo = MockCustomerRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        mock_repo
    }

    #[tokio::test]
    async fn should_record_customer_updated_when_params_are_valid() {
        // Arrange
        let entity = Customer::new(CustomerProps {
            email: Email::new("original".to_string()).expect("valid Email"),
            name: "original".to_string(),
            tags: vec![],
        })
        .unwrap();
        let entity_id = entity.id;
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = UpdateCustomerUseCaseImpl {
            repository: Arc::new(repository_holding(entity)),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(UpdateCustomerParams {
                id: entity_id,
                email: "updated".to_string(),
                name: "updated".to_string(),
                tags: vec![],
            })
            .await;

        // Assert
        assert_eq!(result.unwrap().name, "updated");
        let recorded = uow_factory.recorded();
        assert!(recorded.committed);
        assert_eq!(recorded.events.len(), 1);
        assert_eq!(recorded.events[0].event_type(), "customer_updated");
        assert_eq!(recorded.events[0].aggregate_id(), entity_id.value());
    }

    #[tokio::test]
    async fn should_return_not_found_and_record_nothing_when_the_customer_does_not_exist() {
        // Arrange
        let mut mock_repo = MockCustomerRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = UpdateCustomerUseCaseImpl {
            repository: Arc::new(mock_repo),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(UpdateCustomerParams {
                id: CustomerId::generate(),
                email: "new".to_string(),
                name: "new".to_string(),
                tags: vec![],
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "customer.not_found");
        let recorded = uow_factory.recorded();
        assert!(recorded.saved.is_empty());
        assert!(recorded.events.is_empty());
    }

    #[tokio::test]
    async fn should_return_error_and_record_nothing_when_name_is_empty() {
        // Arrange
        let entity = Customer::new(CustomerProps {
            email: Email::new("original".to_string()).expect("valid Email"),
            name: "original".to_string(),
            tags: vec![],
        })
        .unwrap();
        let entity_id = entity.id;
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = UpdateCustomerUseCaseImpl {
            repository: Arc::new(repository_holding(entity)),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(UpdateCustomerParams {
                id: entity_id,
                email: "updated".to_string(),
                name: "".to_string(),
                tags: vec![],
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(
            result.unwrap_err().to_string(),
            "customer.validation_error.name_empty"
        );
        assert!(uow_factory.recorded().saved.is_empty());
    }

    #[tokio::test]
    async fn should_return_error_and_record_nothing_when_email_is_empty() {
