import { Route, Switch } from "wouter";
import { contarTotal, PAGINA_DE_UNO } from "../api/http";
import { RequiereAuth } from "../auth/auth";
import type { EntradaMenu } from "../compartido/ui/marco";
import { ContenedorDetalleCliente } from "../contextos/cliente/contenedores/detalle-cliente";
import { ContenedorFormularioCliente } from "../contextos/cliente/contenedores/formulario-cliente";
import { ContenedorListaCliente } from "../contextos/cliente/contenedores/lista-cliente";
import { PaginaNoEncontrada } from "../paginas/no-encontrada";
import { PaginaPanel } from "../paginas/panel";
import { PaginaPerfil } from "../paginas/perfil";
import { rutasPropias } from "../rutas";

/** Dónde va la app: en la raíz o, con portada, en `[proyecto] ruta_app` (`puerto nuevo --portada`). */
export const rutaApp = "/";

export const entradasMenu: EntradaMenu[] = [
  {
    etiqueta: "Clientes",
    href: "/clientes",
    total: ({ api }) => contarTotal(api.GET("/clientes", PAGINA_DE_UNO)),
  },
];

export function RutasApp() {
  return (
    <Switch>
      <Route path="/">
        <PaginaPanel entradas={entradasMenu} />
      </Route>
      <Route path="/perfil">
        <RequiereAuth>
          <PaginaPerfil />
        </RequiereAuth>
      </Route>
      {rutasPropias}
      <Route path="/clientes">
        <ContenedorListaCliente />
      </Route>
      <Route path="/clientes/nuevo">
        <RequiereAuth>
          <ContenedorFormularioCliente />
        </RequiereAuth>
      </Route>
      <Route path="/clientes/:id/editar">
        {(params) => (
          <RequiereAuth>
            <ContenedorFormularioCliente id={params.id} />
          </RequiereAuth>
        )}
      </Route>
      <Route path="/clientes/:id">{(params) => <ContenedorDetalleCliente id={params.id} />}</Route>
      <Route>
        <PaginaNoEncontrada />
      </Route>
    </Switch>
  );
}
