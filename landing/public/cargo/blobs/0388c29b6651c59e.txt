// GENERADO por puerto desde puerto.toml — no lo edites a mano. El snapshot sigue los campos de la
// entidad; añade un evento con `puerto generar evento Cliente <nombre>`.
use puerto_nucleo::EventoDominio;
use serde::{Deserialize, Serialize};
use uuid::Uuid;

use crate::modelo::Cliente;

pub const CLIENTE_CREADO: &str = "cliente_creado";
pub const CLIENTE_ACTUALIZADO: &str = "cliente_actualizado";
pub const CLIENTE_BORRADO: &str = "cliente_borrado";

/// Lo que lleva cada evento de cliente: sus datos tal como los muestra la API — primitivos, con
/// los objetos valor desenvueltos. Un contrato público: los consumidores lo decodifican con
/// `evento.decodificar::<SnapshotCliente>()`.
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
#[serde(rename_all = "camelCase")]
pub struct SnapshotCliente {
    pub id: Uuid,
    pub email: String,
    pub nombre: String,
    pub etiquetas: Vec<String>,
}

impl SnapshotCliente {
    pub fn from_entidad(entidad: &Cliente) -> Self {
        Self {
            id: entidad.id.valor(),
            email: entidad.email.valor().to_string(),
            nombre: entidad.nombre.clone(),
            etiquetas: entidad
                .etiquetas
                .iter()
                .map(|v| v.valor().to_string())
                .collect::<Vec<_>>(),
        }
    }
}

/// Lo que le pasa a cada cliente. Sus casos de uso de escritura registran uno con su unidad de
/// trabajo.
#[derive(Debug, Clone, PartialEq)]
pub enum EventoCliente {
    Creado(SnapshotCliente),
    Actualizado(SnapshotCliente),
    Borrado(SnapshotCliente),
}

impl EventoCliente {
    pub fn snapshot(&self) -> &SnapshotCliente {
        match self {
            Self::Creado(snapshot) | Self::Actualizado(snapshot) | Self::Borrado(snapshot) => {
                snapshot
            }
        }
    }
}

impl EventoDominio for EventoCliente {
    fn tipo_evento(&self) -> &'static str {
        match self {
            Self::Creado(_) => CLIENTE_CREADO,
            Self::Actualizado(_) => CLIENTE_ACTUALIZADO,
            Self::Borrado(_) => CLIENTE_BORRADO,
        }
    }

    fn tipo_agregado(&self) -> &'static str {
        "cliente"
    }

    fn id_agregado(&self) -> Uuid {
        self.snapshot().id
    }

    fn version_esquema(&self) -> u16 {
        1
    }

    fn payload(&self) -> serde_json::Value {
        // Un snapshot solo tiene primitivos: serializarlo no puede fallar.
        serde_json::to_value(self.snapshot()).unwrap_or_default()
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::pruebas::madres::MadreCliente;

    #[test]
    fn deberia_describir_cada_evento_cuando_se_registra() {
        // Preparar
        let entidad = MadreCliente::aleatorio();
        let snapshot = SnapshotCliente::from_entidad(&entidad);
        let eventos = [
            (EventoCliente::Creado(snapshot.clone()), "cliente_creado"),
            (
                EventoCliente::Actualizado(snapshot.clone()),
                "cliente_actualizado",
            ),
            (EventoCliente::Borrado(snapshot.clone()), "cliente_borrado"),
        ];

        for (evento, tipo_evento) in eventos {
            // Actuar y comprobar
            assert_eq!(evento.tipo_evento(), tipo_evento);
            assert_eq!(evento.tipo_agregado(), "cliente");
            assert_eq!(evento.id_agregado(), entidad.id.valor());
            assert_eq!(evento.version_esquema(), 1);
            assert_eq!(evento.snapshot(), &snapshot);
        }
    }

    #[test]
    fn deberia_mantener_el_contrato_json_cuando_se_serializa_el_snapshot() {
        // Preparar
        let snapshot = SnapshotCliente::from_entidad(&MadreCliente::aleatorio());

        // Actuar
        let payload = EventoCliente::Creado(snapshot.clone()).payload();

        // Comprobar — una clave renombrada o quitada rompe a cada consumidor de estos eventos.
        let mut claves: Vec<&str> = payload
            .as_object()
            .map(|object| object.keys().map(String::as_str).collect())
            .unwrap_or_default();
        claves.sort_unstable();
        assert_eq!(claves, ["email", "etiquetas", "id", "nombre"]);
        assert_eq!(
            serde_json::from_value::<SnapshotCliente>(payload).ok(),
            Some(snapshot)
        );
    }
}
