import { PedidoError } from "../domain/pedido/errors";
import type { Pedido, PedidoDraft } from "../domain/pedido/model";
import type { PedidoRepository } from "../domain/pedido/repository";

/** Valid pedidos for tests; pass `overrides` for the fields a test is about. */
export const PedidoMother = {
  draft(overrides: Partial<PedidoDraft> = {}): PedidoDraft {
    return {
      importe: 1.5,
      estado: "Pendiente",
      nota: "example",
      ...overrides,
    };
  },

  random(overrides: Partial<Pedido> = {}): Pedido {
    return { id: crypto.randomUUID(), ...PedidoMother.draft(), ...overrides };
  },
};
