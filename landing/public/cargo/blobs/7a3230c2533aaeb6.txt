import { describe, expect, test } from "bun:test";
import { screen, waitFor } from "@testing-library/react";
import { CurrentUserMother, fakeAuthGateway } from "../../test-support/auth";
import { renderScreen } from "../../test-support/render";
import { RequireAuth, useAuth } from "./auth";

function Secret() {
  const { can } = useAuth();
  return <p>{can("admin") ? "admin area" : "member area"}</p>;
}

describe("RequireAuth", () => {
  test("should start a sign-in that comes back here when nobody is signed in", async () => {
    // Arrange
    const auth = fakeAuthGateway({ user: null });

    // Act
    renderScreen(
      <RequireAuth>
        <Secret />
      </RequireAuth>,
      { auth },
      "/orders?page=2",
    );

    // Assert
    await waitFor(() => expect(auth.logins).toEqual(["/orders?page=2"]));
    expect(screen.queryByText("member area")).toBeNull();
  });

  test("should show the page to a signed-in user without signing in again", async () => {
    // Arrange
    const auth = fakeAuthGateway({ user: CurrentUserMother.random() });

    // Act
    renderScreen(
      <RequireAuth>
        <Secret />
      </RequireAuth>,
      { auth },
    );

    // Assert
    expect(await screen.findByText("member area")).toBeTruthy();
    expect(auth.logins).toEqual([]);
  });
});

describe("useAuth", () => {
  test("should allow a role only when the user holds it", async () => {
    // Arrange
    const user = CurrentUserMother.random({ roles: ["admin"] });

    // Act
    renderScreen(<Secret />, {}, "/", { user });

    // Assert
    expect(await screen.findByText("admin area")).toBeTruthy();
  });
});

describe("AuthProvider", () => {
  test("should show the error code when the API does not accept the session", async () => {
    // Arrange
    const auth = {
      ...fakeAuthGateway(),
      currentUser: () => Promise.reject(new Error("auth.unauthenticated")),
    };

    // Act
    renderScreen(<Secret />, { auth });

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe("auth.unauthenticated");
  });
});
