/** Valid customers for tests; pass `overrides` for the fields a test is about. */
export const CustomerMother = {
  draft(overrides: Partial<CustomerDraft> = {}): CustomerDraft {
    return {
      email: "example",
      name: "example",
      tags: ["example"],
      ...overrides,
    };
  },

  random(overrides: Partial<Customer> = {}): Customer {
    return { id: crypto.randomUUID(), ...CustomerMother.draft(), ...overrides };
  },
};
