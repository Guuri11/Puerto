// A success variant carries the whole DTO, so it outgrows the error variants as the entity gains
// fields. A response is built once per request and returned: boxing the DTO would only add an
// allocation.
#![allow(clippy::large_enum_variant)]

use poem::http::StatusCode;
use poem_openapi::{ApiResponse, payload::Json};
use puerto_web::ErrorResponse;

use crate::api::product::dto::{ProductDto, ProductPageDto};

#[derive(ApiResponse)]
pub enum CreateProductResponse {
    #[oai(status = 201)]
    Created(Json<ProductDto>),
    #[oai(status = 422)]
    UnprocessableEntity(Json<ErrorResponse>),
    #[oai(status = 401)]
    Unauthorized(Json<ErrorResponse>),
    #[oai(status = 403)]
    Forbidden(Json<ErrorResponse>),
    #[oai(status = 409)]
    Conflict(Json<ErrorResponse>),
    /// The token could not be checked: the provider or the database is down (`BearerAuth`).
    #[oai(status = 503)]
    Unavailable(Json<ErrorResponse>),
    #[oai(status = 500)]
    InternalError(Json<ErrorResponse>),
}

impl CreateProductResponse {
    pub fn from_status(status: StatusCode, error: Json<ErrorResponse>) -> Self {
        match status {
            StatusCode::UNPROCESSABLE_ENTITY => Self::UnprocessableEntity(error),
            StatusCode::UNAUTHORIZED => Self::Unauthorized(error),
            StatusCode::FORBIDDEN => Self::Forbidden(error),
            StatusCode::CONFLICT => Self::Conflict(error),
            StatusCode::SERVICE_UNAVAILABLE => Self::Unavailable(error),
            _ => Self::InternalError(error),
        }
    }
}
