use chrono::{DateTime, Utc};
use uuid::Uuid;

use crate::errores::ErrorProducto;

/// El id de cada `Producto`: un `Uuid` que no se puede pasar donde se espera el id de otra
/// entidad.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct IdProducto(Uuid);

impl IdProducto {
    /// Un id nuevo y aleatorio — para una entidad que se está creando.
    pub fn generar() -> Self {
        Self(Uuid::new_v4())
    }

    /// Un id leído de fuera (un parámetro de ruta, una fila) — sin validar: cualquier `Uuid` es un
    /// id.
    pub fn from_uuid(valor: Uuid) -> Self {
        Self(valor)
    }

    pub fn valor(&self) -> Uuid {
        self.0
    }
}

impl std::fmt::Display for IdProducto {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        self.0.fmt(f)
    }
}

#[derive(Debug, Clone)]
pub struct PropsProducto {
    pub nombre: String,
    pub precio: i64,
    pub sku: String,
}

#[derive(Debug, Clone)]
pub struct Producto {
    pub id: IdProducto,
    pub creado_en: DateTime<Utc>,
    pub actualizado_en: DateTime<Utc>,
    pub borrado: bool,
    pub borrado_en: Option<DateTime<Utc>>,
    pub nombre: String,
    pub precio: i64,
    pub sku: String,
}

impl Producto {
    pub fn new(props: PropsProducto) -> Result<Self, ErrorProducto> {
        if props.nombre.trim().is_empty() {
            return Err(ErrorProducto::Validacion("nombre_vacio".into()));
        }
        if props.sku.trim().is_empty() {
            return Err(ErrorProducto::Validacion("sku_vacio".into()));
        }
        let ahora = chrono::Utc::now();
        Ok(Self {
            id: IdProducto::generar(),
            creado_en: ahora,
            actualizado_en: ahora,
            borrado: false,
            borrado_en: None,
            nombre: props.nombre,
            precio: props.precio,
            sku: props.sku,
        })
    }

    pub fn from_repositorio(datos: Producto) -> Self {
        datos
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn deberia_crear_producto_cuando_los_campos_son_validos() {
        let resultado = Producto::new(PropsProducto {
            nombre: "example".to_string(),
            precio: 42,
            sku: "example".to_string(),
        });
        assert_eq!(resultado.unwrap().nombre, "example");
    }

    #[test]
    fn deberia_rechazar_producto_cuando_nombre_es_una_cadena_vacia() {
        let resultado = Producto::new(PropsProducto {
            nombre: "".into(),
            precio: 42,
            sku: "example".to_string(),
        });
        assert!(resultado.is_err());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "producto.validacion.nombre_vacio"
        );
    }

    #[test]
    fn deberia_rechazar_producto_cuando_nombre_solo_tiene_espacios() {
        let resultado = Producto::new(PropsProducto {
            nombre: "   ".into(),
            precio: 42,
            sku: "example".to_string(),
        });
        assert!(resultado.is_err());
    }

    #[test]
    fn deberia_rechazar_producto_cuando_sku_es_una_cadena_vacia() {
        let resultado = Producto::new(PropsProducto {
            nombre: "example".to_string(),
            precio: 42,
            sku: "".into(),
        });
        assert!(resultado.is_err());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "producto.validacion.sku_vacio"
        );
    }

    #[test]
    fn deberia_rechazar_producto_cuando_sku_solo_tiene_espacios() {
        let resultado = Producto::new(PropsProducto {
            nombre: "example".to_string(),
            precio: 42,
            sku: "   ".into(),
        });
        assert!(resultado.is_err());
    }
}
