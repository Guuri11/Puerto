import { ClienteError } from "../../domain/cliente/errors";
import type { Cliente, ClienteDraft } from "../../domain/cliente/model";
import type { ClienteRepository } from "../../domain/cliente/repository";
import { type ApiClient, apiErrorCode } from "../http/client";
import type { components } from "../http/schema";

type ClienteDto = components["schemas"]["DtoCliente"];
type ClienteRequest = components["schemas"]["PeticionCrearCliente"];

function toCliente(dto: ClienteDto): Cliente {
  return {
    id: dto.id,
    email: dto.email,
    nombre: dto.nombre,
    etiquetas: dto.etiquetas,
  };
}

function toRequest(draft: ClienteDraft): ClienteRequest {
  return {
    email: draft.email,
    nombre: draft.nombre,
    etiquetas: draft.etiquetas,
  };
}

/** `ClienteRepository` over the REST API (`/clientes`). */
export function clienteRepositoryHttp(api: ApiClient): ClienteRepository {
  return {
    async list(page) {
      const { data, error, response } = await api.GET("/clientes", {
        params: { query: { limite: page.limit, desplazamiento: page.offset } },
      });
      if (data === undefined) throw new ClienteError(apiErrorCode(error, response));
      return {
        items: data.elementos.map(toCliente),
        total: data.total,
        limit: data.limite,
        offset: data.desplazamiento,
      };
    },
    async get(id) {
      const { data, error, response } = await api.GET("/clientes/{id}", {
        params: { path: { id } },
      });
      if (data === undefined) throw new ClienteError(apiErrorCode(error, response));
      return toCliente(data);
    },
    async create(draft) {
      const { data, error, response } = await api.POST("/clientes", {
        body: toRequest(draft),
      });
      if (data === undefined) throw new ClienteError(apiErrorCode(error, response));
      return toCliente(data);
    },
    async update(id, draft) {
      const { data, error, response } = await api.PUT("/clientes/{id}", {
        params: { path: { id } },
        body: toRequest(draft),
      });
      if (data === undefined) throw new ClienteError(apiErrorCode(error, response));
      return toCliente(data);
    },
    async remove(id) {
      const { error, response } = await api.DELETE("/clientes/{id}", {
        params: { path: { id } },
      });
      if (!response.ok) throw new ClienteError(apiErrorCode(error, response));
    },
  };
}
