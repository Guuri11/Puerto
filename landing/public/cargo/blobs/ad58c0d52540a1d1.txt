use std::sync::Arc;

use async_trait::async_trait;
use puerto_core::RepositoryError;

use crate::domain::product::{
    errors::ProductError,
    events::{ProductEvent, ProductSnapshot},
    repository::ProductRepositoryTrait,
    unit_of_work::ProductUnitOfWorkFactory,
    use_cases::delete_product::{DeleteProductParams, DeleteProductUseCaseTrait},
};

pub struct DeleteProductUseCaseImpl {
    pub repository: Arc<dyn ProductRepositoryTrait>,
    pub uow_factory: Arc<dyn ProductUnitOfWorkFactory>,
}

#[async_trait]
impl DeleteProductUseCaseTrait for DeleteProductUseCaseImpl {
    async fn execute(&self, params: DeleteProductParams) -> Result<(), ProductError> {
        tracing::info!(id = %params.id, "delete_product: start");
        let failed =
            |e: &RepositoryError| tracing::error!(error = %e, "delete_product: repository failed");
        let mut entity = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(failed)?
            .ok_or_else(|| {
                let err = ProductError::NotFound;
                tracing::warn!(error = %err, "delete_product: rejected");
                err
            })?;
        let now = chrono::Utc::now();
        entity.deleted = true;
        entity.deleted_at = Some(now);
        entity.updated_at = now;
        let mut uow = self.uow_factory.begin().await.inspect_err(failed)?;
        uow.save(&entity).await.inspect_err(failed)?;
        uow.record(ProductEvent::Deleted(ProductSnapshot::from_entity(&entity)))
            .await
            .inspect_err(failed)?;
        uow.commit().await.inspect_err(failed)?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::product::{
        model::{Product, ProductId, ProductProps},
        repository::mocks::MockProductRepository,
        unit_of_work::mocks::RecordingUnitOfWorkFactory,
    };
    use puerto_core::DomainEvent;

    #[tokio::test]
    async fn should_record_product_deleted_when_id_exists() {
        // Arrange
        let entity = Product::new(ProductProps {
            name: "example".to_string(),
            price: 42,
            sku: "example".to_string(),
        })
        .unwrap();
        let entity_id = entity.id;
        let mut mock_repo = MockProductRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = DeleteProductUseCaseImpl {
            repository: Arc::new(mock_repo),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(DeleteProductParams { id: entity_id })
            .await;

        // Assert
        assert!(result.is_ok());
        let recorded = uow_factory.recorded();
        assert!(recorded.committed);
        assert!(recorded.saved.iter().all(|saved| saved.deleted));
        assert_eq!(recorded.events.len(), 1);
        assert_eq!(recorded.events[0].event_type(), "product_deleted");
        assert_eq!(recorded.events[0].aggregate_id(), entity_id.value());
    }

    #[tokio::test]
    async fn should_return_not_found_and_record_nothing_when_the_product_does_not_exist() {
        // Arrange
        let mut mock_repo = MockProductRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = DeleteProductUseCaseImpl {
            repository: Arc::new(mock_repo),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(DeleteProductParams {
                id: ProductId::generate(),
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "product.not_found");
        assert!(uow_factory.recorded().events.is_empty());
    }
}
