use std::sync::Arc;

use async_trait::async_trait;
use puerto_core::RepositoryError;

use crate::domain::order::{
    errors::OrderError,
    events::{OrderEvent, OrderSnapshot},
    repository::OrderRepositoryTrait,
    unit_of_work::OrderUnitOfWorkFactory,
    use_cases::delete_order::{DeleteOrderParams, DeleteOrderUseCaseTrait},
};

pub struct DeleteOrderUseCaseImpl {
    pub repository: Arc<dyn OrderRepositoryTrait>,
    pub uow_factory: Arc<dyn OrderUnitOfWorkFactory>,
}

#[async_trait]
impl DeleteOrderUseCaseTrait for DeleteOrderUseCaseImpl {
    async fn execute(&self, params: DeleteOrderParams) -> Result<(), OrderError> {
        tracing::info!(id = %params.id, "delete_order: start");
        let failed =
            |e: &RepositoryError| tracing::error!(error = %e, "delete_order: repository failed");
        let mut entity = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(failed)?
            .ok_or_else(|| {
                let err = OrderError::NotFound;
                tracing::warn!(error = %err, "delete_order: rejected");
                err
            })?;
        let now = chrono::Utc::now();
        entity.deleted = true;
        entity.deleted_at = Some(now);
        entity.updated_at = now;
        let mut uow = self.uow_factory.begin().await.inspect_err(failed)?;
        uow.save(&entity).await.inspect_err(failed)?;
        uow.record(OrderEvent::Deleted(OrderSnapshot::from_entity(&entity)))
            .await
            .inspect_err(failed)?;
        uow.commit().await.inspect_err(failed)?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::order::value_objects::Amount;
    use crate::domain::order::value_objects::Status;
    use crate::domain::order::{
        model::{Order, OrderId, OrderProps},
        repository::mocks::MockOrderRepository,
        unit_of_work::mocks::RecordingUnitOfWorkFactory,
    };
    use crate::tests::mothers::principal_mother::PrincipalMother;
    use puerto_core::DomainEvent;

    #[tokio::test]
    async fn should_record_order_deleted_when_id_exists() {
        // Arrange
        let entity = Order::new(OrderProps {
            amount: Amount::new(1.5).expect("valid Amount"),
            status: Status::Pending,
            note: None,
        })
        .unwrap();
        let entity_id = entity.id;
        let mut mock_repo = MockOrderRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = DeleteOrderUseCaseImpl {
            repository: Arc::new(mock_repo),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(DeleteOrderParams {
                actor: PrincipalMother::random(),
                id: entity_id,
            })
            .await;

        // Assert
        assert!(result.is_ok());
        let recorded = uow_factory.recorded();
        assert!(recorded.committed);
        assert!(recorded.saved.iter().all(|saved| saved.deleted));
        assert_eq!(recorded.events.len(), 1);
        assert_eq!(recorded.events[0].event_type(), "order_deleted");
        assert_eq!(recorded.events[0].aggregate_id(), entity_id.value());
    }

    #[tokio::test]
    async fn should_return_not_found_and_record_nothing_when_the_order_does_not_exist() {
        // Arrange
        let mut mock_repo = MockOrderRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let uow_factory = RecordingUnitOfWorkFactory::new();
        let use_case = DeleteOrderUseCaseImpl {
            repository: Arc::new(mock_repo),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Act
        let result = use_case
            .execute(DeleteOrderParams {
                actor: PrincipalMother::random(),
                id: OrderId::generate(),
            })
            .await;

        // Assert
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.not_found");
        assert!(uow_factory.recorded().events.is_empty());
    }
}
