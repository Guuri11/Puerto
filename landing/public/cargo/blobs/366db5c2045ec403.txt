use async_trait::async_trait;
use dominio_cliente::errores::ErrorCliente;
use dominio_cliente::modelo::IdCliente;
use dominio_cliente::repositorio::RepositorioCliente;
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsBorrarCliente {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub id: IdCliente,
}

#[async_trait]
pub trait BorrarCliente: Send + Sync {
    async fn ejecutar(&self, params: ParamsBorrarCliente) -> Result<(), ErrorCliente>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub BorrarCliente {}

        #[async_trait]
        impl BorrarCliente for BorrarCliente {
            async fn ejecutar(&self, params: ParamsBorrarCliente) -> Result<(), ErrorCliente>;
        }
    }
}

pub struct BorrarClienteImpl {
    pub repositorio: Arc<dyn RepositorioCliente>,
}

#[async_trait]
impl BorrarCliente for BorrarClienteImpl {
    async fn ejecutar(&self, params: ParamsBorrarCliente) -> Result<(), ErrorCliente> {
        tracing::info!(id = %params.id, "borrar_cliente: inicio");
        let mut entidad = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "borrar_cliente: fallo del repositorio"))?
            .ok_or_else(|| {
                let err = ErrorCliente::NoEncontrado;
                tracing::warn!(error = %err, "borrar_cliente: rechazado");
                err
            })?;
        let ahora = chrono::Utc::now();
        entidad.borrado = true;
        entidad.borrado_en = Some(ahora);
        entidad.actualizado_en = ahora;
        self.repositorio.guardar(&entidad).await.inspect_err(
            |e| tracing::error!(error = %e, "borrar_cliente: fallo del repositorio"),
        )?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_cliente::{
        modelo::IdCliente, pruebas::madres::MadreCliente,
        repositorio::dobles::MockRepositorioCliente,
    };
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    #[tokio::test]
    async fn deberia_hacer_borrado_logico_de_cliente_cuando_el_id_existe() {
        let entidad = MadreCliente::aleatorio();
        let id_entidad = entidad.id;
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        repositorio_doble.expect_guardar().returning(|_| Ok(()));
        let caso_de_uso = BorrarClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsBorrarCliente {
                actor: MadrePrincipal::aleatorio(),
                id: id_entidad,
            })
            .await;

        assert!(resultado.is_ok());
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_cuando_cliente_no_existe() {
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let caso_de_uso = BorrarClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsBorrarCliente {
                actor: MadrePrincipal::aleatorio(),
                id: IdCliente::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.no_encontrado");
    }
}
