use std::sync::Arc;

use async_trait::async_trait;

use crate::domain::product::{
    errors::ProductError,
    repository::ProductRepositoryTrait,
    use_cases::delete_product::{DeleteProductParams, DeleteProductUseCaseTrait},
};

pub struct DeleteProductUseCaseImpl {
    pub repository: Arc<dyn ProductRepositoryTrait>,
}

#[async_trait]
impl DeleteProductUseCaseTrait for DeleteProductUseCaseImpl {
    async fn execute(&self, params: DeleteProductParams) -> Result<(), ProductError> {
        tracing::info!(id = %params.id, "delete_product: start");
        let mut entity = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "delete_product: repository failed"))?
            .ok_or_else(|| {
                let err = ProductError::NotFound;
                tracing::warn!(error = %err, "delete_product: rejected");
                err
            })?;
        let now = chrono::Utc::now();
        entity.deleted = true;
        entity.deleted_at = Some(now);
        entity.updated_at = now;
        self.repository
            .save(&entity)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "delete_product: repository failed"))?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::product::{
        model::{Product, ProductId, ProductProps},
        repository::mocks::MockProductRepository,
    };

    #[tokio::test]
    async fn should_soft_delete_product_when_id_exists() {
        let entity = Product::new(ProductProps {
            name: "example".to_string(),
            price: 42,
            sku: "example".to_string(),
        })
        .unwrap();
        let entity_id = entity.id;
        let mut mock_repo = MockProductRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        mock_repo.expect_save().returning(|_| Ok(()));
        let use_case = DeleteProductUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(DeleteProductParams { id: entity_id })
            .await;

        assert!(result.is_ok());
    }

    #[tokio::test]
    async fn should_return_not_found_when_product_does_not_exist() {
        let mut mock_repo = MockProductRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let use_case = DeleteProductUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(DeleteProductParams {
                id: ProductId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "product.not_found");
    }
}
