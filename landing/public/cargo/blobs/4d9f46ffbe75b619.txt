use dominio_cliente::errores::ErrorCliente;
use poem::http::StatusCode;
use poem_openapi::payload::Json;
use puerto_web::RespuestaError;

/// Un error de cliente como su estado HTTP y su cuerpo (ADR-0007 de Puerto): los fallos de
/// validación y los objetos valor inválidos son 422 — la petición estaba bien formada (poem
/// responde 400 a una que no lo está), pero incumple una regla del dominio. Exhaustivo a propósito:
/// una variante de error nueva no compila hasta que tiene un estado.
pub fn a_respuesta_error(err: ErrorCliente) -> (StatusCode, Json<RespuestaError>) {
    let status = match &err {
        ErrorCliente::Validacion(_) => StatusCode::UNPROCESSABLE_ENTITY,
        ErrorCliente::NoEncontrado => StatusCode::NOT_FOUND,
        ErrorCliente::Duplicado => StatusCode::CONFLICT,
        ErrorCliente::Repositorio => StatusCode::INTERNAL_SERVER_ERROR,
        ErrorCliente::Desconocido => StatusCode::INTERNAL_SERVER_ERROR,
        ErrorCliente::Prohibido => StatusCode::FORBIDDEN,
        ErrorCliente::EmailInvalido => StatusCode::UNPROCESSABLE_ENTITY,
        ErrorCliente::EtiquetaInvalido => StatusCode::UNPROCESSABLE_ENTITY,
    };
    (status, Json(RespuestaError::from_codigo(err.to_string())))
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn deberia_mapear_validacion_a_422_con_su_codigo() {
        // Actuar
        let (status, Json(cuerpo)) =
            a_respuesta_error(ErrorCliente::Validacion("nombre_vacio".into()));

        // Comprobar
        assert_eq!(status, StatusCode::UNPROCESSABLE_ENTITY);
        assert_eq!(cuerpo.codigo, "cliente.validacion.nombre_vacio");
    }

    #[test]
    fn deberia_mapear_no_encontrado_a_404_con_su_codigo() {
