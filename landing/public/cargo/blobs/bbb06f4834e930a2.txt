use std::sync::Arc;

use async_trait::async_trait;
use puerto_core::Page;

use crate::domain::order::{
    errors::OrderError,
    model::Order,
    repository::OrderRepositoryTrait,
    use_cases::list_orders::{ListOrdersParams, ListOrdersUseCaseTrait},
};

pub struct ListOrdersUseCaseImpl {
    pub repository: Arc<dyn OrderRepositoryTrait>,
}

#[async_trait]
impl ListOrdersUseCaseTrait for ListOrdersUseCaseImpl {
    async fn execute(&self, params: ListOrdersParams) -> Result<Page<Order>, OrderError> {
        tracing::info!(
            limit = params.page.limit(),
            offset = params.page.offset(),
            "list_orders: start"
        );
        self.repository
            .find_all(params.page)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "list_orders: repository failed"))
            .map_err(OrderError::from)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::order::value_objects::Amount;
    use crate::domain::order::value_objects::Status;
    use crate::domain::order::{
        model::{Order, OrderProps},
        repository::mocks::MockOrderRepository,
    };
    use puerto_core::PageRequest;

    #[tokio::test]
    async fn should_return_the_page_of_orders() {
        let entities = vec![
            Order::new(OrderProps {
                amount: Amount::new(1.5).expect("valid Amount"),
                status: Status::Pending,
                note: None,
            })
            .unwrap(),
            Order::new(OrderProps {
                amount: Amount::new(1.5).expect("valid Amount"),
                status: Status::Pending,
                note: None,
            })
            .unwrap(),
        ];
        let mut mock_repo = MockOrderRepository::new();
        mock_repo
            .expect_find_all()
            .returning(move |page| Ok(Page::new(entities.clone(), 2, page)));
        let use_case = ListOrdersUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(ListOrdersParams {
                page: PageRequest::default(),
            })
            .await;

        let page = result.unwrap();
        assert_eq!(page.items.len(), 2);
        assert_eq!(page.total, 2);
    }

    #[tokio::test]
    async fn should_return_an_empty_page_when_no_orders_exist() {
        let mut mock_repo = MockOrderRepository::new();
        mock_repo
            .expect_find_all()
            .returning(|page| Ok(Page::new(vec![], 0, page)));
        let use_case = ListOrdersUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(ListOrdersParams {
                page: PageRequest::default(),
            })
            .await;

        let page = result.unwrap();
        assert!(page.items.is_empty());
        assert_eq!(page.total, 0);
    }

    #[tokio::test]
    async fn should_pass_the_requested_page_to_the_repository() {
        let mut mock_repo = MockOrderRepository::new();
        mock_repo
            .expect_find_all()
            .withf(|page| page.limit() == 5 && page.offset() == 10)
            .returning(|page| Ok(Page::new(vec![], 0, page)));
        let use_case = ListOrdersUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(ListOrdersParams {
                page: PageRequest::new(Some(5), Some(10)),
            })
            .await;

        assert_eq!(result.unwrap().offset, 10);
    }
}
