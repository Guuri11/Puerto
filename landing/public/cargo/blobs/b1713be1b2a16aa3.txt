services:
  postgres:
    image: postgres:16
    container_name: shop-postgres
    ports:
      - "${DB_HOST_PORT:-5432}:5432"
    environment:
      POSTGRES_USER: devuser
      POSTGRES_PASSWORD: password
      POSTGRES_DB: shop
    volumes:
      - postgres-data:/var/lib/postgresql/data
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U devuser -d shop"]
      interval: 5s
      timeout: 5s
      retries: 5
