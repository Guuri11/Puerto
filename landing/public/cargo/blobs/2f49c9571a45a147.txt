import type { Page, PageRequest } from "../shared/page";
import type { Producto, ProductoDraft } from "./model";

/** The port the screens call. Every method rejects with the `ProductoError` the API answered. */
export interface ProductoRepository {
  list(page: PageRequest): Promise<Page<Producto>>;
  get(id: string): Promise<Producto>;
  create(draft: ProductoDraft): Promise<Producto>;
  update(id: string, draft: ProductoDraft): Promise<Producto>;
  remove(id: string): Promise<void>;
}
