use std::sync::Arc;

use async_trait::async_trait;

use crate::domain::product::{
    errors::ProductError,
    model::Product,
    repository::ProductRepositoryTrait,
    use_cases::get_product::{GetProductParams, GetProductUseCaseTrait},
};

pub struct GetProductUseCaseImpl {
    pub repository: Arc<dyn ProductRepositoryTrait>,
}

#[async_trait]
impl GetProductUseCaseTrait for GetProductUseCaseImpl {
    async fn execute(&self, params: GetProductParams) -> Result<Product, ProductError> {
        tracing::info!(id = %params.id, "get_product: start");
        let result = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "get_product: repository failed"))?;
        result.ok_or_else(|| {
            let err = ProductError::NotFound;
            tracing::warn!(error = %err, "get_product: rejected");
            err
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::product::{
        model::{Product, ProductId, ProductProps},
        repository::mocks::MockProductRepository,
    };
    use puerto_core::RepositoryError;

    #[tokio::test]
    async fn should_return_product_when_id_exists() {
        let entity = Product::new(ProductProps {
            name: "example".to_string(),
            price: 42,
            sku: "example".to_string(),
        })
        .unwrap();
        let entity_id = entity.id;
        let mut mock_repo = MockProductRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        let use_case = GetProductUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case.execute(GetProductParams { id: entity_id }).await;

        assert!(result.is_ok());
        assert_eq!(result.unwrap().name, "example");
    }

    #[tokio::test]
    async fn should_return_not_found_when_id_does_not_exist() {
        let mut mock_repo = MockProductRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let use_case = GetProductUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(GetProductParams {
                id: ProductId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "product.not_found");
    }

    #[tokio::test]
    async fn should_return_repository_error_when_the_store_is_unavailable() {
        let mut mock_repo = MockProductRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(|_| Err(RepositoryError::Unavailable));
        let use_case = GetProductUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(GetProductParams {
                id: ProductId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "product.repository_error");
    }
}
