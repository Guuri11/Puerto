//! Las filas que necesitan los tests de Postgres —una entidad pedido válida y otra guardada—,
//! para los tests de este crate y para los de los contextos que la referencian (ADR-0023 § 5 de
//! Puerto).
use dominio_pedido::modelo::{Pedido, PropsPedido};
use dominio_pedido::unit_of_work::UnitOfWorkFactoryPedido;
use dominio_pedido::valores::Estado;
use dominio_pedido::valores::Importe;
use sqlx::PgPool;

use crate::unit_of_work::UnitOfWorkFactoryPedidoPostgres;

/// Una entidad pedido válida, sin guardar.
pub fn construir() -> Pedido {
    Pedido::new(PropsPedido {
        importe: Importe::new(1.5).expect("Importe::new rechazó el valor de ejemplo"),
        estado: Estado::Pendiente,
        nota: None,
    })
    .unwrap()
}

/// Guarda la entidad a través de la unidad de trabajo: el repositorio solo lee.
pub async fn guardar(pool: &PgPool, entidad: &Pedido) {
    let mut uow = UnitOfWorkFactoryPedidoPostgres::new(pool.clone())
        .begin()
        .await
        .unwrap();
    uow.guardar(entidad).await.unwrap();
    uow.commit().await.unwrap();
}

/// Una entidad pedido guardada.
pub async fn sembrar(pool: &PgPool) -> Pedido {
    let entidad = construir();
    guardar(pool, &entidad).await;
    entidad
}
