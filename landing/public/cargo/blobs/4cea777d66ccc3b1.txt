use async_trait::async_trait;
use puerto_core::auth::Principal;

use crate::domain::product::{errors::ProductError, model::Product};

#[derive(Debug, Clone)]
pub struct CreateProductParams {
    /// Who asks (ADR-0021). A rule that depends on the data ("only the owner edits") decides on
    /// it here and answers `Forbidden`.
    pub actor: Principal,
    pub name: String,
    pub price: i64,
    pub sku: String,
}

#[async_trait]
pub trait CreateProductUseCaseTrait: Send + Sync {
    async fn execute(&self, params: CreateProductParams) -> Result<Product, ProductError>;
}

#[cfg(any(test, feature = "test-utils"))]
pub mod mocks {
    use mockall::mock;

    use super::*;

    mock! {
        pub CreateProductUseCase {}

        #[async_trait]
        impl CreateProductUseCaseTrait for CreateProductUseCase {
            async fn execute(&self, params: CreateProductParams) -> Result<Product, ProductError>;
        }
    }
}
