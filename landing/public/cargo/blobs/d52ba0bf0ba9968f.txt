import { Route, Switch } from "wouter";
import { contarTotal, PAGINA_DE_UNO } from "../api/http";
import type { EntradaMenu } from "../compartido/ui/marco";
import { ContenedorDetalleProducto } from "../contextos/producto/contenedores/detalle-producto";
import { ContenedorFormularioProducto } from "../contextos/producto/contenedores/formulario-producto";
import { ContenedorListaProducto } from "../contextos/producto/contenedores/lista-producto";
import { PaginaNoEncontrada } from "../paginas/no-encontrada";
import { PaginaPanel } from "../paginas/panel";
import { rutasPropias } from "../rutas";

/** Dónde va la app: en la raíz o, con portada, en `[proyecto] ruta_app` (`puerto nuevo --portada`). */
export const rutaApp = "/";

export const entradasMenu: EntradaMenu[] = [
  {
    etiqueta: "Productos",
    href: "/productos",
    total: ({ api }) => contarTotal(api.GET("/productos", PAGINA_DE_UNO)),
  },
];

export function RutasApp() {
  return (
    <Switch>
      <Route path="/">
        <PaginaPanel entradas={entradasMenu} />
      </Route>
      {rutasPropias}
      <Route path="/productos">
        <ContenedorListaProducto />
      </Route>
      <Route path="/productos/nuevo">
        <ContenedorFormularioProducto />
      </Route>
      <Route path="/productos/:id/editar">
        {(params) => <ContenedorFormularioProducto id={params.id} />}
      </Route>
      <Route path="/productos/:id">
        {(params) => <ContenedorDetalleProducto id={params.id} />}
      </Route>
      <Route>
        <PaginaNoEncontrada />
      </Route>
    </Switch>
  );
}
