import type { PedidoDraft } from "./model";

/** What a screen shows for an API error: under the input it is about, or in the alert. */
export interface PedidoProblem {
  field: keyof PedidoDraft | null;
  text: string;
}

/**
 * The messages for the codes of the backend's `errores.rs`, generated from the same list. English:
 * this file is the place to translate them. A code missing here shows as it is.
 */
const TEXTS: Record<string, string> = {
  "pedido.importe_invalido": "Importe is not valid",
  "pedido.estado_invalido": "Estado is not valid",
  "pedido.duplicado": "A pedido with these values already exists",
  "pedido.no_encontrado": "This pedido no longer exists",
  "pedido.repositorio": "The pedido could not be saved, try again later",
  "pedido.desconocido": "Something went wrong",
};

/** The input each field's code is about. */
const FIELDS: Record<string, keyof PedidoDraft> = {
  "pedido.importe_invalido": "importe",
  "pedido.estado_invalido": "estado",
};

export function toProblem(code: string): PedidoProblem {
  return { field: FIELDS[code] ?? null, text: TEXTS[code] ?? code };
}
