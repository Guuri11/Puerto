use async_trait::async_trait;

use crate::domain::customer::{
    errors::CustomerError,
    model::{Customer, CustomerId},
};

#[derive(Debug, Clone)]
pub struct GetCustomerParams {
    pub id: CustomerId,
}

#[async_trait]
pub trait GetCustomerUseCaseTrait: Send + Sync {
    async fn execute(&self, params: GetCustomerParams) -> Result<Customer, CustomerError>;
}

        #[async_trait]

            async fn execute(&self, params: GetCustomerParams) -> Result<Customer, CustomerError>;
        }
    }
}
