use async_trait::async_trait;
use dominio_producto::errores::ErrorProducto;
use dominio_producto::eventos::{EventoProducto, SnapshotProducto};
use dominio_producto::modelo::IdProducto;
use dominio_producto::repositorio::RepositorioProducto;
use dominio_producto::unit_of_work::UnitOfWorkFactoryProducto;
use puerto_nucleo::ErrorRepositorio;
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsBorrarProducto {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub id: IdProducto,
}

#[async_trait]
pub trait BorrarProducto: Send + Sync {
    async fn ejecutar(&self, params: ParamsBorrarProducto) -> Result<(), ErrorProducto>;
}

pub struct BorrarProductoImpl {
    pub repositorio: Arc<dyn RepositorioProducto>,
    pub uow_factory: Arc<dyn UnitOfWorkFactoryProducto>,
}

#[async_trait]
impl BorrarProducto for BorrarProductoImpl {
    async fn ejecutar(&self, params: ParamsBorrarProducto) -> Result<(), ErrorProducto> {
        tracing::info!(id = %params.id, "borrar_producto: inicio");
        let fallido = |e: &ErrorRepositorio| tracing::error!(error = %e, "borrar_producto: fallo del repositorio");
        let mut entidad = self
            .repositorio
            .buscar_por_id(params.id)
            .await
            .inspect_err(fallido)?
            .ok_or_else(|| {
                let err = ErrorProducto::NoEncontrado;
                tracing::warn!(error = %err, "borrar_producto: rechazado");
                err
            })?;
        let ahora = chrono::Utc::now();
        entidad.borrado = true;
        entidad.borrado_en = Some(ahora);
        entidad.actualizado_en = ahora;
        let mut uow = self.uow_factory.begin().await.inspect_err(fallido)?;
        uow.guardar(&entidad).await.inspect_err(fallido)?;
        uow.registrar(EventoProducto::Borrado(SnapshotProducto::from_entidad(
            &entidad,
        )))
        .await
        .inspect_err(fallido)?;
        uow.commit().await.inspect_err(fallido)?;
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_producto::{
        modelo::IdProducto, pruebas::madres::MadreProducto,
        repositorio::dobles::MockRepositorioProducto,
        unit_of_work::dobles::UnitOfWorkFactoryGrabadora,
    };
    use puerto_nucleo::EventoDominio;
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    #[tokio::test]
    async fn deberia_registrar_el_evento_producto_borrado_cuando_el_id_existe() {
        // Preparar
        let entidad = MadreProducto::aleatorio();
        let id_entidad = entidad.id;
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = BorrarProductoImpl {
            repositorio: Arc::new(repositorio_doble),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsBorrarProducto {
                actor: MadrePrincipal::aleatorio(),
                id: id_entidad,
            })
            .await;

        // Comprobar
        assert!(resultado.is_ok());
        let grabado = uow_factory.grabado();
        assert!(grabado.confirmado);
        assert!(grabado.guardados.iter().all(|guardados| guardados.borrado));
        assert_eq!(grabado.eventos.len(), 1);
        assert_eq!(grabado.eventos[0].tipo_evento(), "producto_borrado");
        assert_eq!(grabado.eventos[0].id_agregado(), id_entidad.valor());
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_sin_registrar_nada_cuando_producto_no_existe() {
        // Preparar
        let mut repositorio_doble = MockRepositorioProducto::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = BorrarProductoImpl {
            repositorio: Arc::new(repositorio_doble),
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsBorrarProducto {
                actor: MadrePrincipal::aleatorio(),
                id: IdProducto::generar(),
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.no_encontrado");
        assert!(uow_factory.grabado().eventos.is_empty());
    }
}
