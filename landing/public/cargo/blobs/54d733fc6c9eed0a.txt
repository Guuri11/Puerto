use business::domain::order::errors::OrderError;
use poem::http::StatusCode;
use poem_openapi::payload::Json;
use puerto_web::ErrorResponse;

use crate::api::error::IntoErrorResponse;

impl IntoErrorResponse for OrderError {
    /// Exhaustive on purpose: a new error variant does not compile until it has a status.
    fn into_error_response(self) -> (StatusCode, Json<ErrorResponse>) {
        let status = match &self {
            OrderError::ValidationError(_) => StatusCode::UNPROCESSABLE_ENTITY,
            OrderError::NotFound => StatusCode::NOT_FOUND,
            OrderError::Duplicate => StatusCode::CONFLICT,
            OrderError::RepositoryError => StatusCode::INTERNAL_SERVER_ERROR,
            OrderError::Unknown => StatusCode::INTERNAL_SERVER_ERROR,
            OrderError::InvalidAmount => StatusCode::UNPROCESSABLE_ENTITY,
            OrderError::InvalidStatus => StatusCode::UNPROCESSABLE_ENTITY,
        };
        (status, Json(ErrorResponse::from_code(self.to_string())))
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn should_map_validation_error_to_422_with_its_code() {
        // Act
        let (status, Json(body)) =
            OrderError::ValidationError("name_empty".into()).into_error_response();

        // Assert
        assert_eq!(status, StatusCode::UNPROCESSABLE_ENTITY);
        assert_eq!(body.name, "order.validation_error.name_empty");
    }

    #[test]
    fn should_map_not_found_to_404_with_its_code() {

        let (status, Json(body)) = OrderError::NotFound.into_error_response();
