use async_trait::async_trait;
use dominio_cliente::errores::ErrorCliente;
use dominio_cliente::modelo::{Cliente, IdCliente};
use dominio_cliente::repositorio::RepositorioCliente;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsObtenerCliente {
    pub id: IdCliente,
}

#[async_trait]
pub trait ObtenerCliente: Send + Sync {
    async fn ejecutar(&self, params: ParamsObtenerCliente) -> Result<Cliente, ErrorCliente>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub ObtenerCliente {}

        #[async_trait]
        impl ObtenerCliente for ObtenerCliente {
            async fn ejecutar(&self, params: ParamsObtenerCliente) -> Result<Cliente, ErrorCliente>;
        }
    }
}

pub struct ObtenerClienteImpl {
    pub repositorio: Arc<dyn RepositorioCliente>,
}

#[async_trait]
impl ObtenerCliente for ObtenerClienteImpl {
    #[tracing::instrument(name = "obtener_cliente", skip_all, fields(id = %params.id), err(level = "warn"))]
    async fn ejecutar(&self, params: ParamsObtenerCliente) -> Result<Cliente, ErrorCliente> {
        self.repositorio
            .buscar_por_id(params.id)
            .await?
            .ok_or(ErrorCliente::NoEncontrado)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_cliente::{
        modelo::IdCliente, pruebas::madres::MadreCliente,
        repositorio::dobles::MockRepositorioCliente,
    };
    use puerto_nucleo::ErrorRepositorio;

    #[tokio::test]
    async fn deberia_devolver_cliente_cuando_el_id_existe() {
        let entidad = MadreCliente::aleatorio();
        let id_entidad = entidad.id;
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(move |_| Ok(Some(entidad.clone())));
        let caso_de_uso = ObtenerClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerCliente { id: id_entidad })
            .await;

        assert_eq!(resultado.unwrap().id, id_entidad);
    }

    #[tokio::test]
    async fn deberia_devolver_no_encontrado_cuando_el_id_no_existe() {
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Ok(None));
        let caso_de_uso = ObtenerClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerCliente {
                id: IdCliente::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.no_encontrado");
    }

    #[tokio::test]
    async fn deberia_devolver_error_de_repositorio_cuando_el_almacen_no_esta_disponible() {
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble
            .expect_buscar_por_id()
            .returning(|_| Err(ErrorRepositorio::NoDisponible));
        let caso_de_uso = ObtenerClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        let resultado = caso_de_uso
            .ejecutar(ParamsObtenerCliente {
                id: IdCliente::generar(),
            })
            .await;

        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.repositorio");
    }
}
