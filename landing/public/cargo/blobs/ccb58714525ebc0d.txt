use super::value_objects::Amount;
use super::value_objects::Status;
use chrono::{DateTime, Utc};
use uuid::Uuid;

use super::errors::OrderError;

/// The id of every `Order`: a `Uuid` that cannot be passed where another entity's id is expected.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct OrderId(Uuid);

impl OrderId {
    /// A new, random id — for an entity being created.
    pub fn generate() -> Self {
        Self(Uuid::new_v4())
    }

    /// An id read from the outside (a path parameter, a row) — no validation: any `Uuid` is an id.
    pub fn from_uuid(value: Uuid) -> Self {
        Self(value)
    }

    pub fn value(&self) -> Uuid {
        self.0
    }
}

impl std::fmt::Display for OrderId {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        self.0.fmt(f)
    }
}

#[derive(Debug, Clone)]
pub struct OrderProps {
    pub amount: Amount,
    pub status: Status,
    pub note: Option<String>,
}

#[derive(Debug, Clone)]
pub struct Order {
    pub id: OrderId,
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
    pub deleted: bool,
    pub deleted_at: Option<DateTime<Utc>>,
    pub amount: Amount,
    pub status: Status,
    pub note: Option<String>,
}

impl Order {
    pub fn new(props: OrderProps) -> Result<Self, OrderError> {
        let now = chrono::Utc::now();
        Ok(Self {
            id: OrderId::generate(),
            created_at: now,
            updated_at: now,
            deleted: false,
            deleted_at: None,
            amount: props.amount,
            status: props.status,
            note: props.note,
        })
    }

    pub fn from_repository(data: Order) -> Self {
        data
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn should_create_order_when_fields_are_valid() {
        let result = Order::new(OrderProps {
            amount: Amount::new(1.5).expect("valid Amount"),
            status: Status::Pending,
            note: None,
        });
        assert!(result.is_ok());
    }

    #[test]
    fn should_reject_order_when_status_is_invalid() {
        let result = "InvalidVariant".parse::<Status>();
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.invalid_status");
    }
}
