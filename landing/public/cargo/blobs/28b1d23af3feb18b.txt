import { describe, expect, mock, test } from "bun:test";
import { fireEvent, screen, waitFor } from "@testing-library/react";
import { PedidoError } from "../../../../domain/pedido/errors";
import { CurrentUserMother } from "../../../../test-support/auth";
import { fakePedidoRepository, PedidoMother } from "../../../../test-support/pedido";
import { renderScreen } from "../../../../test-support/render";
import { PedidoDetailContainer } from "./pedido-detail-container";

describe("PedidoDetailContainer", () => {
  test("should show the pedido when it exists", async () => {
    // Arrange
    const seeded = PedidoMother.random();
    const pedidoRepository = fakePedidoRepository([seeded]);

    // Act
    renderScreen(<PedidoDetailContainer id={seeded.id} />, { pedidoRepository });

    // Assert
    expect(await screen.findByText("Edit")).toBeTruthy();
  });

  test("should show the not-found code when the pedido does not exist", async () => {
    // Arrange
    const pedidoRepository = fakePedidoRepository();

    // Act
    renderScreen(<PedidoDetailContainer id={crypto.randomUUID()} />, {
      pedidoRepository,
    });

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe("pedido.no_encontrado");
  });

  test("should delete the pedido only once the deletion is confirmed", async () => {
    // Arrange
    const seeded = PedidoMother.random();
    const fake = fakePedidoRepository([seeded]);
    const remove = mock(fake.remove);
    const { history } = renderScreen(<PedidoDetailContainer id={seeded.id} />, {
      pedidoRepository: { ...fake, remove },
    });
    fireEvent.click(await screen.findByText("Delete", { selector: ".card-actions button" }));
    expect(remove).not.toHaveBeenCalled();

    // Act
    fireEvent.click(screen.getByText("Delete", { selector: ".modal-action button" }));

    // Assert
    await waitFor(() => expect(history.at(-1)).toBe("/pedidos"));
    expect(remove).toHaveBeenCalledWith(seeded.id);
  });

  test("should not offer Delete to a user without its role", async () => {
    // Arrange
    const seeded = PedidoMother.random();
    const pedidoRepository = fakePedidoRepository([seeded]);
    const user = CurrentUserMother.random();

    // Act
    renderScreen(<PedidoDetailContainer id={seeded.id} />, { pedidoRepository }, "/", { user });

    // Assert
    expect(await screen.findByText("Importe", { selector: "dt" })).toBeTruthy();
    expect(screen.queryByText("Delete", { selector: ".card-actions *" })).toBeNull();
  });

  test("should show auth.prohibido when the API refuses the deletion", async () => {
    // Arrange
    const seeded = PedidoMother.random();
    const pedidoRepository = {
      ...fakePedidoRepository([seeded]),
      remove: () => Promise.reject(new PedidoError("auth.prohibido")),
    };
    renderScreen(<PedidoDetailContainer id={seeded.id} />, { pedidoRepository });
    fireEvent.click(await screen.findByText("Delete", { selector: ".card-actions button" }));

    // Act
    fireEvent.click(screen.getByText("Delete", { selector: ".modal-action button" }));

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe("auth.prohibido");
  });
});
