use std::sync::Arc;

use async_trait::async_trait;

use crate::domain::order::{
    errors::OrderError,
    model::Order,
    repository::OrderRepositoryTrait,
    use_cases::get_order::{GetOrderParams, GetOrderUseCaseTrait},
};

pub struct GetOrderUseCaseImpl {
    pub repository: Arc<dyn OrderRepositoryTrait>,
}

#[async_trait]
impl GetOrderUseCaseTrait for GetOrderUseCaseImpl {
    async fn execute(&self, params: GetOrderParams) -> Result<Order, OrderError> {
        tracing::info!(id = %params.id, "get_order: start");
        let result = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "get_order: repository failed"))?;
        result.ok_or_else(|| {
            let err = OrderError::NotFound;
            tracing::warn!(error = %err, "get_order: rejected");
            err
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::order::value_objects::Amount;
    use crate::domain::order::value_objects::Status;
    use crate::domain::order::{
        model::{Order, OrderId, OrderProps},
        repository::mocks::MockOrderRepository,
    };
    use puerto_core::RepositoryError;

    #[tokio::test]
    async fn should_return_order_when_id_exists() {
        let entity = Order::new(OrderProps {
            amount: Amount::new(1.5).expect("valid Amount"),
            status: Status::Pending,
            note: None,
        })
        .unwrap();
        let entity_id = entity.id;
        let mut mock_repo = MockOrderRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        let use_case = GetOrderUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case.execute(GetOrderParams { id: entity_id }).await;

        assert!(result.is_ok());
    }

    #[tokio::test]
    async fn should_return_not_found_when_id_does_not_exist() {
        let mut mock_repo = MockOrderRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let use_case = GetOrderUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(GetOrderParams {
                id: OrderId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.not_found");
    }

    #[tokio::test]
    async fn should_return_repository_error_when_the_store_is_unavailable() {
        let mut mock_repo = MockOrderRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(|_| Err(RepositoryError::Unavailable));
        let use_case = GetOrderUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(GetOrderParams {
                id: OrderId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "order.repository_error");
    }
}
