use business::domain::order::model::OrderId;
use business::domain::order::use_cases::{

    delete_order::{DeleteOrderParams, DeleteOrderUseCaseTrait},
    get_order::{GetOrderParams, GetOrderUseCaseTrait},
    list_orders::{ListOrdersParams, ListOrdersUseCaseTrait},
    update_order::{UpdateOrderParams, UpdateOrderUseCaseTrait},
};
use poem_openapi::{

    param::{Path, Query},
    payload::Json,
};
use puerto_core::PageRequest;
use puerto_web::auth::BearerAuth;
use puerto_web::require_role;

use crate::api::error::IntoErrorResponse;
use crate::api::order::dto::{CreateOrderRequest, OrderDto, OrderPageDto, UpdateOrderRequest};
use crate::api::order::responses::{
    CreateOrderResponse, DeleteOrderResponse, GetOrderResponse, ListOrdersResponse,
    UpdateOrderResponse,
};
use crate::api::tags::ApiTags;

/// Holds the use case ports; `generated/bootstrap.rs` plugs in the implementations.
pub struct OrderApi {
    pub create_order: Arc<dyn CreateOrderUseCaseTrait>,
    pub get_order: Arc<dyn GetOrderUseCaseTrait>,
    pub list_orders: Arc<dyn ListOrdersUseCaseTrait>,
    pub update_order: Arc<dyn UpdateOrderUseCaseTrait>,
    pub delete_order: Arc<dyn DeleteOrderUseCaseTrait>,
}

#[OpenApi]
impl OrderApi {
    /// Create a new Order
    #[oai(path = "/orders", method = "post", tag = "ApiTags::Order")]
    async fn create(
        &self,
        auth: BearerAuth,
        body: Json<CreateOrderRequest>,
    ) -> CreateOrderResponse {
        match self
            .create_order
            .execute(CreateOrderParams {
                actor: auth.0,
                amount: body.amount,
                status: body.status.clone(),
                note: body.note.clone(),
            })
            .await
        {
            Ok(entity) => CreateOrderResponse::Created(Json(OrderDto::from_domain(&entity))),
            Err(err) => {
                let (status, error) = err.into_error_response();
                tracing::warn!(code = %error.0.name, "create_order failed");
                CreateOrderResponse::from_status(status, error)
            }
        }
    }

    /// Get a Order by ID
    #[oai(path = "/orders/:id", method = "get", tag = "ApiTags::Order")]
    async fn get_by_id(&self, id: Path<Uuid>) -> GetOrderResponse {
        match self
            .get_order
            .execute(GetOrderParams {
                id: OrderId::from_uuid(id.0),
            })
            .await
        {
            Ok(entity) => GetOrderResponse::Ok(Json(OrderDto::from_domain(&entity))),
            Err(err) => {
                let (status, error) = err.into_error_response();
                tracing::warn!(code = %error.0.name, "get_order failed");
                GetOrderResponse::from_status(status, error)
            }
        }
    }

    /// List Orders, one page at a time (`limit` is capped by the domain)
    #[oai(path = "/orders", method = "get", tag = "ApiTags::Order")]
    async fn list(
        &self,
        limit: Query<Option<u32>>,
        offset: Query<Option<u32>>,
    ) -> ListOrdersResponse {
        match self
            .list_orders
            .execute(ListOrdersParams {
                page: PageRequest::new(limit.0, offset.0),
            })
            .await
        {
            Ok(page) => ListOrdersResponse::Ok(Json(OrderPageDto::from_domain(&page))),
            Err(err) => {
                let (status, error) = err.into_error_response();
                tracing::error!(code = %error.0.name, "list_orders failed");
                ListOrdersResponse::from_status(status, error)
            }
        }
    }

    /// Update a Order
    #[oai(path = "/orders/:id", method = "put", tag = "ApiTags::Order")]
    async fn update(
        &self,
        auth: BearerAuth,
        id: Path<Uuid>,
        body: Json<UpdateOrderRequest>,
    ) -> UpdateOrderResponse {
        match self
            .update_order
            .execute(UpdateOrderParams {
                actor: auth.0,
                id: OrderId::from_uuid(id.0),
                amount: body.amount,
                status: body.status.clone(),
                note: body.note.clone(),
            })
            .await
        {
            Ok(entity) => UpdateOrderResponse::Ok(Json(OrderDto::from_domain(&entity))),
            Err(err) => {
                let (status, error) = err.into_error_response();
                tracing::warn!(code = %error.0.name, "update_order failed");
                UpdateOrderResponse::from_status(status, error)
            }
        }
    }

    /// Delete a Order
    #[oai(path = "/orders/:id", method = "delete", tag = "ApiTags::Order")]
    #[require_role("admin")]
    async fn delete(&self, auth: BearerAuth, id: Path<Uuid>) -> DeleteOrderResponse {
        match self
            .delete_order
            .execute(DeleteOrderParams {
                actor: auth.0,
                id: OrderId::from_uuid(id.0),
            })
            .await
        {
            Ok(()) => DeleteOrderResponse::NoContent,
            Err(err) => {
                let (status, error) = err.into_error_response();
                tracing::warn!(code = %error.0.name, "delete_order failed");
                DeleteOrderResponse::from_status(status, error)
            }
        }
    }
}

#[cfg(test)]
mod tests {
    use business::domain::order::errors::OrderError;
    use business::domain::order::use_cases::{
        create_order::mocks::MockCreateOrderUseCase, delete_order::mocks::MockDeleteOrderUseCase,
        get_order::mocks::MockGetOrderUseCase, list_orders::mocks::MockListOrdersUseCase,
        update_order::mocks::MockUpdateOrderUseCase,
    };
    use business::tests::mothers::order_mother::OrderMother;
    use business::tests::mothers::principal_mother::PrincipalMother;

    use super::*;

    /// One mock per use case. A test sets expectations only on the one its route should call: a
    /// route that reaches for another use case panics instead of passing.
    #[derive(Default)]
    struct Mocks {
        create: MockCreateOrderUseCase,
        get: MockGetOrderUseCase,
        list: MockListOrdersUseCase,
        update: MockUpdateOrderUseCase,
        delete: MockDeleteOrderUseCase,
    }

    /// Accepts `caller` (every role a route of this API requires) and `nobody` (no role).
    fn authenticator() -> Arc<dyn Authenticator> {
        let mut authenticator = MockAuthenticator::new();
        authenticator
            .expect_authenticate()
            .returning(|token| match token {
                "caller" => Ok(PrincipalMother::new().with_roles(["admin"]).build()),
                "nobody" => Ok(PrincipalMother::random()),
                _ => Err(AuthError::Invalid),
            });
        Arc::new(authenticator)
    }

    impl Mocks {
        /// Every request signed in as `caller`.
        fn client(self) -> TestClient<impl poem::Endpoint> {
            self.client_with(Some("Bearer caller"))
        }

        /// `authorization`: the header every request sends; `None` sends none.
        fn client_with(self, authorization: Option<&str>) -> TestClient<impl poem::Endpoint> {
            let api = OrderApi {
                create_order: Arc::new(self.create),
                get_order: Arc::new(self.get),
                list_orders: Arc::new(self.list),
                update_order: Arc::new(self.update),
                delete_order: Arc::new(self.delete),
            };
            let client = TestClient::new(
                Route::new()
                    .nest("/", OpenApiService::new(api, "test", "1.0"))
                    .data(authenticator()),
            );
            match authorization {
                Some(value) => client.default_header("Authorization", value),
                None => client,
            }
        }
    }

    fn create_request() -> CreateOrderRequest {
        CreateOrderRequest {
            amount: 1.5,
            status: "example".to_string(),
            note: None,
        }
    }

    fn update_request() -> UpdateOrderRequest {
        UpdateOrderRequest {
            amount: 1.5,
            status: "example".to_string(),
            note: None,
        }
    }

    fn item_path() -> String {
        format!("/orders/{}", Uuid::new_v4())
    }

    #[tokio::test]
    async fn should_respond_201_when_order_is_created() {
        // Arrange
        let mut mocks = Mocks::default();
        mocks
            .create
            .expect_execute()
            .returning(|_| Ok(OrderMother::random()));

        // Act
        let response = mocks
            .client()
            .post("/orders")
            .body_json(&create_request().to_json())
            .send()
            .await;

        // Assert
        response.assert_status(StatusCode::CREATED);
    }

    async fn should_respond_200_with_a_page_of_orders_when_listing() {
        // Arrange
        let mut mocks = Mocks::default();
        mocks
            .list
            .expect_execute()
            .returning(|params| Ok(Page::new(OrderMother::random_vec(2), 7, params.page)));

        // Act
        let response = mocks.client().get("/orders").send().await;

        // Assert
        response.assert_status(StatusCode::OK);
        let json = response.json().await;
        let body = json.value().object();
        body.get("items").array().assert_len(2);
        body.get("total").assert_i64(7);
    }

            .list

        // Act
        let response = mocks.client().get("/orders?limit=5&offset=10").send().await;

        // Assert
        response.assert_status(StatusCode::OK);
