import { describe, expect, test } from "bun:test";
import { fireEvent, screen, waitFor } from "@testing-library/react";
import { ProductoError } from "../../../../domain/producto/errors";
import type { ProductoDraft } from "../../../../domain/producto/model";
import { fakeProductoRepository, ProductoMother } from "../../../../test-support/producto";
import { renderScreen } from "../../../../test-support/render";
import { ProductoFormContainer } from "./producto-form-container";

function type(label: string, value: string) {
  fireEvent.change(screen.getByLabelText(label), { target: { value } });
}

/** Fills the empty form with `draft`, the way a user would. */
async function fillIn(draft: ProductoDraft) {
  await screen.findByText("Save");
  type("Nombre", draft.nombre);
  type("Precio", String(draft.precio));
  type("Sku", draft.sku);
}

describe("ProductoFormContainer", () => {
  test("should create the producto from what was typed and show it", async () => {
    // Arrange
    const draft = ProductoMother.draft();
    const productoRepository = fakeProductoRepository();
    const { history } = renderScreen(<ProductoFormContainer />, { productoRepository });
    await fillIn(draft);

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    await waitFor(() => expect(history.at(-1)).toStartWith("/productos/"));
    const {
      items: [created],
    } = await productoRepository.list({ limit: 1, offset: 0 });
    expect(created).toMatchObject(draft);
    expect(history.at(-1)).toBe(`/productos/${created?.id}`);
  });

  test("should save the edited producto when an existing one is submitted", async () => {
    // Arrange
    const seeded = ProductoMother.random();
    const productoRepository = fakeProductoRepository([seeded]);
    const { history } = renderScreen(<ProductoFormContainer id={seeded.id} />, {
      productoRepository,
    });

    // Act
    fireEvent.click(await screen.findByText("Save"));

    // Assert
    await waitFor(() => expect(history.at(-1)).toBe(`/productos/${seeded.id}`));
    expect(await productoRepository.get(seeded.id)).toEqual(seeded);
  });

  test("should show the code itself when the API answers one the form does not know", async () => {
    // Arrange
    const code = "producto.validacion.invalid";
    const productoRepository = {
      ...fakeProductoRepository(),
      create: () => Promise.reject(new ProductoError(code)),
    };
    renderScreen(<ProductoFormContainer />, { productoRepository });
    await fillIn(ProductoMother.draft());

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe(code);
  });

  test("should say a producto with these values exists when the API answers a duplicate", async () => {
    // Arrange
    const productoRepository = {
      ...fakeProductoRepository(),
      create: () => Promise.reject(new ProductoError("producto.duplicado")),
    };
    renderScreen(<ProductoFormContainer />, { productoRepository });
    await fillIn(ProductoMother.draft());

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe(
      "A producto with these values already exists",
    );
  });

  test("should show the message under its input when the API rejects one field", async () => {
    // Arrange
    const productoRepository = {
      ...fakeProductoRepository(),
      create: () => Promise.reject(new ProductoError("producto.validacion.nombre_vacio")),
    };
    renderScreen(<ProductoFormContainer />, { productoRepository });
    await fillIn(ProductoMother.draft());

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    expect(await screen.findByText("Nombre is required")).toBeTruthy();
    const input = screen.getByLabelText("Nombre");
    expect(input.getAttribute("aria-invalid")).toBe("true");
    expect(screen.queryByRole("alert")).toBeNull();
  });
});
