use super::value_objects::Email;
use super::value_objects::Tag;
use chrono::{DateTime, Utc};
use uuid::Uuid;

use super::errors::CustomerError;

/// The id of every `Customer`: a `Uuid` that cannot be passed where another entity's id is expected.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct CustomerId(Uuid);

impl CustomerId {
    /// A new, random id — for an entity being created.
    pub fn generate() -> Self {
        Self(Uuid::new_v4())
    }

    /// An id read from the outside (a path parameter, a row) — no validation: any `Uuid` is an id.
    pub fn from_uuid(value: Uuid) -> Self {
        Self(value)
    }

    pub fn value(&self) -> Uuid {
        self.0
    }
}

impl std::fmt::Display for CustomerId {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        self.0.fmt(f)
    }
}

#[derive(Debug, Clone)]
pub struct CustomerProps {
    pub email: Email,
    pub name: String,
    pub tags: Vec<Tag>,
}

#[derive(Debug, Clone)]
pub struct Customer {
    pub id: CustomerId,
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
    pub deleted: bool,
    pub deleted_at: Option<DateTime<Utc>>,
    pub email: Email,
    pub name: String,
    pub tags: Vec<Tag>,
}

impl Customer {
    pub fn new(props: CustomerProps) -> Result<Self, CustomerError> {
        if props.name.trim().is_empty() {
            return Err(CustomerError::ValidationError("name_empty".into()));
        }
        let now = chrono::Utc::now();
        Ok(Self {
            id: CustomerId::generate(),
            created_at: now,
            updated_at: now,
            deleted: false,
            deleted_at: None,
            email: props.email,
            name: props.name,
            tags: props.tags,
        })
    }

    pub fn from_repository(data: Customer) -> Self {
        data
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn should_create_customer_when_fields_are_valid() {
        let result = Customer::new(CustomerProps {
            email: Email::new("example".to_string()).expect("valid Email"),
            name: "example".to_string(),
            tags: vec![],
        });
        assert_eq!(result.unwrap().name, "example");
    }

    #[test]
    fn should_reject_customer_when_name_is_empty() {
        let result = Customer::new(CustomerProps {
            email: Email::new("example".to_string()).expect("valid Email"),
            name: "".into(),
            tags: vec![],
        });
        assert!(result.is_err());
        assert_eq!(
            result.unwrap_err().to_string(),
            "customer.validation_error.name_empty"
        );
    }

    #[test]
    fn should_reject_customer_when_name_is_only_whitespace() {
        let result = Customer::new(CustomerProps {
            email: Email::new("example".to_string()).expect("valid Email"),
            name: "   ".into(),
            tags: vec![],
        });
        assert!(result.is_err());
    }

    #[test]
    fn should_reject_customer_when_email_is_empty() {
        let result = Email::new("".to_string());
        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "customer.invalid_email");
    }
}
