use async_trait::async_trait;
use sqlx::PgPool;

use business::domain::product::{
    events::ProductEvent,
    model::Product,
    unit_of_work::{ProductUnitOfWork, ProductUnitOfWorkFactory},
};
use puerto_core::RepositoryError;
use puerto_sqlx::uow::PgTransaction;

use super::entity::ProductDb;
use crate::db::map_sqlx_error;

/// Opens a Postgres transaction per write: the product row and its outbox event commit together.
pub struct PgProductUnitOfWorkFactory {
    pub pool: PgPool,
}

impl PgProductUnitOfWorkFactory {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }
}

#[async_trait]
impl ProductUnitOfWorkFactory for PgProductUnitOfWorkFactory {
    async fn begin(&self) -> Result<Box<dyn ProductUnitOfWork>, RepositoryError> {
        let tx = PgTransaction::begin(&self.pool).await?;
        Ok(Box::new(PgProductUnitOfWork { tx }))
    }
}

struct PgProductUnitOfWork {
    tx: PgTransaction,
}

#[async_trait]
impl ProductUnitOfWork for PgProductUnitOfWork {
    async fn save(&mut self, entity: &Product) -> Result<(), RepositoryError> {
        tracing::debug!(id = %entity.id, "save");
        let db = ProductDb::from(entity);
        sqlx::query!(
            "INSERT INTO products (id, created_at, updated_at, deleted, deleted_at, name, price, sku)
             VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
             ON CONFLICT (id) DO UPDATE
             SET updated_at = $3, deleted = $4, deleted_at = $5, name = $6, price = $7, sku = $8",
            db.id,
            db.created_at,
            db.updated_at,
            db.deleted,
            db.deleted_at,
            db.name,
            db.price,
            db.sku,
        )
        .execute(self.tx.tx())
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "save failed");
            map_sqlx_error(&e)
        })?;
        Ok(())
    }

    async fn record(&mut self, event: ProductEvent) -> Result<(), RepositoryError> {
        self.tx.record(&event).await
    }

    async fn commit(self: Box<Self>) -> Result<(), RepositoryError> {
        self.tx.commit().await
    }
}

#[cfg(test)]
mod integration_tests {
    use super::*;
    use business::domain::product::{
        events::ProductSnapshot, model::ProductProps, repository::ProductRepositoryTrait,
    };

    use crate::product::repository::PgProductRepository;
    use uuid::Uuid;

    fn new_product() -> Product {
        Product::new(ProductProps {
            name: "example".to_string(),
            price: 42,
            sku: Uuid::new_v4().to_string(),
        })
        .unwrap()
    }

    /// What a create use case does: save, record, commit.
    async fn create(pool: &PgPool, entity: &Product) -> Result<(), RepositoryError> {
        let mut uow = PgProductUnitOfWorkFactory::new(pool.clone())
            .begin()
            .await?;
        uow.save(entity).await?;
        uow.record(ProductEvent::Created(ProductSnapshot::from_entity(entity)))
            .await?;
        uow.commit().await
    }

    async fn count(pool: &PgPool, sql: &str) -> i64 {
        sqlx::query_scalar(sql).fetch_one(pool).await.unwrap()
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_write_the_product_and_its_event_when_committed(pool: PgPool) {
        // Arrange
        let entity = new_product();

        // Act
        create(&pool, &entity).await.unwrap();

        // Assert
        let found = PgProductRepository::new(pool.clone())
            .find_by_id(entity.id)
            .await
            .unwrap();
        assert!(found.is_some());
        let (event_type, aggregate_id): (String, uuid::Uuid) =
            sqlx::query_as("SELECT event_type, aggregate_id FROM outbox")
                .fetch_one(&pool)
                .await
                .unwrap();
        assert_eq!(event_type, "product_created");
        assert_eq!(aggregate_id, entity.id.value());
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_write_nothing_when_dropped_without_commit(pool: PgPool) {
        // Arrange
        let entity = new_product();

        // Act
        {
            let mut uow = PgProductUnitOfWorkFactory::new(pool.clone())
                .begin()
                .await
                .unwrap();
            uow.save(&entity).await.unwrap();
            uow.record(ProductEvent::Created(ProductSnapshot::from_entity(&entity)))
                .await
                .unwrap();
        }

        // Assert
        assert_eq!(count(&pool, "SELECT COUNT(*) FROM products").await, 0);
        assert_eq!(count(&pool, "SELECT COUNT(*) FROM outbox").await, 0);
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_update_the_product_when_saved_again(pool: PgPool) {
        // Arrange
        let mut entity = new_product();
        create(&pool, &entity).await.unwrap();
        entity.name = "updated".to_string();
        entity.updated_at = chrono::Utc::now();

        // Act
        let mut uow = PgProductUnitOfWorkFactory::new(pool.clone())
            .begin()
            .await
            .unwrap();
        uow.save(&entity).await.unwrap();
        uow.commit().await.unwrap();

        // Assert
        let found = PgProductRepository::new(pool)
            .find_by_id(entity.id)
            .await
            .unwrap()
            .unwrap();
        assert_eq!(found.name, "updated");
    }

    #[sqlx::test(migrations = "./migrations")]
    async fn should_write_neither_row_nor_event_when_the_sku_is_taken(pool: PgPool) {
        // Arrange
        let first = new_product();
        create(&pool, &first).await.unwrap();
        let mut second = new_product();
        second.sku = first.sku.clone();

        // Act
        let result = create(&pool, &second).await;

        // Assert
        assert_eq!(result.unwrap_err(), RepositoryError::Duplicate);
        assert_eq!(count(&pool, "SELECT COUNT(*) FROM products").await, 1);
        assert_eq!(count(&pool, "SELECT COUNT(*) FROM outbox").await, 1);
    }
}
