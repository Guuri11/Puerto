use business::domain::product::errors::ProductError;
use poem::http::StatusCode;
use poem_openapi::payload::Json;
use puerto_web::ErrorResponse;

use crate::api::error::IntoErrorResponse;

impl IntoErrorResponse for ProductError {
    /// Exhaustive on purpose: a new error variant does not compile until it has a status.
    fn into_error_response(self) -> (StatusCode, Json<ErrorResponse>) {
        let status = match &self {
            ProductError::ValidationError(_) => StatusCode::UNPROCESSABLE_ENTITY,
            ProductError::NotFound => StatusCode::NOT_FOUND,
            ProductError::Duplicate => StatusCode::CONFLICT,
            ProductError::RepositoryError => StatusCode::INTERNAL_SERVER_ERROR,
            ProductError::Unknown => StatusCode::INTERNAL_SERVER_ERROR,
        };
        (status, Json(ErrorResponse::from_code(self.to_string())))
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn should_map_validation_error_to_422_with_its_code() {
        // Act
        let (status, Json(body)) =
            ProductError::ValidationError("name_empty".into()).into_error_response();

        // Assert
        assert_eq!(status, StatusCode::UNPROCESSABLE_ENTITY);
        assert_eq!(body.name, "product.validation_error.name_empty");
    }

    #[test]
    fn should_map_not_found_to_404_with_its_code() {
        // Act
        let (status, Json(body)) = ProductError::NotFound.into_error_response();

        // Assert
        assert_eq!(status, StatusCode::NOT_FOUND);
        assert_eq!(body.name, "product.not_found");
    }

    #[test]
    fn should_map_duplicate_to_409_with_its_code() {
        // Act
        let (status, Json(body)) = ProductError::Duplicate.into_error_response();

        // Assert
        assert_eq!(status, StatusCode::CONFLICT);
        assert_eq!(body.name, "product.duplicate");
    }

    #[test]
    fn should_map_repository_error_to_500_with_its_code() {
        // Act
        let (status, Json(body)) = ProductError::RepositoryError.into_error_response();

        // Assert
        assert_eq!(status, StatusCode::INTERNAL_SERVER_ERROR);
        assert_eq!(body.name, "product.repository_error");
    }

    #[test]
    fn should_map_unknown_to_500_with_its_code() {
        // Act
        let (status, Json(body)) = ProductError::Unknown.into_error_response();

        // Assert
        assert_eq!(status, StatusCode::INTERNAL_SERVER_ERROR);
        assert_eq!(body.name, "product.unknown");
    }
}
