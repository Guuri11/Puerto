use async_trait::async_trait;
use dominio_cliente::errores::ErrorCliente;
use dominio_cliente::modelo::{Cliente, PropsCliente};
use dominio_cliente::repositorio::RepositorioCliente;
use dominio_cliente::valores::{Email, Etiqueta};
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsCrearCliente {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub email: String,
    pub nombre: String,
    pub etiquetas: Vec<String>,
}

#[async_trait]
pub trait CrearCliente: Send + Sync {
    async fn ejecutar(&self, params: ParamsCrearCliente) -> Result<Cliente, ErrorCliente>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub CrearCliente {}

        #[async_trait]
        impl CrearCliente for CrearCliente {
            async fn ejecutar(&self, params: ParamsCrearCliente) -> Result<Cliente, ErrorCliente>;
        }
    }
}

pub struct CrearClienteImpl {
    pub repositorio: Arc<dyn RepositorioCliente>,
}

#[async_trait]
impl CrearCliente for CrearClienteImpl {
    async fn ejecutar(&self, params: ParamsCrearCliente) -> Result<Cliente, ErrorCliente> {
        tracing::info!("crear_cliente: inicio");
        let email = Email::new(params.email)?;
        let etiquetas: Vec<Etiqueta> = params
            .etiquetas
            .into_iter()
            .map(Etiqueta::new)
            .collect::<Result<Vec<_>, _>>()?;
        let entidad = Cliente::new(PropsCliente {
            email,
            nombre: params.nombre,
            etiquetas,
        })
        .inspect_err(|e| tracing::warn!(error = %e, "crear_cliente: rechazado"))?;
        self.repositorio
            .guardar(&entidad)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "crear_cliente: fallo del repositorio"))?;
        tracing::info!(id = %entidad.id, "crear_cliente: fin");
        Ok(entidad)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_cliente::repositorio::dobles::MockRepositorioCliente;
    use puerto_nucleo::ErrorRepositorio;
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    #[tokio::test]
    async fn deberia_crear_cliente_cuando_los_campos_son_validos() {
        // Preparar
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble.expect_guardar().returning(|_| Ok(()));
        let caso_de_uso = CrearClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearCliente {
                actor: MadrePrincipal::aleatorio(),
                email: "example".to_string(),
                nombre: "example".to_string(),
                etiquetas: vec![],
            })
            .await;

        // Comprobar
        assert!(resultado.is_ok());
        assert_eq!(resultado.unwrap().nombre, "example");
    }

    #[tokio::test]
    async fn deberia_devolver_duplicado_cuando_el_almacen_rechaza_un_duplicado() {
        // Preparar
        let mut repositorio_doble = MockRepositorioCliente::new();
        repositorio_doble
            .expect_guardar()
            .returning(|_| Err(ErrorRepositorio::Duplicado));
        let caso_de_uso = CrearClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearCliente {
                actor: MadrePrincipal::aleatorio(),
                email: "example".to_string(),
                nombre: "example".to_string(),
                etiquetas: vec![],
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.duplicado");
    }

    #[tokio::test]
    async fn deberia_devolver_error_cuando_nombre_es_una_cadena_vacia() {
        // Preparar
        let repositorio_doble = MockRepositorioCliente::new();
        let caso_de_uso = CrearClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearCliente {
                actor: MadrePrincipal::aleatorio(),
                email: "example".to_string(),
                nombre: "".to_string(),
                etiquetas: vec![],
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(
            resultado.unwrap_err().to_string(),
            "cliente.validacion.nombre_vacio"
        );
    }

    #[tokio::test]
    async fn deberia_devolver_error_cuando_email_es_una_cadena_vacia() {
        // Preparar
        let repositorio_doble = MockRepositorioCliente::new();
        let caso_de_uso = CrearClienteImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearCliente {
                actor: MadrePrincipal::aleatorio(),
                email: "".to_string(),
                nombre: "example".to_string(),
                etiquetas: vec![],
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "cliente.email_invalido");
