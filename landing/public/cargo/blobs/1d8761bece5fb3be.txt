import "./index.css";

import { StrictMode } from "react";
import { createRoot } from "react-dom/client";
import { Route, Switch } from "wouter";
import { ProveedorAuth } from "./auth/auth";
import { MarcoAcceso } from "./compartido/ui/marco-acceso";
import { ProveedorDependencias } from "./dependencias";
import { crearDependencias } from "./generado/dependencias";
import { MarcoApp, RutasApp } from "./generado/rutas";
import { marca } from "./marca/marca";
import { PaginaLogin } from "./paginas/login";
import { PaginaRegistro } from "./paginas/registro";

const raiz = document.getElementById("root");
if (raiz === null) {
  throw new Error("index.html no tiene el elemento #root");
}
document.title = marca.nombre;

createRoot(raiz).render(
  <StrictMode>
    <ProveedorDependencias valor={crearDependencias()}>
      <ProveedorAuth>
        <Switch>
          <Route path="/login">
            <MarcoAcceso>
              <PaginaLogin />
            </MarcoAcceso>
          </Route>
          <Route path="/registro">
            <MarcoAcceso>
              <PaginaRegistro />
            </MarcoAcceso>
          </Route>
          <Route>
            <MarcoApp>
              <RutasApp />
            </MarcoApp>
          </Route>
        </Switch>
      </ProveedorAuth>
    </ProveedorDependencias>
  </StrictMode>,
);
