use async_trait::async_trait;
use sqlx::PgPool;

use dominio_cliente::{
    modelo::{Cliente, IdCliente},
    repositorio::RepositorioCliente,
};
use puerto_nucleo::{ErrorRepositorio, Pagina, PeticionPagina};
use puerto_sqlx::mapear_error_sqlx;

use crate::fila::FilaCliente;

pub struct RepositorioClientePostgres {
    pub pool: PgPool,
}

impl RepositorioClientePostgres {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }
}

/// Una fila como entidad del dominio. Una fila que el dominio rechaza es un dato guardado que se ha
/// estropeado, no culpa de quien llama: va al log con el motivo del dominio y se devuelve como
/// `DatosInvalidos`.
fn to_dominio(fila: FilaCliente) -> Result<Cliente, ErrorRepositorio> {
    Cliente::try_from(fila).map_err(|e| {
        tracing::error!(error = %e, "la fila guardada no es una entidad cliente válida");
        ErrorRepositorio::DatosInvalidos
    })
}

#[async_trait]
impl RepositorioCliente for RepositorioClientePostgres {
    async fn buscar_todos(
        &self,
        pagina: PeticionPagina,
    ) -> Result<Pagina<Cliente>, ErrorRepositorio> {
        tracing::debug!(
            limite = pagina.limite(),
            desplazamiento = pagina.desplazamiento(),
            "buscar_todos"
        );
        let filas = sqlx::query_as!(
            FilaCliente,
            "SELECT id, creado_en, actualizado_en, borrado, borrado_en, email, nombre, etiquetas
             FROM clientes WHERE borrado = false
             ORDER BY creado_en DESC, id LIMIT $1 OFFSET $2",
            i64::from(pagina.limite()),
            i64::from(pagina.desplazamiento())
        )
        .fetch_all(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "buscar_todos falló");
            mapear_error_sqlx(&e)
        })?;
        let total = sqlx::query_scalar!(
            r#"SELECT COUNT(*) AS "total!" FROM clientes WHERE borrado = false"#
        )
        .fetch_one(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "falló el COUNT de buscar_todos");
            mapear_error_sqlx(&e)
        })?;

        let elementos = filas
            .into_iter()
            .map(to_dominio)
            .collect::<Result<Vec<_>, _>>()?;
        // COUNT(*) nunca es negativo.
        let total = u64::try_from(total).unwrap_or_default();
        Ok(Pagina::new(elementos, total, pagina))
    }

    async fn buscar_por_id(&self, id: IdCliente) -> Result<Option<Cliente>, ErrorRepositorio> {
        tracing::debug!(%id, "buscar_por_id");
        let fila = sqlx::query_as!(
            FilaCliente,
            "SELECT id, creado_en, actualizado_en, borrado, borrado_en, email, nombre, etiquetas
             FROM clientes WHERE id = $1 AND borrado = false",
            id.valor()
        )
        .fetch_optional(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "buscar_por_id falló");
            mapear_error_sqlx(&e)
        })?;

        fila.map(to_dominio).transpose()
    }

    async fn guardar(&self, entidad: &Cliente) -> Result<(), ErrorRepositorio> {
        tracing::debug!(id = %entidad.id, "guardar");
        let db = FilaCliente::from(entidad);
        sqlx::query!(
            "INSERT INTO clientes (id, creado_en, actualizado_en, borrado, borrado_en, email, nombre, etiquetas)
             VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
             ON CONFLICT (id) DO UPDATE
             SET actualizado_en = $3, borrado = $4, borrado_en = $5, email = $6, nombre = $7, etiquetas = $8",
            db.id,
            db.creado_en,
            db.actualizado_en,
            db.borrado,
            db.borrado_en,
            db.email,
            db.nombre,
            &db.etiquetas,
        )
        .execute(&self.pool)
        .await
        .map_err(|e| {
            tracing::error!(error = %e, "guardar falló");
            mapear_error_sqlx(&e)
        })?;
        Ok(())
    }
}

#[cfg(test)]
mod pruebas_integracion {
    use super::*;
    use crate::pruebas::siembra::{construir, sembrar};

    fn repositorio_de_prueba(pool: PgPool) -> RepositorioClientePostgres {
        RepositorioClientePostgres::new(pool)
    }

    #[sqlx::test(migrator = "migrador::MIGRADOR")]
    async fn deberia_guardar_y_recuperar_por_id(pool: PgPool) {
        // Preparar
        let entidad = sembrar(&pool).await;

        // Actuar
        let encontrado = repositorio_de_prueba(pool)
            .buscar_por_id(entidad.id)
            .await
            .unwrap()
            .unwrap();

        // Comprobar
        assert_eq!(encontrado.id, entidad.id);
        assert_eq!(encontrado.email, entidad.email);
        assert_eq!(encontrado.nombre, entidad.nombre);
        assert_eq!(encontrado.etiquetas, entidad.etiquetas);
        assert!(!encontrado.borrado);
    }

    #[sqlx::test(migrator = "migrador::MIGRADOR")]
    async fn deberia_devolver_none_cuando_el_id_no_existe(pool: PgPool) {
        // Actuar
        let resultado = repositorio_de_prueba(pool)
            .buscar_por_id(IdCliente::generar())
            .await
            .unwrap();

        // Comprobar
        assert!(resultado.is_none());
    }

    #[sqlx::test(migrator = "migrador::MIGRADOR")]
    async fn deberia_actualizar_la_entidad_al_guardar_un_id_existente(pool: PgPool) {
        // Preparar
        let mut entidad = sembrar(&pool).await;
        entidad.nombre = "updated".to_string();
        entidad.actualizado_en = chrono::Utc::now();

        // Actuar
        repositorio_de_prueba(pool.clone())
            .guardar(&entidad)
            .await
            .unwrap();

        // Comprobar
        let encontrado = repositorio_de_prueba(pool)
            .buscar_por_id(entidad.id)
            .await
            .unwrap()
            .unwrap();
        assert_eq!(encontrado.nombre, "updated");
    }

    #[sqlx::test(migrator = "migrador::MIGRADOR")]
    async fn deberia_listar_clientes_sin_las_filas_borradas(pool: PgPool) {
        // Preparar
        sembrar(&pool).await;
        let mut borrado = sembrar(&pool).await;
        borrado.borrado = true;
        borrado.borrado_en = Some(borrado.actualizado_en);
        repositorio_de_prueba(pool.clone())
            .guardar(&borrado)
            .await
            .unwrap();

        // Actuar
        let pagina = repositorio_de_prueba(pool)
            .buscar_todos(PeticionPagina::default())
            .await
            .unwrap();

        // Comprobar
        assert_eq!(pagina.elementos.len(), 1);
        assert_eq!(pagina.total, 1);
    }

    #[sqlx::test(migrator = "migrador::MIGRADOR")]
    async fn deberia_devolver_la_pagina_pedida_de_una_lista_mayor(pool: PgPool) {
        // Preparar
        for _ in 0..3 {
            sembrar(&pool).await;
        }

        // Actuar
        let pagina = repositorio_de_prueba(pool)
            .buscar_todos(PeticionPagina::new(Some(2), Some(2)))
            .await
            .unwrap();

        // Comprobar
        assert_eq!(pagina.elementos.len(), 1);
        assert_eq!(pagina.total, 3);
        assert_eq!((pagina.limite, pagina.desplazamiento), (2, 2));
    }

    #[sqlx::test(migrator = "migrador::MIGRADOR")]
    async fn deberia_rechazar_un_duplicado_de_email(pool: PgPool) {
        // Preparar
        let primero = sembrar(&pool).await;
        let mut segundo = construir();
        segundo.email = primero.email.clone();

        // Actuar
        let resultado = repositorio_de_prueba(pool).guardar(&segundo).await;

        // Comprobar
        assert_eq!(resultado.unwrap_err(), ErrorRepositorio::Duplicado);
    }
}
