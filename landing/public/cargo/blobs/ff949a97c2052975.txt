use std::sync::Arc;

use async_trait::async_trait;

use crate::domain::customer::{
    errors::CustomerError,
    model::Customer,
    repository::CustomerRepositoryTrait,
    use_cases::get_customer::{GetCustomerParams, GetCustomerUseCaseTrait},
};

pub struct GetCustomerUseCaseImpl {
    pub repository: Arc<dyn CustomerRepositoryTrait>,
}

#[async_trait]
impl GetCustomerUseCaseTrait for GetCustomerUseCaseImpl {
    async fn execute(&self, params: GetCustomerParams) -> Result<Customer, CustomerError> {
        tracing::info!(id = %params.id, "get_customer: start");
        let result = self
            .repository
            .find_by_id(params.id)
            .await
            .inspect_err(|e| tracing::error!(error = %e, "get_customer: repository failed"))?;
        result.ok_or_else(|| {
            let err = CustomerError::NotFound;
            tracing::warn!(error = %err, "get_customer: rejected");
            err
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::domain::customer::value_objects::Email;
    use crate::domain::customer::{
        model::{Customer, CustomerId, CustomerProps},
        repository::mocks::MockCustomerRepository,
    };
    use puerto_core::RepositoryError;

    #[tokio::test]
    async fn should_return_customer_when_id_exists() {
        let entity = Customer::new(CustomerProps {
            email: Email::new("example".to_string()).expect("valid Email"),
            name: "example".to_string(),
            tags: vec![],
        })
        .unwrap();
        let entity_id = entity.id;
        let mut mock_repo = MockCustomerRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(move |_| Ok(Some(entity.clone())));
        let use_case = GetCustomerUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case.execute(GetCustomerParams { id: entity_id }).await;

        assert!(result.is_ok());
        assert_eq!(result.unwrap().name, "example");
    }

    #[tokio::test]
    async fn should_return_not_found_when_id_does_not_exist() {
        let mut mock_repo = MockCustomerRepository::new();
        mock_repo.expect_find_by_id().returning(|_| Ok(None));
        let use_case = GetCustomerUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(GetCustomerParams {
                id: CustomerId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "customer.not_found");
    }

    #[tokio::test]
    async fn should_return_repository_error_when_the_store_is_unavailable() {
        let mut mock_repo = MockCustomerRepository::new();
        mock_repo
            .expect_find_by_id()
            .returning(|_| Err(RepositoryError::Unavailable));
        let use_case = GetCustomerUseCaseImpl {
            repository: Arc::new(mock_repo),
        };

        let result = use_case
            .execute(GetCustomerParams {
                id: CustomerId::generate(),
            })
            .await;

        assert!(result.is_err());
        assert_eq!(result.unwrap_err().to_string(), "customer.repository_error");
    }
}
