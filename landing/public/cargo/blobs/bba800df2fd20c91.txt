import { describe, expect, test } from "bun:test";
import { ProductoError } from "../../domain/producto/errors";
import { ProductoMother } from "../../test-support/producto";
import { createApiClient } from "../http/client";
import { productoRepositoryHttp } from "./producto-repository-http";

/** A repository whose every request is answered with `status` and `body`. */
function answering(status: number, body?: unknown) {
  const requests: Request[] = [];
  const api = createApiClient({
    baseUrl: "http://localhost/api",
    fetch: async (request) => {
      requests.push(request);
      return body === undefined ? new Response(null, { status }) : Response.json(body, { status });
    },
  });
  return { repository: productoRepositoryHttp(api), requests };
}

describe("productoRepositoryHttp", () => {
  test("should ask for the page and return its productos", async () => {
    // Arrange
    const producto = ProductoMother.random();
    const { repository, requests } = answering(200, {
      elementos: [producto],
      total: 21,
      limite: 20,
      desplazamiento: 20,
    });

    // Act
    const listed = await repository.list({ limit: 20, offset: 20 });

    // Assert
    const url = new URL(requests[0]?.url ?? "");
    expect(url.pathname).toBe("/api/productos");
    expect(url.search).toBe("?limite=20&desplazamiento=20");
    expect(listed).toEqual({ items: [producto], total: 21, limit: 20, offset: 20 });
  });

  test("should send the draft when creating a producto", async () => {
    // Arrange
    const draft = ProductoMother.draft();
    const { repository, requests } = answering(201, { id: crypto.randomUUID(), ...draft });

    // Act
    const created = await repository.create(draft);

    // Assert
    expect(requests[0]?.method).toBe("POST");
    expect(await requests[0]?.json()).toEqual(draft);
    expect(created).toMatchObject(draft);
  });

  test("should reject with the API's error code when the producto does not exist", async () => {
    // Arrange
    const code = "producto.no_encontrado";
    const { repository } = answering(404, { codigo: code, mensaje: code });

    // Act
    const result = repository.get(crypto.randomUUID());

    // Assert
    await expect(result).rejects.toBeInstanceOf(ProductoError);
    await expect(result).rejects.toMatchObject({ code });
  });

  test("should resolve when the API deletes the producto", async () => {
    // Arrange
    const { repository, requests } = answering(204);

    // Act
    await repository.remove("42");

    // Assert
    expect(requests[0]?.url).toBe("http://localhost/api/productos/42");
  });
});
