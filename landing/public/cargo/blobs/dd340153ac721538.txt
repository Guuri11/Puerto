use async_trait::async_trait;
use dominio_producto::errores::ErrorProducto;
use dominio_producto::eventos::{EventoProducto, SnapshotProducto};
use dominio_producto::modelo::{Producto, PropsProducto};
use dominio_producto::unit_of_work::UnitOfWorkFactoryProducto;
use puerto_nucleo::ErrorRepositorio;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsCrearProducto {
    pub nombre: String,
    pub precio: i64,
    pub sku: String,
}

#[async_trait]
pub trait CrearProducto: Send + Sync {
    async fn ejecutar(&self, params: ParamsCrearProducto) -> Result<Producto, ErrorProducto>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub CrearProducto {}

        #[async_trait]
        impl CrearProducto for CrearProducto {
            async fn ejecutar(&self, params: ParamsCrearProducto) -> Result<Producto, ErrorProducto>;
        }
    }
}

pub struct CrearProductoImpl {
    pub uow_factory: Arc<dyn UnitOfWorkFactoryProducto>,
}

#[async_trait]
impl CrearProducto for CrearProductoImpl {
    async fn ejecutar(&self, params: ParamsCrearProducto) -> Result<Producto, ErrorProducto> {
        tracing::info!("crear_producto: inicio");
        let fallido = |e: &ErrorRepositorio| tracing::error!(error = %e, "crear_producto: fallo del repositorio");
        let entidad = Producto::new(PropsProducto {
            nombre: params.nombre,
            precio: params.precio,
            sku: params.sku,
        })
        .inspect_err(|e| tracing::warn!(error = %e, "crear_producto: rechazado"))?;
        let mut uow = self.uow_factory.begin().await.inspect_err(fallido)?;
        uow.guardar(&entidad).await.inspect_err(fallido)?;
        uow.registrar(EventoProducto::Creado(SnapshotProducto::from_entidad(
            &entidad,
        )))
        .await
        .inspect_err(fallido)?;
        uow.commit().await.inspect_err(fallido)?;
        tracing::info!(id = %entidad.id, "crear_producto: fin");
        Ok(entidad)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_producto::unit_of_work::dobles::{FallarEn, UnitOfWorkFactoryGrabadora};
    use puerto_nucleo::EventoDominio;

    #[tokio::test]
    async fn deberia_registrar_el_evento_producto_creado_cuando_los_campos_son_validos() {
        // Preparar
        let uow_factory = UnitOfWorkFactoryGrabadora::new();
        let caso_de_uso = CrearProductoImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearProducto {
                nombre: "example".to_string(),
                precio: 42,
                sku: "example".to_string(),
            })
            .await;

        // Comprobar
        let entidad = match resultado {
            Ok(entidad) => entidad,
            Err(e) => panic!("la creación falló: {e}"),
        };
        assert_eq!(entidad.nombre, "example");
        let grabado = uow_factory.grabado();
        assert!(grabado.confirmado);
        assert_eq!(grabado.guardados.len(), 1);
        assert_eq!(grabado.eventos.len(), 1);
        assert_eq!(grabado.eventos[0].tipo_evento(), "producto_creado");
        assert_eq!(grabado.eventos[0].id_agregado(), entidad.id.valor());
    }

    #[tokio::test]
    async fn no_deberia_hacer_commit_cuando_el_almacen_rechaza_un_duplicado() {
        // Preparar
        let uow_factory =
            UnitOfWorkFactoryGrabadora::fallando_en(FallarEn::Guardar, ErrorRepositorio::Duplicado);
        let caso_de_uso = CrearProductoImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearProducto {
                nombre: "example".to_string(),
                precio: 42,
                sku: "example".to_string(),
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.duplicado");
        let grabado = uow_factory.grabado();
        assert!(!grabado.confirmado);
        assert!(grabado.eventos.is_empty());
    }

    #[tokio::test]
    async fn deberia_devolver_error_de_repositorio_cuando_falla_el_commit() {
        // Preparar
        let uow_factory = UnitOfWorkFactoryGrabadora::fallando_en(
            FallarEn::Commit,
            ErrorRepositorio::NoDisponible,
        );
        let caso_de_uso = CrearProductoImpl {
            uow_factory: Arc::new(uow_factory.clone()),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearProducto {
                nombre: "example".to_string(),
                precio: 42,
                sku: "example".to_string(),
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "producto.repositorio");
        assert!(!uow_factory.grabado().confirmado);
    }

    #[tokio::test]
    async fn deberia_devolver_error_sin_registrar_nada_cuando_nombre_es_una_cadena_vacia() {
        // Preparar

        };
