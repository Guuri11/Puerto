services:
  postgres:
    image: postgres:16
    container_name: shop-postgres
    ports:
      - "${DB_HOST_PORT:-5432}:5432"
    environment:
      POSTGRES_USER: devuser
      POSTGRES_PASSWORD: password
      POSTGRES_DB: shop
    volumes:
      - postgres-data:/var/lib/postgresql/data
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U devuser -d shop"]
      interval: 5s
      timeout: 5s
      retries: 5

  redpanda:
    image: docker.redpanda.com/redpandadata/redpanda:v24.2.7
    container_name: shop-redpanda
    command:
      - redpanda
      - start
      - --mode
      - dev-container
      - --smp
      - "1"
      - --kafka-addr
      - PLAINTEXT://0.0.0.0:9092
      - --advertise-kafka-addr
      - PLAINTEXT://localhost:${KAFKA_HOST_PORT:-9092}
    ports:
      - "${KAFKA_HOST_PORT:-9092}:9092"
    volumes:
      - redpanda-data:/var/lib/redpanda/data
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "rpk cluster health | grep -q 'Healthy:.*true'"]
      interval: 5s
      timeout: 5s
      retries: 10

  keycloak:
    image: quay.io/keycloak/keycloak:26.8.0
    container_name: shop-keycloak
    # Imports infrastructure/keycloak/realm.json on the first start: the `web` client, the
    # `admin` role and a user holding it (`make auth/ui`).
    command: start-dev --import-realm
    ports:
      - "8180:8080"
    environment:
      KC_BOOTSTRAP_ADMIN_USERNAME: admin
      KC_BOOTSTRAP_ADMIN_PASSWORD: admin
      KC_HEALTH_ENABLED: "true"
    volumes:
      - ./infrastructure/keycloak:/opt/keycloak/data/import:ro
    restart: unless-stopped
    healthcheck:
      # The image has no curl: bash's /dev/tcp asks the management port.
      test: ["CMD-SHELL", "exec 3<>/dev/tcp/localhost/9000 && printf 'GET /health/ready HTTP/1.0\r\n\r\n' >&3 && grep -q UP <&3"]
      interval: 5s
      timeout: 5s
      retries: 30

volumes:
  postgres-data:
  redpanda-data:
