import type { PedidoProblem } from "../../../../domain/pedido/messages";
import { type Pedido, type PedidoDraft, parseEstado } from "../../../../domain/pedido/model";
import { textOf, textOrNull } from "../../../shared/forms";
import { FieldError } from "../../../shared/ui/field-error";

/** What the inputs hold: the text of every input, `checked` for a checkbox. */
export interface PedidoFormValues {
  importe: string;
  estado: string;
  nota: string;
}

/** The inputs for `pedido`, empty when there is none yet. */
export function toPedidoForm(pedido: Pedido | null): PedidoFormValues {
  if (pedido === null) {
    return {
      importe: "",
      estado: "",
      nota: "",
    };
  }
  return {
    importe: String(pedido.importe),
    estado: pedido.estado,
    nota: textOf(pedido.nota),
  };
}

export function toPedidoDraft(values: PedidoFormValues): PedidoDraft {
  return {
    importe: Number(values.importe),
    estado: parseEstado(values.estado),
    nota: textOrNull(values.nota),
  };
}

export function PedidoFormView({
  values,
  saving,
  problem,
  onChange,
  onSubmit,
}: {
  values: PedidoFormValues;
  saving: boolean;
  /** What the API said about the last submit: shown under its input. */
  problem: PedidoProblem | null;
  onChange: (values: PedidoFormValues) => void;
  onSubmit: () => void;
}) {
  return (
    <form
      className="space-y-2"
      onSubmit={(event) => {
        event.preventDefault();
        onSubmit();
      }}
    >
      <label className="fieldset">
        <span className="fieldset-legend">Importe</span>
        <input
          className="input w-full"
          type="number"
          aria-invalid={problem?.field === "importe"}
          step="any"
          required
          value={values.importe}
          onChange={(event) => onChange({ ...values, importe: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="importe" />
      <label className="fieldset">
        <span className="fieldset-legend">Estado</span>
        <select
          className="select w-full"
          aria-invalid={problem?.field === "estado"}
          required
          value={values.estado}
          onChange={(event) => onChange({ ...values, estado: event.target.value })}
        >
          <option value="">Choose…</option>
          <option value="Pendiente">Pendiente</option>
          <option value="Confirmado">Confirmado</option>
          <option value="Cancelado">Cancelado</option>
        </select>
      </label>
      <FieldError problem={problem} field="estado" />
      <label className="fieldset">
        <span className="fieldset-legend">Nota</span>
        <input
          className="input w-full"
          type="text"
          aria-invalid={problem?.field === "nota"}
          value={values.nota}
          onChange={(event) => onChange({ ...values, nota: event.target.value })}
        />
      </label>
      <FieldError problem={problem} field="nota" />
      <button type="submit" className="btn btn-primary" disabled={saving}>
        Save
      </button>
    </form>
  );
}
