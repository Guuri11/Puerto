use async_trait::async_trait;
use puerto_core::auth::Principal;

use crate::domain::order::{errors::OrderError, model::OrderId};

#[derive(Debug, Clone)]
pub struct DeleteOrderParams {
    /// Who asks (ADR-0021). A rule that depends on the data ("only the owner edits") decides on
    /// it here and answers `Forbidden`.
    pub actor: Principal,
    pub id: OrderId,
}

#[async_trait]
pub trait DeleteOrderUseCaseTrait: Send + Sync {
    async fn execute(&self, params: DeleteOrderParams) -> Result<(), OrderError>;
}

#[cfg(any(test, feature = "test-utils"))]
pub mod mocks {
    use mockall::mock;

    use super::*;

    mock! {
        pub DeleteOrderUseCase {}

        #[async_trait]
        impl DeleteOrderUseCaseTrait for DeleteOrderUseCase {
            async fn execute(&self, params: DeleteOrderParams) -> Result<(), OrderError>;
        }
    }
}
