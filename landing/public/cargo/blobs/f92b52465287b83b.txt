import { useState } from "react";
import { Link, useLocation } from "wouter";
import { desenvolver } from "../../../api/http";
import { useAuth } from "../../../auth/auth";
import { AlertaError } from "../../../compartido/ui/alerta-error";
import { DialogoConfirmar } from "../../../compartido/ui/dialogo-confirmar";
import { useCarga } from "../../../compartido/use-carga";
import { useDependencias } from "../../../dependencias";
import { VistaDetalleCliente } from "../vistas/detalle-cliente";

export function ContenedorDetalleCliente({ id }: { id: string }) {
  const { api } = useDependencias();
  const { puede } = useAuth();
  const cliente = useCarga(
    () => desenvolver(api.GET("/clientes/{id}", { params: { path: { id } } })),
    [api, id],
  );
  const [, navegar] = useLocation();
  const [confirmando, setConfirmando] = useState(false);
  const [fallo, setFallo] = useState<unknown>(null);

  const borrar = async () => {
    setConfirmando(false);
    try {
      await desenvolver(api.DELETE("/clientes/{id}", { params: { path: { id } } }));
      navegar("/clientes");
    } catch (error) {
      setFallo(error);
    }
  };

  return (
    <section className="space-y-4">
      <h1 className="text-2xl font-bold">Cliente</h1>
      {fallo !== null && <AlertaError error={fallo} />}
      {cliente.estado === "cargando" && <span className="loading loading-spinner" />}
      {cliente.estado === "error" && <AlertaError error={cliente.error} />}
      {cliente.estado === "listo" && (
        <div className="card bg-base-100">
          <div className="card-body">
            <VistaDetalleCliente cliente={cliente.datos} />
            <div className="card-actions justify-end">
              <Link href={`/clientes/${id}/editar`} className="btn">
                Editar
              </Link>
              {puede("admin") && (
                <button
                  type="button"
                  className="btn btn-error"
                  onClick={() => setConfirmando(true)}
                >
                  Borrar
                </button>
              )}
            </div>
          </div>
        </div>
      )}
      <Link href="/clientes" className="link">
        Volver a clientes
      </Link>
      <DialogoConfirmar
        abierto={confirmando}
        mensaje="¿Borrar cliente?"
        alConfirmar={borrar}
        alCancelar={() => setConfirmando(false)}
      />
    </section>
  );
}
