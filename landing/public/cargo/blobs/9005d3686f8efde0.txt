use std::collections::HashMap;
use std::sync::{PoisonError, RwLock, RwLockReadGuard, RwLockWriteGuard};

use async_trait::async_trait;
use dominio_cliente::{
    modelo::{Cliente, IdCliente},
    repositorio::RepositorioCliente,
};
use puerto_nucleo::{ErrorRepositorio, Pagina, PeticionPagina};

/// Guarda clientes en la memoria del proceso: se pierden cuando el proceso se para.
/// Como el adaptador de Postgres, oculta las filas borradas y lista primero las más recientes;
/// un valor único que ya tiene otra fila viva es `Duplicado`.
/// Cubre el hueco hasta que haya base de datos (`puerto nuevo --db` genera repositorios Postgres).
#[derive(Default)]
pub struct RepositorioClienteEnMemoria {
    filas: RwLock<HashMap<IdCliente, Cliente>>,
}

impl RepositorioClienteEnMemoria {
    pub fn new() -> Self {
        Self::default()
    }

    // Un cerrojo envenenado significa que un escritor hizo panic; cada escritura es un solo
    // insert, así que el mapa sigue entero y se usa tal cual.
    fn leer(&self) -> RwLockReadGuard<'_, HashMap<IdCliente, Cliente>> {
        self.filas.read().unwrap_or_else(PoisonError::into_inner)
    }

    fn escribir(&self) -> RwLockWriteGuard<'_, HashMap<IdCliente, Cliente>> {
        self.filas.write().unwrap_or_else(PoisonError::into_inner)
    }
}

#[async_trait]
impl RepositorioCliente for RepositorioClienteEnMemoria {
    async fn buscar_todos(
        &self,
        pagina: PeticionPagina,
    ) -> Result<Pagina<Cliente>, ErrorRepositorio> {
        tracing::debug!(
            limite = pagina.limite(),
            desplazamiento = pagina.desplazamiento(),
            "buscar_todos"
        );
        let filas = self.leer();
        let mut vivos: Vec<&Cliente> = filas.values().filter(|fila| !fila.borrado).collect();
        vivos.sort_by(|a, b| {
            b.creado_en
                .cmp(&a.creado_en)
                .then_with(|| a.id.valor().cmp(&b.id.valor()))
        });
        let total = vivos.len() as u64;
        let elementos = vivos
            .into_iter()
            .skip(pagina.desplazamiento() as usize)
            .take(pagina.limite() as usize)
            .cloned()
            .collect();
        Ok(Pagina::new(elementos, total, pagina))
    }

    async fn buscar_por_id(&self, id: IdCliente) -> Result<Option<Cliente>, ErrorRepositorio> {
        tracing::debug!(%id, "buscar_por_id");
        Ok(self.leer().get(&id).filter(|fila| !fila.borrado).cloned())
    }

    async fn guardar(&self, entidad: &Cliente) -> Result<(), ErrorRepositorio> {
        tracing::debug!(id = %entidad.id, "guardar");
        let mut filas = self.escribir();
        // Como los índices únicos parciales: solo las filas vivas ocupan sus valores.
        let ocupado = !entidad.borrado
            && filas
                .values()
                .any(|fila| fila.id != entidad.id && !fila.borrado && fila.email == entidad.email);
        if ocupado {
            tracing::warn!(id = %entidad.id, "guardar rechazado: un valor único ya está ocupado");
            return Err(ErrorRepositorio::Duplicado);
        }
        filas.insert(entidad.id, entidad.clone());
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_cliente::modelo::PropsCliente;
    use dominio_cliente::pruebas::madres::MadreCliente;
    use dominio_cliente::valores::Email;
    use uuid::Uuid;

    /// Una entidad válida, con los valores de la madre salvo en los campos únicos (uno
    /// nuevo cada vez) y las fechas (un segundo exacto).
    fn nuevo_cliente() -> Cliente {
        Cliente::new(PropsCliente {
            email: Email::new(Uuid::new_v4().to_string())
                .expect("Email::new rechazó el valor de ejemplo"),
            ..MadreCliente::new().construir_props()
        })
        .unwrap()
    }

    #[tokio::test]
    async fn deberia_encontrar_cliente_por_id_tras_guardar() {
        // Preparar
        let repositorio = RepositorioClienteEnMemoria::new();
        let entidad = nuevo_cliente();
        repositorio.guardar(&entidad).await.unwrap();

        // Actuar
        let encontrado = repositorio.buscar_por_id(entidad.id).await.unwrap();

        // Comprobar
        assert_eq!(encontrado.map(|fila| fila.id), Some(entidad.id));
    }

    #[tokio::test]
    async fn deberia_actualizar_la_fila_cuando_se_guarda_de_nuevo() {
        // Preparar
        let repositorio = RepositorioClienteEnMemoria::new();
        let mut entidad = nuevo_cliente();
        repositorio.guardar(&entidad).await.unwrap();
        entidad.actualizado_en = chrono::Utc::now();

        // Actuar
        repositorio.guardar(&entidad).await.unwrap();

        // Comprobar
        let pagina = repositorio
            .buscar_todos(PeticionPagina::default())
            .await
            .unwrap();
        assert_eq!(pagina.total, 1);
        assert_eq!(pagina.elementos[0].actualizado_en, entidad.actualizado_en);
    }

    #[tokio::test]
    async fn no_deberia_encontrar_cliente_tras_su_borrado() {
        // Preparar
        let repositorio = RepositorioClienteEnMemoria::new();
        let mut entidad = nuevo_cliente();
        entidad.borrado = true;
        entidad.borrado_en = Some(entidad.actualizado_en);
        repositorio.guardar(&entidad).await.unwrap();

        // Actuar
        let encontrado = repositorio.buscar_por_id(entidad.id).await.unwrap();
        let pagina = repositorio
            .buscar_todos(PeticionPagina::default())
            .await
            .unwrap();

        // Comprobar
        assert!(encontrado.is_none());
        assert_eq!(pagina.total, 0);
    }

    #[tokio::test]
    async fn deberia_devolver_la_pagina_pedida_de_una_lista_mayor() {
        // Preparar
        let repositorio = RepositorioClienteEnMemoria::new();
        for _ in 0..3 {
            repositorio.guardar(&nuevo_cliente()).await.unwrap();
        }

        // Actuar
        let pagina = repositorio
            .buscar_todos(PeticionPagina::new(Some(2), Some(2)))
            .await
            .unwrap();

        // Comprobar
        assert_eq!(pagina.elementos.len(), 1);
        assert_eq!(pagina.total, 3);
        assert_eq!((pagina.limite, pagina.desplazamiento), (2, 2));
    }

    #[tokio::test]
    async fn deberia_rechazar_un_duplicado_de_email() {
        // Preparar
        let repositorio = RepositorioClienteEnMemoria::new();
        let primero = nuevo_cliente();
        repositorio.guardar(&primero).await.unwrap();
        let mut segundo = nuevo_cliente();
        segundo.email = primero.email.clone();

        // Actuar
        let resultado = repositorio.guardar(&segundo).await;

        // Comprobar
        assert_eq!(resultado.unwrap_err(), ErrorRepositorio::Duplicado);
    }
}
