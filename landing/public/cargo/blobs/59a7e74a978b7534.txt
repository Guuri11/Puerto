use async_trait::async_trait;
use dominio_pedido::errores::ErrorPedido;
use dominio_pedido::modelo::{Pedido, PropsPedido};
use dominio_pedido::repositorio::RepositorioPedido;
use dominio_pedido::valores::{Estado, Importe};
use puerto_nucleo::auth::Principal;
use std::sync::Arc;

#[derive(Debug, Clone)]
pub struct ParamsCrearPedido {
    /// Quien lo pide (ADR-0021). Una regla que depende de los datos ("solo edita el propietario")
    /// se decide aquí, sobre él, y responde `Prohibido`.
    pub actor: Principal,
    pub importe: f64,
    pub estado: String,
    pub nota: Option<String>,
}

#[async_trait]
pub trait CrearPedido: Send + Sync {
    async fn ejecutar(&self, params: ParamsCrearPedido) -> Result<Pedido, ErrorPedido>;
}

#[cfg(any(test, feature = "pruebas"))]
pub mod dobles {
    use mockall::mock;

    use super::*;

    mock! {
        pub CrearPedido {}

        #[async_trait]
        impl CrearPedido for CrearPedido {
            async fn ejecutar(&self, params: ParamsCrearPedido) -> Result<Pedido, ErrorPedido>;
        }
    }
}

pub struct CrearPedidoImpl {
    pub repositorio: Arc<dyn RepositorioPedido>,
}

#[async_trait]
impl CrearPedido for CrearPedidoImpl {
    #[tracing::instrument(name = "crear_pedido", skip_all, err(level = "warn"))]
    async fn ejecutar(&self, params: ParamsCrearPedido) -> Result<Pedido, ErrorPedido> {
        let importe = Importe::new(params.importe)?;
        let estado = params.estado.parse::<Estado>()?;
        let entidad = Pedido::new(PropsPedido {
            importe,
            estado,
            nota: params.nota,
        })?;
        self.repositorio.guardar(&entidad).await?;
        Ok(entidad)
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use dominio_pedido::pruebas::madres::MadrePedido;
    use dominio_pedido::repositorio::dobles::MockRepositorioPedido;
    use puerto_nucleo::ErrorRepositorio;
    use puerto_nucleo::auth::dobles::MadrePrincipal;

    #[tokio::test]
    async fn deberia_crear_pedido_cuando_los_campos_son_validos() {
        // Preparar
        let props = MadrePedido::new().construir_props();
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble.expect_guardar().returning(|_| Ok(()));
        let caso_de_uso = CrearPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearPedido {
                actor: MadrePrincipal::aleatorio(),
                importe: props.importe.valor(),
                estado: props.estado.as_str().to_string(),
                nota: props.nota,
            })
            .await;

        // Comprobar
        assert!(resultado.is_ok());
        assert_eq!(
            resultado.unwrap().estado,
            MadrePedido::new().construir_props().estado
        );
    }

    #[tokio::test]
    async fn deberia_devolver_duplicado_cuando_el_almacen_rechaza_un_duplicado() {
        // Preparar
        let props = MadrePedido::new().construir_props();
        let mut repositorio_doble = MockRepositorioPedido::new();
        repositorio_doble
            .expect_guardar()
            .returning(|_| Err(ErrorRepositorio::Duplicado));
        let caso_de_uso = CrearPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearPedido {
                actor: MadrePrincipal::aleatorio(),
                importe: props.importe.valor(),
                estado: props.estado.as_str().to_string(),
                nota: props.nota,
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "pedido.duplicado");
    }

    #[tokio::test]
    async fn deberia_devolver_error_cuando_estado_es_una_cadena_vacia() {
        // Preparar
        let props = MadrePedido::new().construir_props();
        let repositorio_doble = MockRepositorioPedido::new();
        let caso_de_uso = CrearPedidoImpl {
            repositorio: Arc::new(repositorio_doble),
        };

        // Actuar
        let resultado = caso_de_uso
            .ejecutar(ParamsCrearPedido {
                actor: MadrePrincipal::aleatorio(),
                importe: props.importe.valor(),
                estado: "".to_string(),
                nota: props.nota,
            })
            .await;

        // Comprobar
        assert!(resultado.is_err());
        assert_eq!(resultado.unwrap_err().to_string(), "pedido.estado_invalido");
    }
}
