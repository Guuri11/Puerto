import { type ClienteApi, crearClienteApi } from "../api/cliente";

/** Lo que guarda la API falsa bajo `ruta`: los DTO de la API, por id. */
export interface Coleccion<T extends { id: string }> {
  ruta: string;
  /** El código que responde un elemento que no está: `producto.no_encontrado`. */
  noEncontrado: string;
  elementos: Map<string, T>;
  /** Los campos que `POST <ruta>/{id}/<accion>` pone en el elemento, por acción. */
  acciones: Record<string, Partial<T>>;
  /** Lo que `PUT <ruta>/{id}` conserva aunque no se mande: los campos generados (27.17). */
  conservados: string[];
}

/**
 * Los elementos bajo `ruta` (`/productos`) de `entidad` (`producto`), empezando por `semilla`, las
 * `acciones` que entiende (`{ cancelar: { estado: "Cancelado" } }`) y los campos que actualizar
 * `conservados` (`["codigo"]`).
 */
export function coleccion<T extends { id: string }>(
  ruta: string,
  entidad: string,
  semilla: T[] = [],
  acciones: Record<string, Partial<T>> = {},
  conservados: string[] = [],
): Coleccion<T> {
  const elementos = new Map(semilla.map((elemento) => [elemento.id, elemento]));
  return { ruta, noEncontrado: `${entidad}.no_encontrado`, elementos, acciones, conservados };
}

/** Un `ClienteApi` sobre {@link fetchFalso}. */
export function apiFalsa(
  colecciones: Coleccion<{ id: string }>[],
  fallos: Record<string, string> = {},
): ClienteApi {
  return crearClienteApi({
    baseUrl: "http://localhost/api",
    fetch: fetchFalso(colecciones, fallos),
  });
}

/**
 * Un `fetch` que responde como la API generada, sobre colecciones en memoria: una página por
 * `limite` / `desplazamiento`, un `404` con `<entidad>.no_encontrado`, `201`, `200` y `204`, y
 * `200` con el elemento cambiado a una acción. Cada uno de `fallos` responde su código en su lugar:
 * `{ "POST /productos": "producto.duplicado" }` (también a las acciones), o solo el de una acción,
 * `{ "POST /productos/{id}/publicar": "producto.validacion.ya_publicado" }`.
 */
export function fetchFalso(
  colecciones: Coleccion<{ id: string }>[],
  fallos: Record<string, string> = {},
) {
  return async (peticion: Request): Promise<Response> => {
    const url = new URL(peticion.url);
    const ruta = url.pathname.replace(/^\/api/, "");
    const objetivo = colecciones.find((c) => ruta === c.ruta || ruta.startsWith(`${c.ruta}/`));
    if (objetivo === undefined) return responderError(404, "http.404");
    const [id = "", accion] = ruta.slice(objetivo.ruta.length + 1).split("/");
    const fallo =
      fallos[`${peticion.method} ${objetivo.ruta}/{id}/${accion}`] ??
      fallos[`${peticion.method} ${objetivo.ruta}`];
    if (fallo !== undefined) return responderError(400, fallo);
    if (accion !== undefined) return servirAccion(objetivo, peticion, id, accion);
    return id === ""
      ? servirLista(objetivo, peticion, url)
      : servirElemento(objetivo, peticion, id);
  };
}

async function servirLista(objetivo: Coleccion<{ id: string }>, peticion: Request, url: URL) {
  if (peticion.method === "POST") {
    const creado = { ...(await peticion.json()), id: crypto.randomUUID() };
    objetivo.elementos.set(creado.id, creado);
    return Response.json(creado, { status: 201 });
  }
  const limite = Number(url.searchParams.get("limite") ?? 20);
  const desplazamiento = Number(url.searchParams.get("desplazamiento") ?? 0);
  const todos = [...objetivo.elementos.values()];
  const elementos = todos.slice(desplazamiento, desplazamiento + limite);
  return Response.json({ elementos, total: todos.length, limite, desplazamiento });
}

async function servirElemento(objetivo: Coleccion<{ id: string }>, peticion: Request, id: string) {
  const encontrado = objetivo.elementos.get(id);
  if (encontrado === undefined) return responderError(404, objetivo.noEncontrado);
  if (peticion.method === "DELETE") {
    objetivo.elementos.delete(id);
    return new Response(null, { status: 204 });
  }
  if (peticion.method === "PUT") {
    // Como la API, actualizar reemplaza todos los campos: el que no se manda queda vacío, salvo
    // los que la API pone ella (los generados), que se conservan.
    const previo: Record<string, unknown> = encontrado;
    const conservado = Object.fromEntries(
      objetivo.conservados.map((campo) => [campo, previo[campo]]),
    );
    const actualizado = { ...conservado, ...(await peticion.json()), id };
    objetivo.elementos.set(id, actualizado);
    return Response.json(actualizado);
  }
  return Response.json(encontrado);
}

function servirAccion(
  objetivo: Coleccion<{ id: string }>,
  peticion: Request,
  id: string,
  accion: string,
) {
  const cambios = objetivo.acciones[accion];
  if (peticion.method !== "POST" || cambios === undefined) return responderError(404, "http.404");
  const encontrado = objetivo.elementos.get(id);
  if (encontrado === undefined) return responderError(404, objetivo.noEncontrado);
  const cambiado = { ...encontrado, ...cambios };
  objetivo.elementos.set(id, cambiado);
  return Response.json(cambiado);
}

/** Un `RespuestaError` como lo responde la API. */
function responderError(estado: number, codigo: string) {
  return Response.json({ codigo, mensaje: codigo }, { status: estado });
}
