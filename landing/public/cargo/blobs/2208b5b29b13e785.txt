import { useState } from "react";
import { Link, useLocation } from "wouter";
import { useDependencies } from "../../../core/dependencies";
import { useAsync } from "../../../core/hooks/use-async";
import { ConfirmDialog } from "../../../shared/ui/confirm-dialog";
import { ErrorAlert } from "../../../shared/ui/error-alert";
import { ClienteDetailView } from "../views/cliente-detail-view";

export function ClienteDetailContainer({ id }: { id: string }) {
  const { clienteRepository } = useDependencies();
  const cliente = useAsync(() => clienteRepository.get(id), [clienteRepository, id]);
  const [, navigate] = useLocation();
  const [confirming, setConfirming] = useState(false);
  const [failure, setFailure] = useState<unknown>(null);

  const remove = async () => {
    setConfirming(false);
    try {
      await clienteRepository.remove(id);
      navigate("/clientes");
    } catch (error) {
      setFailure(error);
    }
  };

  return (
    <section className="space-y-4">
      <h1 className="text-2xl font-bold">Cliente</h1>
      {failure !== null && <ErrorAlert error={failure} />}
      {cliente.status === "loading" && <span className="loading loading-spinner" />}
      {cliente.status === "error" && <ErrorAlert error={cliente.error} />}
      {cliente.status === "ok" && (
        <div className="card bg-base-100">
          <div className="card-body">
            <ClienteDetailView cliente={cliente.data} />
            <div className="card-actions justify-end">
              <Link href={`/clientes/${id}/edit`} className="btn">
                Edit
              </Link>
              <button type="button" className="btn btn-error" onClick={() => setConfirming(true)}>
                Delete
              </button>
            </div>
          </div>
        </div>
      )}
      <Link href="/clientes" className="link">
        Back to clientes
      </Link>
      <ConfirmDialog
        open={confirming}
        message="Delete this cliente?"
        onConfirm={remove}
        onCancel={() => setConfirming(false)}
      />
    </section>
  );
}
