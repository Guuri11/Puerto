import { describe, expect, test } from "bun:test";
import { fireEvent, screen, waitFor } from "@testing-library/react";
import { PedidoError } from "../../../../domain/pedido/errors";
import type { PedidoDraft } from "../../../../domain/pedido/model";
import { fakePedidoRepository, PedidoMother } from "../../../../test-support/pedido";
import { renderScreen } from "../../../../test-support/render";
import { textOf } from "../../../shared/forms";
import { PedidoFormContainer } from "./pedido-form-container";

function type(label: string, value: string) {
  fireEvent.change(screen.getByLabelText(label), { target: { value } });
}

/** Fills the empty form with `draft`, the way a user would. */
async function fillIn(draft: PedidoDraft) {
  await screen.findByText("Save");
  type("Importe", String(draft.importe));
  type("Estado", draft.estado);
  type("Nota", textOf(draft.nota));
}

describe("PedidoFormContainer", () => {
  test("should create the pedido from what was typed and show it", async () => {
    // Arrange
    const draft = PedidoMother.draft();
    const pedidoRepository = fakePedidoRepository();
    const { history } = renderScreen(<PedidoFormContainer />, { pedidoRepository });
    await fillIn(draft);

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    await waitFor(() => expect(history.at(-1)).toStartWith("/pedidos/"));
    const {
      items: [created],
    } = await pedidoRepository.list({ limit: 1, offset: 0 });
    expect(created).toMatchObject(draft);
    expect(history.at(-1)).toBe(`/pedidos/${created?.id}`);
  });

  test("should save the edited pedido when an existing one is submitted", async () => {
    // Arrange
    const seeded = PedidoMother.random();
    const pedidoRepository = fakePedidoRepository([seeded]);
    const { history } = renderScreen(<PedidoFormContainer id={seeded.id} />, {
      pedidoRepository,
    });

    // Act
    fireEvent.click(await screen.findByText("Save"));

    // Assert
    await waitFor(() => expect(history.at(-1)).toBe(`/pedidos/${seeded.id}`));
    expect(await pedidoRepository.get(seeded.id)).toEqual(seeded);
  });

  test("should show the code itself when the API answers one the form does not know", async () => {
    // Arrange
    const code = "pedido.validacion.invalid";
    const pedidoRepository = {
      ...fakePedidoRepository(),
      create: () => Promise.reject(new PedidoError(code)),
    };
    renderScreen(<PedidoFormContainer />, { pedidoRepository });
    await fillIn(PedidoMother.draft());

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe(code);
  });

  test("should say a pedido with these values exists when the API answers a duplicate", async () => {
    // Arrange
    const pedidoRepository = {
      ...fakePedidoRepository(),
      create: () => Promise.reject(new PedidoError("pedido.duplicado")),
    };
    renderScreen(<PedidoFormContainer />, { pedidoRepository });
    await fillIn(PedidoMother.draft());

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    expect((await screen.findByRole("alert")).textContent).toBe(
      "A pedido with these values already exists",
    );
  });

  test("should show the message under its input when the API rejects one field", async () => {
    // Arrange
    const pedidoRepository = {
      ...fakePedidoRepository(),
      create: () => Promise.reject(new PedidoError("pedido.importe_invalido")),
    };
    renderScreen(<PedidoFormContainer />, { pedidoRepository });
    await fillIn(PedidoMother.draft());

    // Act
    fireEvent.click(screen.getByText("Save"));

    // Assert
    expect(await screen.findByText("Importe is not valid")).toBeTruthy();
    const input = screen.getByLabelText("Importe");
    expect(input.getAttribute("aria-invalid")).toBe("true");
    expect(screen.queryByRole("alert")).toBeNull();
  });
});
