import { Link } from "wouter";
import type { Esquemas } from "../../../api/cliente";

type Cliente = Esquemas["DtoCliente"];

export function VistaListaCliente({ clientes }: { clientes: Cliente[] }) {
  return (
    <div className="overflow-x-auto rounded-box bg-base-100">
      <table className="table">
        <thead>
          <tr>
            <th>Email</th>
            <th>Nombre</th>
            <th>Etiquetas</th>
            <th />
          </tr>
        </thead>
        <tbody>
          {clientes.map((cliente) => (
            <tr key={cliente.id}>
              <td>{cliente.email}</td>
              <td>{cliente.nombre}</td>
              <td>{cliente.etiquetas.join(", ")}</td>
              <td className="text-right">
                <Link href={`/clientes/${cliente.id}`} className="link">
                  Ver
                </Link>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
